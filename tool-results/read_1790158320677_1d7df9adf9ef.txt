   493→/** GET/PATCH /api/user/profile payload (route returns this shape). */
   494→type ProfilePayload = {
   495→  id: string;
   496→  name: string | null;
   497→  handle: string | null;
   498→  bio: string | null;
   499→  image: string | null;
   500→  role: string;

   504→const BIO_MAX = 200;
   505→
   506→/**
   507→ * Compact profile editor in a Popover anchored to the avatar chip.
   508→ * Opening (the parent owns `open`) pulls the full profile so bio and avatar
   509→ * are always current (fetch continuation, not an effect body, per the
   510→ * repo's setState-in-effect lint ban); Save → PATCH, then onSaved() lets the
   511→ * parent refresh the session so the header chip and the dropdown row pick up
   512→ * the new avatar/handle without a reload.
   513→ */
   514→function ProfileEditor({
   515→  open,
   516→  onOpenChange,
   517→  initial,
   518→  onSaved,
   519→}: {
   520→  open: boolean;
   521→  onOpenChange: (open: boolean) => void;
   522→  initial: { name: string; handle: string; email: string | null };
   523→  onSaved: () => void | Promise<void>;
   524→}) {
   525→  const { toast } = useToast();
   526→  const [saving, setSaving] = useState(false);
   527→  const [name, setName] = useState(initial.name);
   528→  const [handle, setHandle] = useState(initial.handle);
   529→  const [bio, setBio] = useState("");
   530→  const [image, setImage] = useState<string | null>(null);
   531→  const [error, setError] = useState<string | null>(null);
   532→
   533→  // Every open: pull the authoritative profile (bio and avatar are not on
   534→  // the session) and clear stale errors. All state writes happen in async
   535→  // continuations so react-hooks/set-state-in-effect stays satisfied.
   536→  useEffect(() => {
   537→    if (!open) return;
   538→    let alive = true;
   539→    setError(null);
   540→    fetch("/api/user/profile")
   541→      .then((r) =>
   542→        r.ok ? (r.json() as Promise<{ profile: ProfilePayload }>) : null
   543→      )
   544→      .then((d) => {
   545→        if (!alive || !d?.profile) return;
   546→        setName(d.profile.name || initial.name);
   547→        setHandle(d.profile.handle || initial.handle);
   548→        setBio(d.profile.bio ?? "");
   549→        setImage(d.profile.image ?? null);
   550→      })
   551→      .catch(() => {
   552→        /* fall back to the session-derived values already in the form */
   553→      });
   554→    return () => {
   555→      alive = false;
   556→    };
   557→  }, [open, initial.name, initial.handle]);
   558→
   559→  const save = async () => {
   560→    setSaving(true);
   561→    setError(null);
   562→    try {
   563→      const res = await fetch("/api/user/profile", {
   564→        method: "PATCH",
   565→        headers: { "Content-Type": "application/json" },
   566→        body: JSON.stringify({ name, handle, bio, image }),
   567→      });
   568→      const payload = (await res.json().catch(() => null)) as
   569→        | { profile?: ProfilePayload; error?: string }
   570→        | null;
   571→      if (!res.ok || !payload?.profile) {
   572→        throw new Error(
   573→          payload?.error ?? "Could not save the profile. Try again."
   574→        );
   575→      }
   576→      toast({ title: "Profile saved." });
   577→      await onSaved();
   578→      onOpenChange(false);
   579→    } catch (err) {
   580→      setError(
   581→        err instanceof Error ? err.message : "Could not save the profile."
   582→      );
   583→    } finally {
   584→      setSaving(false);
   585→    }
   586→  };
   587→
   588→  return (
   589→    <PopoverContent
   590→      align="end"
   591→      sideOffset={8}
   592→      className="z-[70] w-[340px] rounded-xl border border-white/10 bg-coal p-4 text-white shadow-xl shadow-black/40"
   593→    >
   594→      <p className={MONO_LABEL}>Profile</p>
   595→      <p className="mt-1 text-xs leading-snug text-white/60">
   596→        Signed in as {initial.email ?? "a maker"}.
   597→      </p>
   598→
   599→      <form
   600→        onSubmit={(e) => {
   601→          e.preventDefault();
   602→          if (!saving) void save();
   603→        }}
   604→        noValidate
   605→        className="mt-4 space-y-3"
   606→      >
   607→        <div className="flex items-start gap-3">
   608→          <ImageUploadField
   609→            value={image}
   610→            onChange={(url) => setImage(url)}
   611→            purpose="avatar"
   612→            endpoint="user"
   613→            square
   614→            label="Avatar"
   615→            hint="Square works best"
   616→            className="w-[116px] shrink-0"
   617→            disabled={saving}
   618→          />
   619→          <div className="min-w-0 flex-1 space-y-2">
   620→            <div>
   621→              <Label htmlFor="prother-profile-name" className={MONO_LABEL}>
   622→                Name
   623→              </Label>
   624→              <Input
   625→                id="prother-profile-name"
   626→                value={name}
   627→                onChange={(e) => setName(e.target.value)}
   628→                maxLength={40}
   629→                autoComplete="name"
   630→                disabled={saving}
   631→                className="mt-1 h-10 rounded-lg border-white/15 bg-white/[0.03] text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   632→                placeholder="Your name"
   633→              />
   634→            </div>
   635→            <div>
   636→              <Label htmlFor="prother-profile-handle" className={MONO_LABEL}>
   637→                Handle
   638→              </Label>
   639→              <div className="relative mt-1">
   640→                <span
   641→                  aria-hidden
   642→                  className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 font-mono text-sm text-white/55"
   643→                >
   644→                  @
   645→                </span>
   646→                <Input
   647→                  id="prother-profile-handle"
   648→                  value={handle}
   649→                  onChange={(e) =>
   650→                    setHandle(e.target.value.replace(/\s/g, ""))
   651→                  }
   652→                  maxLength={24}
   653→                  autoComplete="off"
   654→                  spellCheck={false}
   655→                  disabled={saving}
   656→                  className="h-10 rounded-lg border-white/15 bg-white/[0.03] pl-7 text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   657→                  placeholder="handle"
   658→                />
   659→              </div>
   660→            </div>
   661→          </div>
   662→        </div>
   663→
   664→        <div>
   665→          <div className="flex items-baseline justify-between gap-2">
   666→            <Label htmlFor="prother-profile-bio" className={MONO_LABEL}>
   667→              Bio
   668→            </Label>
   669→            <span
   670→              className={cn(
   671→                "font-mono text-xs tabular-nums",
   672→                bio.length >= BIO_MAX ? "text-ember" : "text-white/55"
   673→              )}
   674→            >
   675→              {bio.length}/{BIO_MAX}
   676→            </span>
   677→          </div>
   678→          <Textarea
   679→            id="prother-profile-bio"
   680→            value={bio}
   681→            onChange={(e) => setBio(e.target.value.slice(0, BIO_MAX))}
   682→            maxLength={BIO_MAX}
   683→            rows={3}
   684→            disabled={saving}
   685→            className="mt-1 rounded-lg border-white/15 bg-white/[0.03] text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   686→            placeholder="What are you building?"
   687→          />
   688→        </div>
   689→
   690→        {error && (
   691→          <p role="alert" className="text-xs leading-snug text-red-400">
   692→            {error}
   693→          </p>
   694→        )}
   695→
   696→        <Button
   697→          type="submit"
   698→          disabled={saving}
   699→          className="h-11 w-full rounded-lg bg-ember font-semibold text-[#0A0A0A] shadow-none hover:bg-ember/90"
   700→        >
   701→          {saving ? (
   702→            <>
   703→              <Loader2 className="size-4 animate-spin" aria-hidden />
   704→              Saving…
   705→            </>
   706→          ) : (
   707→            "Save profile"
   708→          )}
   709→        </Button>
   710→      </form>
   711→    </PopoverContent>
   712→  );
   713→}
