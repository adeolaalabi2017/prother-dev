  1473→type Delta = { text: string; dir: "up" | "down" | "flat"; title: string };
  1474→
  1475→/** Honest delta vs yesterday — no fabrication when the base is zero. */
  1476→function computeDelta(cur: number, prev: number): Delta {
  1477→  if (prev === 0 && cur === 0)
  1478→    return { text: "±0", dir: "flat", title: "No change vs yesterday" };
  1479→  if (prev === 0)
  1480→    return { text: "↑ new", dir: "up", title: `${cur} today · none yesterday` };
  1481→  const pct = Math.round(((cur - prev) / prev) * 100);
  1482→  if (pct === 0)
  1483→    return { text: "±0%", dir: "flat", title: `${cur} today vs ${prev} yesterday` };
  1484→  return pct > 0
  1485→    ? { text: `↑ ${pct}%`, dir: "up", title: `${cur} today vs ${prev} yesterday` }
  1486→    : { text: `↓ ${Math.abs(pct)}%`, dir: "down", title: `${cur} today vs ${prev} yesterday` };
  1487→}
  1488→
  1489→const DELTA_CX: Record<Delta["dir"], string> = {
  1490→  up: "bg-emerald-400/10 text-emerald-400/80",
  1491→  down: "bg-red-400/10 text-red-400/80",
  1492→  flat: "bg-white/5 text-white/60",
