    88→function useAdminKey() {
    89→  // Start locked on BOTH server and client first paint — reading
    90→  // sessionStorage during the hydration render would mismatch the SSR'd
    91→  // locked chip/gate (React 19 hydration error). The stored key is
    92→  // restored in a microtask right after mount (async, so the
    93→  // react-hooks/set-state-in-effect rule stays satisfied).
    94→  const [key, setKey] = useState<string | null>(null);
    95→  useEffect(() => {
    96→    Promise.resolve().then(() => {
    97→      const stored = sessionStorage.getItem(KEY_STORAGE);
    98→      if (stored) setKey(stored);
    99→    });
   100→  }, []);
   101→  const unlock = useCallback((k: string) => {
   102→    sessionStorage.setItem(KEY_STORAGE, k);
   103→    setKey(k);
   104→  }, []);
   105→  const lock = useCallback(() => {
   106→    sessionStorage.removeItem(KEY_STORAGE);
   107→    setKey(null);
   108→  }, []);
   109→  return { key, unlock, lock };
   110→}

   112→function adminFetch(key: string, url: string, init?: RequestInit) {
   113→  return fetch(url, {
   114→    ...init,
   115→    headers: {
   116→      ...(init?.body ? { "Content-Type": "application/json" } : {}),
   117→      "x-editor-key": key,
   118→      ...init?.headers,
   119→    },
   120→  }).then(async (r) => {
   121→    if (r.status === 401) throw new Error("401");
   122→    return r;
   123→  });
   124→}

   374→function QueueTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   375→  const { toast } = useToast();
   376→  const [subs, setSubs] = useState<PendingSub[] | null>(null);
   377→  const [rejecting, setRejecting] = useState<string | null>(null);
   378→  const [failed, setFailed] = useState<string[]>([]);
   379→  const [busy, setBusy] = useState<string | null>(null);
   380→
   381→  const load = useCallback(() => {
   382→    adminFetch(apiKey, "/api/editor/queue")
   383→      .then((r) => r.json())
   384→      .then((d: { pending?: PendingSub[] }) => setSubs(d.pending ?? []))
   385→      .catch(() => setSubs([]));
   386→  }, [apiKey]);
   387→
   388→  useEffect(load, [load]);
   389→
   390→  const decide = useCallback(
   391→    async (id: string, decision: "approve" | "reject") => {
   392→      setBusy(id);
   393→      try {
   394→        const res = await adminFetch(apiKey, "/api/editor/decision", {
   395→          method: "POST",
   396→          body: JSON.stringify(
   397→            decision === "approve"
   398→              ? { decision, id }
   399→              : { decision, id, failedStandards: failed, note: "" }
   400→          ),
   401→        });
   402→        const d = (await res.json()) as { ok?: boolean; error?: string };
   403→        if (!res.ok || !d.ok) {
   404→          toast({ title: d.error ?? "Decision failed", variant: "destructive" });
   405→          return;
   406→        }
   407→        toast({
   408→          title: decision === "approve" ? "Approved & scheduled" : "Rejected",
   409→          description:
   410→            decision === "approve"
   411→              ? "Launches tomorrow at 00:00 UTC."
   412→              : "Maker email names the failed standards.",
   413→        });
   414→        setRejecting(null);
   415→        setFailed([]);
   416→        load();
   417→        onChanged();
   418→      } catch {
   419→        toast({ title: "Network error", variant: "destructive" });
   420→      } finally {
   421→        setBusy(null);
   422→      }
   423→    },
   424→    [apiKey, failed, load, onChanged, toast]
   425→  );
   426→
   427→  if (subs === null) return <Spinner />;
   428→
   429→  return (
   430→    <div className="space-y-3">
   431→      {subs.length === 0 && (
   432→        <div className="rounded-xl border border-dashed border-white/15 bg-white/[0.02] p-10 text-center">
   433→          <CheckCircle2 className="mx-auto size-8 text-emerald-400" aria-hidden />
   434→          <p className="mt-3 font-mono text-sm text-white/60">
   435→            Queue clear — nothing waiting.
   436→          </p>
   437→        </div>
   438→      )}
   439→      {subs.map((s) => (
   440→        <div key={s.id} className="rounded-xl border border-white/10 bg-white/[0.02] p-4">
   441→          <div className="flex items-center gap-3">
   442→            <div
   443→              aria-hidden
   444→              className={cn(
   445→                "flex size-10 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br text-lg",
   446→                s.logoGradient
   447→              )}
   448→            >
   449→              {s.logoEmoji}
   450→            </div>
   451→            <div className="min-w-0 flex-1">
   452→              <p className="truncate text-sm font-bold text-white">{s.name}</p>
   453→              <p className="truncate text-xs text-white/50">{s.tagline}</p>
   454→            </div>
   455→            <span className="hidden shrink-0 font-mono text-[10px] text-white/40 sm:block">
   456→              {s.domain} · {s.ageH}h
   457→            </span>
   458→          </div>
   459→
   460→          {rejecting === s.id ? (
   461→            <div className="mt-3 space-y-2.5 rounded-lg border border-red-400/25 bg-red-400/[0.05] p-3">
   462→              <p className="font-mono text-[10px] tracking-widest text-red-300">
   463→                REJECTIONS MUST CITE FAILED STANDARD(S)
   464→              </p>
   465→              <div className="flex flex-wrap gap-1.5">
   466→                {STANDARD_DEFS.map((sd) => {
   467→                  const on = failed.includes(sd.id);
   468→                  return (
   469→                    <label
   470→                      key={sd.id}
   471→                      className={cn(
   472→                        "flex cursor-pointer items-center gap-1 rounded-full border px-2 py-0.5 font-mono text-[10px]",
   473→                        on ? "border-red-400 bg-red-400/15 text-red-200" : "border-white/15 text-white/60"
   474→                      )}
   475→                    >
   476→                      <Checkbox
   477→                        checked={on}
   478→                        onCheckedChange={(v) =>
   479→                          setFailed((p) => (v ? [...p, sd.id] : p.filter((x) => x !== sd.id)))
   480→                        }
   481→                        className="size-3 border-white/30 data-[state=checked]:border-red-400 data-[state=checked]:bg-red-400"
   482→                      />
   483→                      {sd.id}
   484→                    </label>
   485→                  );
   486→                })}
   487→              </div>
   488→              <div className="flex gap-2">
   489→                <Button
   490→                  size="sm"
   491→                  disabled={failed.length === 0 || busy === s.id}
   492→                  onClick={() => void decide(s.id, "reject")}
   493→                  className="rounded-lg bg-red-500 text-white shadow-none hover:bg-red-400 disabled:opacity-40"
   494→                >
   495→                  Reject ({failed.length})
   496→                </Button>
   497→                <Button
   498→                  size="sm"
   499→                  variant="ghost"
   500→                  onClick={() => setRejecting(null)}
   501→                  className="rounded-lg text-white/60 hover:text-white"
   502→                >
   503→                  Back
   504→                </Button>
   505→              </div>
   506→            </div>
   507→          ) : (
   508→            <div className="mt-3 flex gap-2">
   509→              <Button
   510→                size="sm"
   511→                disabled={busy === s.id}
   512→                onClick={() => void decide(s.id, "approve")}
   513→                className="rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   514→              >
   515→                {busy === s.id ? (
   516→                  <Loader2 className="size-3.5 animate-spin" aria-hidden />
   517→                ) : (
   518→                  <CalendarClock className="size-3.5" aria-hidden />
   519→                )}
   520→                Approve → tomorrow
   521→              </Button>
   522→              <Button
   523→                size="sm"
   524→                variant="outline"
   525→                onClick={() => setRejecting(s.id)}
   526→                className="rounded-lg border-red-400/30 text-red-400 hover:bg-red-400/10 hover:text-red-300"
   527→              >
   528→                Reject…
   529→              </Button>
   530→              <a
   531→                href={s.email ? `mailto:${s.email}` : "#"}
   532→                className="ml-auto hidden items-center gap-1 font-mono text-[10px] text-white/40 hover:text-white/70 sm:flex"
   533→              >
   534→                <Mail className="size-3" aria-hidden /> maker
   535→              </a>
   536→            </div>
   537→          )}
   538→        </div>
   539→      ))}
   540→    </div>
   541→  );
   542→}

   546→function CalendarTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   547→  const { toast } = useToast();
   548→  const [data, setData] = useState<ScheduleData | null>(null);
   549→  const [dates, setDates] = useState<Record<string, string>>({});
   550→  const [busy, setBusy] = useState<string | null>(null);
   551→
   552→  const load = useCallback(() => {
   553→    adminFetch(apiKey, "/api/admin/schedule")
   554→      .then((r) => r.json())
   555→      .then((d: ScheduleData) => setData(d))
   556→      .catch(() => toast({ title: "Calendar unavailable", variant: "destructive" }));
   557→  }, [apiKey, toast]);
   558→
   559→  useEffect(load, [load]);
   560→
   561→  const schedule = useCallback(
   562→    async (toolId: string, date: string | null, name: string) => {
   563→      setBusy(toolId);
   564→      try {
   565→        const res = await adminFetch(apiKey, "/api/admin/schedule", {
   566→          method: "POST",
   567→          body: JSON.stringify({ toolId, date }),
   568→        });
   569→        const d = (await res.json()) as { ok?: boolean; error?: string };
   570→        if (!res.ok || !d.ok) {
   571→          toast({ title: d.error ?? "Scheduling failed", variant: "destructive" });
   572→          return;
   573→        }
   574→        toast({ title: date ? `${name} → ${date}` : `${name} unscheduled` });
   575→        load();
   576→        onChanged();
   577→      } catch {
   578→        toast({ title: "Network error", variant: "destructive" });
   579→      } finally {
   580→        setBusy(null);
   581→      }
   582→    },
   583→    [apiKey, load, onChanged, toast]
   584→  );
   585→
   586→  if (!data) return <Spinner />;
   587→
   588→  const chip = (n: number) => {
   589→    if (n < data.floor) return { emoji: "🔴", label: "below floor" };
   590→    if (n >= 11) return { emoji: "🟡", label: "11–15" };
   591→    return { emoji: "🟢", label: `${data.floor}–10` };
   592→  };
   593→
   594→  return (
   595→    <div className="space-y-5">
   596→      <p className="font-mono text-[10px] tracking-wider text-white/35 uppercase">
   597→        Floor ≥{data.floor} · Cap ≤{data.cap} · today locked (live ranking)
   598→      </p>
   599→
   600→      {/* 14-day grid */}
   601→      <div className="max-h-72 space-y-1.5 overflow-y-auto pr-1">
   602→        {data.days.map((d) => {
   603→          const c = chip(d.count);
   604→          return (
   605→            <div
   606→              key={d.date}
   607→              className="rounded-xl border border-white/10 bg-white/[0.02] px-3 py-2.5"
   608→            >
   609→              <div className="flex items-center gap-2.5">

  1738→type Delta = { text: string; dir: "up" | "down" | "flat"; title: string };

  1740→/** Honest delta vs yesterday — no fabrication when the base is zero. */
  1741→function computeDelta(cur: number, prev: number): Delta {
  1742→  if (prev === 0 && cur === 0)
  1743→    return { text: "±0", dir: "flat", title: "No change vs yesterday" };
  1744→  if (prev === 0)
  1745→    return { text: "▲ new", dir: "up", title: `${cur} today · none yesterday` };
  1746→  const pct = Math.round(((cur - prev) / prev) * 100);
  1747→  if (pct === 0)
  1748→    return { text: "±0%", dir: "flat", title: `${cur} today vs ${prev} yesterday` };
  1749→  return pct > 0
  1750→    ? { text: `▲ ${pct}%`, dir: "up", title: `${cur} today vs ${prev} yesterday` }
  1751→    : { text: `▼ ${Math.abs(pct)}%`, dir: "down", title: `${cur} today vs ${prev} yesterday` };
  1752→}
