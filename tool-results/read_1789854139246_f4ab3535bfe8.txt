    51→function fmtDate(iso: string | null): string {
    52→  if (!iso) return "—";
    53→  return new Date(iso).toLocaleDateString("en-US", {
    54→    month: "short",
    55→    day: "numeric",
    56→    timeZone: "UTC",
    57→  });
    58→}

    97→function DetailBody({ slug, onClose }: { slug: string; onClose: () => void }) {
    98→  const { toast } = useToast();
    99→  const [detail, setDetail] = useState<ToolDetailResponse | null>(null);
   100→  const [loading, setLoading] = useState(true);
   101→  const [error, setError] = useState<string | null>(null);
   102→  const [vote, setVote] = useState<{ votes: number; voted: boolean } | null>(
   103→    null,
   104→  );
   105→  const [copied, setCopied] = useState(false);
   106→
   107→  useEffect(() => {
   108→    let alive = true;
   109→    fetch(`/api/tools/${slug}?vk=${encodeURIComponent(getVoterKey())}`)
   110→      .then(async (res) => {
   111→        if (!res.ok)
   112→          throw new Error(
   113→            res.status === 404 ? "Tool not found" : "Failed to load tool",
   114→          );
   115→        return (await res.json()) as ToolDetailResponse;
   116→      })
   117→      .then((data) => {
   118→        if (alive) {
   119→          setDetail(data);
   120→          setVote({ votes: data.votes, voted: data.voted });
   121→        }
   122→      })
   123→      .catch((err: unknown) => {
   124→        if (alive)
   125→          setError(err instanceof Error ? err.message : "Failed to load tool");
   126→      })
   127→      .finally(() => {
   128→        if (alive) setLoading(false);
   129→      });
   130→    return () => {
   131→      alive = false;
   132→    };
   133→  }, [slug]);
   134→
   135→  const onVote = useCallback(async () => {
   136→    if (!detail?.launchId || !vote) return;
   137→    const nextVoted = !vote.voted;
   138→    setVote({ votes: vote.votes + (nextVoted ? 1 : -1), voted: nextVoted });
   139→    try {
   140→      const res = await fetch("/api/vote", {
   141→        method: "POST",
   142→        headers: { "Content-Type": "application/json" },
   143→        body: JSON.stringify({
   144→          launchId: detail.launchId,
   145→          voterKey: getVoterKey(),
   146→        }),
   147→      });
   148→      const data = (await res.json()) as { voted: boolean; votes: number };
   149→      setVote({ votes: data.votes, voted: data.voted });
   150→      // Sync the feed list behind the modal.
   151→      window.dispatchEvent(
   152→        new CustomEvent("prother:vote", {
   153→          detail: { launchId: detail.launchId, votes: data.votes, voted: data.voted },
   154→        }),
   155→      );
   156→    } catch {
   157→      setVote(vote);
   158→      toast({
   159→        title: "Vote failed",
   160→        description: "Please try again.",
   161→        variant: "destructive",
   162→      });
   163→    }
   164→  }, [detail, vote, toast]);
   165→
   166→  const passedCount = detail?.standards.filter((s) => s.passed).length ?? 0;
   167→
   168→  const shareUrl =
   169→    typeof window !== "undefined"
   170→      ? `${window.location.origin}/${toolHash(slug)}`
   171→      : "";
   172→  const shareText = detail
   173→    ? `🚀 ${detail.name} — ${detail.tagline} is on Prother`
   174→    : "";
   175→
   176→  const onCopyLink = useCallback(async () => {
   177→    try {
   178→      await navigator.clipboard.writeText(shareUrl);
   179→      setCopied(true);
   180→      window.setTimeout(() => setCopied(false), 1600);
   181→    } catch {
   182→      toast({
   183→        title: "Could not copy the link",
   184→        description: shareUrl,
   185→        variant: "destructive",
   186→      });
   187→    }
   188→  }, [shareUrl, toast]);
   189→
   190→  return (
   191→    <>
   192→      {loading && <DetailSkeleton />}
   193→
   194→      {!loading && error && (
   195→        <div className="p-10 text-center">
   196→          <p className="text-sm text-white/60">{error}</p>
   197→          <button
   198→            type="button"
   199→            onClick={onClose}
   200→            className="mt-4 font-mono text-xs text-ember hover:underline"
   201→          >
   202→            ← Close
   203→          </button>
   204→        </div>
   205→      )}

   207→      {!loading && !error && detail && vote && (
   208→        <>
   209→          {/* Header with gradient banner */}

   211→            <div
   212→              aria-hidden

   215→                detail.gradient,

   220→                aria-hidden

   520→function CommandPalette() {
   521→  const { feed } = useFeed();
   522→  const searchOpen = useExplorer((s) => s.searchOpen);
   523→  const setSearch = useExplorer((s) => s.setSearch);
   524→  const openTool = useExplorer((s) => s.openTool);
   525→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   526→
   527→  const pickTool = useCallback(
   528→    (slug: string) => {
   529→      setSearch(false);
   530→      // Let the command dialog finish closing before opening the detail modal.
   531→      window.setTimeout(() => openTool(slug), 80);
   532→    },
   533→    [setSearch, openTool],
   534→  );
   535→
   536→  const goTo = useCallback(
   537→    (hash: string) => {
   538→      setSearch(false);
   539→      window.setTimeout(() => {
   540→        document.querySelector(hash)?.scrollIntoView({ behavior: "smooth" });
   541→      }, 80);
   542→    },
   543→    [setSearch],
   544→  );
   545→
   546→  const filterCategory = useCallback(
   547→    (slug: string) => {
   548→      setCategoryFilter(slug);
   549→      goTo("#feed");
   550→    },
   551→    [setCategoryFilter, goTo],
   552→  );
   553→
   554→  const seen = new Set<string>();
   555→  const todayTools = (feed?.top ?? []).filter((r) => {
   556→    if (seen.has(r.slug)) return false;
   557→    seen.add(r.slug);
   558→    return true;
   559→  });
   560→
   561→  return (
   562→    <CommandDialog
   563→      open={searchOpen}
   564→      onOpenChange={setSearch}
   565→      className="border-white/10 bg-coal text-white [&_[cmdk-group-heading]]:text-white/40 [&_[cmdk-input]]:text-white [&_[cmdk-input]::placeholder]:text-white/30 [&_[cmdk-item]]:text-white/80 [&_[cmdk-item][data-selected=true]]:bg-ember/15 [&_[cmdk-item][data-selected=true]]:text-ember [&_[cmdk-separator]]:bg-white/10"
   566→    >

   705→// ── Mounted-once explorer + ⌘K shortcut ──────────────────────────────────
   706→
   707→export function ToolExplorer() {
   708→  const setSearch = useExplorer((s) => s.setSearch);
   709→  const openTool = useExplorer((s) => s.openTool);
   710→
   711→  // ⌘K / ctrl+K opens the command palette.
   712→  useEffect(() => {
   713→    const onKey = (e: KeyboardEvent) => {
   714→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   715→        e.preventDefault();
   716→        setSearch(true);
   717→      }
   718→    };
   719→    window.addEventListener("keydown", onKey);
   720→    return () => window.removeEventListener("keydown", onKey);
   721→  }, [setSearch]);
   722→
   723→  // Shareable deep links: /#tool=<slug> opens the detail modal on load.
   724→  useEffect(() => {
   725→    const m = window.location.hash.match(/^#tool=([^&]+)$/);
   726→    if (m?.[1]) openTool(decodeURIComponent(m[1]));
   727→  }, [openTool]);
   728→
   729→  return (
   730→    <>
   731→      <CommandPalette />
   732→      <ToolDetailDialog />
   733→    </>
   734→  );
   735→}
