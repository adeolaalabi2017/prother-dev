    84→function urlOk(value: string): boolean {
    85→  try {
    86→    const u = new URL(value.trim());
    87→    return u.protocol === "http:" || u.protocol === "https:";
    88→  } catch {
    89→    return false;
    90→  }
    91→}

    93→const fmtCtr = (ctr: number) => `${ctr.toFixed(1)}%`;

    95→function fmtMoney(cents: number | null | undefined): string {
    96→  if (!cents || cents <= 0) return "no cap";
    97→  const dollars = cents / 100;
    98→  return `$${dollars % 1 === 0 ? dollars.toLocaleString("en-US") : dollars.toFixed(2)} cap`;
    99→}

   101→function fmtWindow(c: AdCampaign): string {
   102→  const s = c.startsAt ? c.startsAt.slice(0, 10) : "—";
   103→  const e = c.endsAt ? c.endsAt.slice(0, 10) : "—";
   104→  return `${s} → ${e}`;
   105→}

   160→function CampaignDialog({
   161→  apiKey,
   162→  campaign,

   172→  const isNew = !campaign;
   173→
   174→  const weightNum = Math.round(Number(f.weight));
   175→  const valid =
   176→    f.name.trim().length > 0 &&
   177→    f.advertiser.trim().length > 0 &&
   178→    f.headline.trim().length > 0 &&
   179→    urlOk(f.clickUrl) &&
   180→    weightNum >= 1 &&

   183→  const submit = async () => {
   184→    if (!valid || busy) return;
   185→    setBusy(true);
   186→    try {
   187→      const payload = {
   188→        name: f.name.trim(),
   189→        advertiser: f.advertiser.trim(),
   190→        placement: f.placement,

   192→        headline: f.headline.trim(),

   238→          <DialogDescription className="mt-1 text-sm text-white/50">
   239→            Sponsored slots across the feed, journal and category pages.

   247→                value={f.name}

   421→export function AdsTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   422→  const { toast } = useToast();
   423→  const [campaigns, setCampaigns] = useState<AdCampaign[] | null>(null);
   424→  const [stats, setStats] = useState<AdsStats | null>(null);
   425→  const [error, setError] = useState(false);
   426→  const [busy, setBusy] = useState<string | null>(null);
   427→  const [dialog, setDialog] = useState<{ open: boolean; campaign: AdCampaign | null }>({
   428→    open: false,
   429→    campaign: null,
   430→  });
   431→  const [confirmDelete, setConfirmDelete] = useState<string | null>(null);
   432→  const deleteTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   433→
   434→  const load = useCallback(() => {
   435→    adminFetch(apiKey, "/api/admin/ads")
   436→      .then((r) => {
   437→        if (!r.ok) throw new Error(String(r.status));
   438→        return r.json() as Promise<{ campaigns: AdCampaign[]; stats: AdsStats }>;
   439→      })
   440→      .then((d) => {
   441→        setCampaigns(d.campaigns);
   442→        setStats(d.stats);
   443→        setError(false);
   444→      })
   445→      .catch(() => setError(true));
   446→  }, [apiKey]);
   447→
   448→  useEffect(load, [load]);
   449→
   450→  const retry = useCallback(() => {
   451→    setCampaigns(null);
   452→    setError(false);
   453→    load();
   454→  }, [load]);
   455→
   456→  const patchStatus = useCallback(
   457→    async (c: AdCampaign, nextStatus: CampaignStatus, note?: string) => {
   458→      const snapshot = campaigns;
   459→      setCampaigns((cur) =>
   460→        cur?.map((x) => (x.id === c.id ? { ...x, status: nextStatus } : x)) ?? cur
   461→      );
   462→      setBusy(c.id);
   463→      try {
   464→        const res = await adminFetch(apiKey, `/api/admin/ads/${c.id}`, {
   465→          method: "PATCH",
   466→          body: JSON.stringify({ status: nextStatus }),
   467→        });
   468→        const d = (await res.json()) as { ok?: boolean; error?: string };
   469→        if (!res.ok || !d.ok) throw new Error(d.error ?? "Status update failed");
   470→        toast({ title: `${c.name} → ${nextStatus}`, description: note });
   471→        onChanged();
   472→      } catch (err) {
   473→        setCampaigns(snapshot);
   474→        toast({
   475→          title: err instanceof Error ? err.message : "Network error",
   476→          variant: "destructive",
   477→        });
   478→      } finally {
   479→        setBusy(null);
   480→      }
   481→    },
   482→    [apiKey, campaigns, toast, onChanged]
   483→  );
   484→
   485→  const doDelete = useCallback(
   486→    async (c: AdCampaign) => {
   487→      setBusy(c.id);
   488→      try {
   489→        const res = await adminFetch(apiKey, `/api/admin/ads/${c.id}`, { method: "DELETE" });
   490→        const d = (await res.json()) as { ok?: boolean; error?: string };
   491→        if (!res.ok || !d.ok) throw new Error(d.error ?? "Delete failed");
   492→        toast({ title: `${c.name} deleted` });
   493→        setCampaigns((cur) => cur?.filter((x) => x.id !== c.id) ?? cur);
   494→        onChanged();
   495→      } catch (err) {
   496→        toast({
   497→          title: err instanceof Error ? err.message : "Network error",
   498→          variant: "destructive",
   499→        });
   500→      } finally {
   501→        setBusy(null);
   502→      }
   503→    },
   504→    [apiKey, toast, onChanged]
   505→  );
   506→
   507→  const onDeleteClick = useCallback(
   508→    (c: AdCampaign) => {
   509→      if (confirmDelete === c.id) {
   510→        if (deleteTimer.current) clearTimeout(deleteTimer.current);
   511→        setConfirmDelete(null);
   512→        void doDelete(c);
   513→      } else {
   514→        setConfirmDelete(c.id);
   515→        if (deleteTimer.current) clearTimeout(deleteTimer.current);
   516→        deleteTimer.current = setTimeout(() => setConfirmDelete(null), 3000);
   517→      }
   518→    },
   519→    [confirmDelete, doDelete]
   520→  );
   521→
   522→  return (
   523→    <div className="space-y-3">
   524→      {/* stats strip */}
   525→      <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-4">
   526→        <MiniStat label="Active campaigns" value={stats ? `${stats.active}/${stats.total}` : "…"} />
   527→        <MiniStat label="Impressions" value={stats?.impressions ?? "…"} />
   528→        <MiniStat label="Clicks" value={stats?.clicks ?? "…"} />
   529→        <MiniStat label="Avg CTR" value={stats ? fmtCtr(stats.ctr) : "…"} accent />
   530→      </div>
   531→
   532→      <div className="flex items-center justify-between">
   533→        <p className="font-mono text-[10px] tracking-wider text-white/35 uppercase">
   534→          {campaigns ? `${campaigns.length} campaign${campaigns.length === 1 ? "" : "s"}` : "…"} ·
   535→          weight picks the slot winner
   536→        </p>
   537→        <Button
   538→          size="sm"
   539→          onClick={() => setDialog({ open: true, campaign: null })}
   540→          className="rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   541→        >
   542→          <Plus className="size-4" aria-hidden />
   543→          New campaign
   544→        </Button>
   545→      </div>
   546→
   547→      {error ? (
   548→        <LoadError label="Campaigns" onRetry={retry} />
   549→      ) : campaigns === null ? (
   550→        <CampaignsSkeleton />
   551→      ) : campaigns.length === 0 ? (
   552→        <div className="rounded-xl border border-dashed border-white/15 bg-white/[0.02] p-10 text-center">
   553→          <Megaphone className="mx-auto size-8 text-white/25" aria-hidden />
   554→          <p className="mt-3 font-mono text-sm text-white/60">
   555→            No campaigns yet — create the first one
   556→          </p>
   557→          <Button
   558→            onClick={() => setDialog({ open: true, campaign: null })}
   559→            className="mt-4 rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   560→          >
   561→            <Plus className="size-4" aria-hidden />
   562→            New campaign
   563→          </Button>
   564→        </div>
   565→      ) : (
   566→        <div className="max-h-[52vh] space-y-2 overflow-y-auto pr-1">
