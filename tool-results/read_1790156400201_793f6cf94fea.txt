   468→  name: string | null;

   470→  bio: string | null;

   476→const BIO_MAX = 200;
   477→
   478→/**
   479→ * Compact profile editor in a Popover anchored to the avatar chip.
   480→ * Opens → GET /api/user/profile fills the draft (async callback, not an
   481→ * effect, per the repo's setState-in-effect lint ban); Save → PATCH, then
   482→ * onSaved() lets the parent refresh the session so the header chip and the
   483→ * dropdown row pick up the new avatar/handle without a reload.
   484→ */
   485→function ProfileEditor({
   486→  onOpenChange,
   487→  initial,
   488→  onSaved,
   489→}: {
   490→  onOpenChange: (open: boolean) => void;
   491→  initial: { name: string; handle: string; email: string | null };
   492→  onSaved: () => void | Promise<void>;
   493→}) {
   494→  const { toast } = useToast();
   495→  const [loading, setLoading] = useState(false);
   496→  const [saving, setSaving] = useState(false);
   497→  const [name, setName] = useState(initial.name);
   498→  const [handle, setHandle] = useState(initial.handle);
   499→  const [bio, setBio] = useState("");
   500→  const [image, setImage] = useState<string | null>(null);
   501→  const [error, setError] = useState<string | null>(null);
   502→
   503→  const handleOpenChange = (next: boolean) => {
   504→    onOpenChange(next);
   505→    if (!next) return;
   506→    // Fresh open: clear stale errors and pull the full profile (bio and
   507→    // avatar are not on the session).
   508→    setError(null);
   509→    setLoading(true);
   510→    fetch("/api/user/profile")
   511→      .then((r) =>
   512→        r.ok ? (r.json() as Promise<{ profile: ProfilePayload }>) : null
   513→      )
   514→      .then((d) => {
   515→        if (d?.profile) {
   516→          setName(d.profile.name || initial.name);
   517→          setHandle(d.profile.handle || initial.handle);
   518→          setBio(d.profile.bio ?? "");
   519→          setImage(d.profile.image ?? null);
   520→        }
   521→      })
   522→      .catch(() => {
   523→        /* fall back to the session-derived initials below */
   524→      })
   525→      .finally(() => setLoading(false));
   526→  };
   527→
   528→  const save = async () => {
   529→    setSaving(true);
   530→    setError(null);
   531→    try {
   532→      const res = await fetch("/api/user/profile", {
   533→        method: "PATCH",
   534→        headers: { "Content-Type": "application/json" },
   535→        body: JSON.stringify({ name, handle, bio, image }),
   536→      });
   537→      const payload = (await res.json().catch(() => null)) as
   538→        | { profile?: ProfilePayload; error?: string }
   539→        | null;
   540→      if (!res.ok || !payload?.profile) {
   541→        throw new Error(
   542→          payload?.error ?? "Could not save the profile. Try again."
   543→        );
   544→      }
   545→      toast({ title: "Profile saved." });
   546→      await onSaved();
   547→      onOpenChange(false);
   548→    } catch (err) {
   549→      setError(
   550→        err instanceof Error ? err.message : "Could not save the profile."
   551→      );
   552→    } finally {
   553→      setSaving(false);
   554→    }
   555→  };
   556→
   557→  return (
   558→    <PopoverContent
   559→      align="end"
   560→      sideOffset={8}
   561→      className="z-[70] w-[340px] rounded-xl border border-white/10 bg-coal p-4 text-white shadow-xl shadow-black/40"
   562→    >
   563→      <p className={MONO_LABEL}>Profile</p>
   564→      <p className="mt-1 text-xs leading-snug text-white/60">
   565→        Signed in as {initial.email ?? "a maker"}.
   566→      </p>
   567→
   568→      <form
   569→        onSubmit={(e) => {
   570→          e.preventDefault();
   571→          if (!saving) void save();
   572→        }}
   573→        noValidate
   574→        className="mt-4 space-y-3"
   575→      >
   576→        <div className="flex items-start gap-3">
   577→          <ImageUploadField
   578→            value={image}
   579→            onChange={(url) => setImage(url)}
   580→            purpose="avatar"
   581→            endpoint="user"
   582→            square
   583→            label="Avatar"
   584→            hint="Square works best"
   585→            className="w-[116px] shrink-0"
   586→            disabled={loading || saving}
   587→          />
   588→          <div className="min-w-0 flex-1 space-y-2">
   589→            <div>
   590→              <Label htmlFor="prother-profile-name" className={MONO_LABEL}>
   591→                Name
   592→              </Label>
   593→              <Input
   594→                id="prother-profile-name"
   595→                value={name}
   596→                onChange={(e) => setName(e.target.value)}
   597→                maxLength={40}
   598→                autoComplete="name"
   599→                disabled={loading || saving}
   600→                className="mt-1 h-10 rounded-lg border-white/15 bg-white/[0.03] text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   601→                placeholder="Your name"
   602→              />
   603→            </div>
   604→            <div>
   605→              <Label htmlFor="prother-profile-handle" className={MONO_LABEL}>
   606→                Handle
   607→              </Label>
   608→              <div className="relative mt-1">
   609→                <span
   610→                  aria-hidden
   611→                  className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 font-mono text-sm text-white/55"
   612→                >
   613→                  @
   614→                </span>
   615→                <Input
   616→                  id="prother-profile-handle"
   617→                  value={handle}
   618→                  onChange={(e) =>
   619→                    setHandle(e.target.value.replace(/\s/g, ""))
   620→                  }
   621→                  maxLength={24}
   622→                  autoComplete="off"
   623→                  spellCheck={false}
   624→                  disabled={loading || saving}
   625→                  className="h-10 rounded-lg border-white/15 bg-white/[0.03] pl-7 text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   626→                  placeholder="handle"
   627→                />
   628→              </div>
   629→            </div>
   630→          </div>
   631→        </div>
   632→
   633→        <div>
   634→          <div className="flex items-baseline justify-between gap-2">
   635→            <Label htmlFor="prother-profile-bio" className={MONO_LABEL}>
   636→              Bio
   637→            </Label>
   638→            <span
   639→              className={cn(
   640→                "font-mono text-xs tabular-nums",
   641→                bio.length >= BIO_MAX ? "text-ember" : "text-white/55"
   642→              )}
   643→            >
   644→              {bio.length}/{BIO_MAX}
   645→            </span>
   646→          </div>
   647→          <Textarea
   648→            id="prother-profile-bio"
   649→            value={bio}
   650→            onChange={(e) => setBio(e.target.value.slice(0, BIO_MAX))}
   651→            maxLength={BIO_MAX}
   652→            rows={3}
   653→            disabled={loading || saving}
   654→            className="mt-1 rounded-lg border-white/15 bg-white/[0.03] text-sm text-white placeholder:text-white/55 focus-visible:border-ember/50 focus-visible:ring-ember/20"
   655→            placeholder="What are you building?"
   656→          />
   657→        </div>
   658→
   659→        {error && (
   660→          <p role="alert" className="text-xs leading-snug text-red-400">
   661→            {error}
   662→          </p>
   663→        )}
   664→
   665→        <Button
   666→          type="submit"
   667→          disabled={loading || saving}
   668→          className="h-11 w-full rounded-lg bg-ember font-semibold text-[#0A0A0A] shadow-none hover:bg-ember/90"
   669→        >
   670→          {saving ? (
   671→            <>
   672→              <Loader2 className="size-4 animate-spin" aria-hidden />
   673→              Saving…
   674→            </>
   675→          ) : (
   676→            "Save profile"
   677→          )}
   678→        </Button>
   679→      </form>
   680→    </PopoverContent>
   681→  );
   682→}
