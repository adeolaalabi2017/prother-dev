    20→function formatCountdown(totalSec: number): string {
    21→  const s = Math.max(0, totalSec);
    22→  const hh = String(Math.floor(s / 3600)).padStart(2, "0");
    23→  const mm = String(Math.floor((s % 3600) / 60)).padStart(2, "0");
    24→  const ss = String(s % 60).padStart(2, "0");
    25→  return `${hh}:${mm}:${ss}`;
    26→}

    28→function pricingLabel(row: FeedRow): string {
    29→  const { model, price } = row.pricing;
    30→  switch (model) {
    31→    case "free":
    32→      return "Free";
    33→    case "freemium":
    34→      return `Freemium ${price ?? ""}`.trim();
    35→    case "paid":
    36→      return `Paid ${price ?? ""}`.trim();
    37→    case "open_source":
    38→      return "Open Source";
    39→    default:
    40→      return "Free";
    41→  }
    42→}

   371→export function LaunchFeed() {
   372→  const { toast } = useToast();
   373→  const { feed, loading, error, refresh } = useFeed();
   374→  const openTool = useExplorer((s) => s.openTool);
   375→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   376→  const categoryFilter = useExplorer((s) => s.categoryFilter);
   377→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   378→  // ?tab= persistence — shareable feed states (new | top | tomorrow | yesterday).
   379→  // Read AFTER mount only: reading window.location during the initial render makes
   380→  // the server HTML ("new") and the hydrated client (?tab=top) diverge, which
   381→  // React reports as a hydration mismatch. Post-mount sync keeps SSR markup valid.
   382→  const [tab, setTab] = useState<Tab>("new");
   383→  useEffect(() => {
   384→    const t = new URLSearchParams(window.location.search).get("tab");
   385→    if (t === "top" || t === "tomorrow" || t === "yesterday") setTab(t);
   386→  }, []);
   387→
   388→  // Archive day browser: null = yesterday (served from the main feed payload),
   389→  // otherwise an ISO day from the strip — fetched once, then cached.
   390→  const [archiveDate, setArchiveDate] = useState<string | null>(null);
   391→  const [dayCache, setDayCache] = useState<Record<string, DayArchiveResponse>>({});
   392→  const [dayLoading, setDayLoading] = useState(false);
   393→
   394→  // Fetch (and cache) a selected archive day's final standings.
   395→  useEffect(() => {
   396→    if (!archiveDate || dayCache[archiveDate]) return;
   397→    let alive = true;
   398→    setDayLoading(true);
   399→    fetch(`/api/feed/day?date=${encodeURIComponent(archiveDate)}`)
   400→      .then(async (res) => {
   401→        if (!res.ok) throw new Error("Failed to load this archive day");
   402→        return (await res.json()) as DayArchiveResponse;
   403→      })
   404→      .then((data) => {
   405→        if (alive) setDayCache((c) => ({ ...c, [data.date]: data }));
   406→      })
   407→      .catch(() => {
   408→        if (alive) setArchiveDate(null); // gracefully fall back to yesterday
   409→      })
   410→      .finally(() => {
   411→        if (alive) setDayLoading(false);
   412→      });
   413→    return () => {
   414→      alive = false;
   415→    };
   416→  }, [archiveDate, dayCache]);
   417→
   418→  // UTC rollover shifts the whole archive window — drop cached days + selection.
   419→  useEffect(() => {
   420→    setDayCache({});
   421→    setArchiveDate(null);
   422→  }, [feed?.date]);
   423→
   424→  // Keep the selected tab in the URL (replaceState — no history spam).
   425→  const setTabSync = useCallback((next: Tab) => {
   426→    setTab(next);
   427→    if (typeof window === "undefined") return;
   428→    const url = new URL(window.location.href);
   429→    const prev = url.searchParams.get("tab");
   430→    if (next === "new") {
   431→      if (prev) {
   432→        url.searchParams.delete("tab");
   433→        window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   434→      }
   435→    } else if (prev !== next) {
   436→      url.searchParams.set("tab", next);
   437→      window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   438→    }
   439→  }, []);
   440→  const [now, setNow] = useState<number>(() => Date.now());
   441→  const [voteState, setVoteState] = useState<VoteState>({});
   442→
   443→  // Countdown deadline — computed from the feed at render (no effect-synced state).
   444→  const deadlineRef = useRef<{ date: string | null; at: number }>({ date: null, at: 0 });
   445→  if (feed && deadlineRef.current.date !== feed.date) {
   446→    deadlineRef.current = { date: feed.date, at: Date.now() + feed.resetsInSec * 1000 };
   447→  }
   448→  const deadline = deadlineRef.current.at;
   449→
   450→  // Tick the clock every second (setState inside the interval callback is fine).
   451→  useEffect(() => {
   452→    const iv = setInterval(() => setNow(Date.now()), 1000);
   453→    return () => clearInterval(iv);
   454→  }, []);
   455→
   456→  // Votes cast inside the tool detail modal sync back into the feed list.
   457→  useEffect(() => {
   458→    const onModalVote = (e: Event) => {
   459→      const d = (e as CustomEvent).detail as {
   460→        launchId: string;
   461→        votes: number;
   462→        voted: boolean;
   463→      };
   464→      setVoteState((prev) => ({
   465→        ...prev,
   466→        [d.launchId]: { votes: d.votes, voted: d.voted },
   467→      }));
   468→    };
   469→    window.addEventListener("prother:vote", onModalVote);
   470→    return () => window.removeEventListener("prother:vote", onModalVote);
   471→  }, []);
   472→
   473→  const remaining = feed ? Math.max(0, Math.round((deadline - now) / 1000)) : null;
   474→
   475→  // UTC-midnight rollover: refetch the feed exactly once when the countdown hits zero.
   476→  const rolledRef = useRef(false);
   477→  useEffect(() => {
   478→    if (remaining !== null && remaining > 0) {
   479→      rolledRef.current = false;
   480→    } else if (remaining === 0 && !rolledRef.current) {
   481→      rolledRef.current = true;
   482→      void refresh();
   483→    }
   484→  }, [remaining, refresh]);
   485→
   486→  const onVote = useCallback(
   487→    async (row: FeedRow) => {
   488→      const current = voteState[row.launchId] ?? { votes: row.votes, voted: row.voted };
   489→      const nextVoted = !current.voted;
   490→      const nextVotes = current.votes + (nextVoted ? 1 : -1);
   491→      setVoteState((prev) => ({
   492→        ...prev,
   493→        [row.launchId]: { votes: nextVotes, voted: nextVoted },
   494→      }));
   495→      try {
   496→        const res = await fetch("/api/vote", {
   497→          method: "POST",
   498→          headers: { "Content-Type": "application/json" },
   499→          body: JSON.stringify({ launchId: row.launchId, voterKey: getVoterKey() }),
   500→        });
   501→        const data = (await res.json()) as { voted: boolean; votes: number };
   502→        setVoteState((prev) => ({
   503→          ...prev,
   504→          [row.launchId]: { votes: data.votes, voted: data.voted },
   505→        }));
   506→      } catch {
   507→        setVoteState((prev) => ({ ...prev, [row.launchId]: current }));
   508→        toast({ title: "Vote failed", description: "Please try again.", variant: "destructive" });
   509→      }
   510→    },
   511→    [voteState, toast]
   512→  );
   513→
   514→  const rows = useMemo<FeedRow[]>(() => {
   515→    if (!feed) return [];
   516→    const base =
   517→      tab === "top"
   518→        ? feed.top ?? []
   519→        : tab === "yesterday"
   520→          ? archiveDate
   521→            ? dayCache[archiveDate]?.rows ?? []
   522→            : feed.yesterday ?? []
   523→          : feed.new ?? [];
   524→    if (!categoryFilter) return base;
   525→    return base.filter((r) => r.category?.slug === categoryFilter);
   526→  }, [feed, tab, categoryFilter, archiveDate, dayCache]);
   527→
   528→  const tomorrowRows = useMemo(() => {
   529→    if (!feed) return [];
   530→    if (!categoryFilter) return feed.tomorrow ?? [];
   531→    return (feed.tomorrow ?? []).filter((t) => t.category?.slug === categoryFilter);
   532→  }, [feed, categoryFilter]);
   533→
