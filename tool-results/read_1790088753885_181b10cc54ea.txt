    90→function ToolPicker({
    91→  label,
    92→  current,
    93→  excludeSlug,
    94→  rows,
    95→  onPick,
    96→  onClear,
    97→}: {
    98→  label: string;
    99→  current: DirectoryRow | null;
   100→  excludeSlug: string | null;
   101→  rows: DirectoryRow[];
   102→  onPick: (slug: string) => void;
   103→  onClear: () => void;
   104→}) {
   105→  const [open, setOpen] = useState(false);
   106→  const [q, setQ] = useState("");
   107→
   108→  const filtered = useMemo(() => {
   109→    const needle = q.trim().toLowerCase();
   110→    return rows
   111→      .filter((r) => r.slug !== excludeSlug)
   112→      .filter(
   113→        (r) =>
   114→          needle === "" ||
   115→          r.name.toLowerCase().includes(needle) ||
   116→          r.slug.includes(needle) ||
   117→          r.tagline.toLowerCase().includes(needle)
   118→      )
   119→      .slice(0, 40);
   120→  }, [rows, excludeSlug, q]);
   121→
   122→  return (
   123→    <div className="flex min-w-0 items-center gap-3">
   124→      <span className={cn(MONO, "w-14 shrink-0")}>{label}</span>
   125→      <Popover
   126→        open={open}
   127→        onOpenChange={(next) => {
   128→          setOpen(next);
   129→          if (next) setQ("");
   130→        }}
   131→      >
   132→        <PopoverTrigger asChild>
   133→          <button
   134→            type="button"
   135→            aria-label={current ? `Change tool ${label}` : `Pick tool for ${label}`}
   136→            className={cn(
   137→              "flex h-11 min-w-0 flex-1 items-center gap-2.5 rounded-xl border px-3 text-left transition-colors",
   138→              current
   139→                ? "border-white/10 bg-white/[0.02] hover:border-ember/40"
   140→                : "border-dashed border-white/20 text-white/40 hover:border-ember/40 hover:text-white/70"
   141→            )}
   142→          >
   143→            {current ? (
   144→              <>
   145→                <span
   146→                  aria-hidden
   147→                  className={cn(
   148→                    "grid size-7 shrink-0 place-items-center rounded-md bg-gradient-to-br text-sm",
   149→                    current.gradient
   150→                  )}
   151→                >
   152→                  {current.emoji}
   153→                </span>
   154→                <span className="min-w-0 flex-1 truncate text-sm font-semibold text-white/90">
   155→                  {current.name}
   156→                </span>
   157→                <span className="shrink-0 font-mono text-xs tabular-nums text-ember">▲{current.votes}</span>
   158→              </>
   159→            ) : (
   160→              <span className="flex items-center gap-2 text-sm">
   161→                <Search className="size-4" aria-hidden /> Pick a tool…
   162→              </span>
   163→            )}
   164→          </button>
   165→        </PopoverTrigger>
   166→        <PopoverContent
   167→          align="start"
   168→          sideOffset={8}
   169→          className="w-80 border-white/10 bg-coal p-0 text-white"
   170→          aria-label={`Choose ${label}`}
   171→        >
   172→          <div className="border-b border-white/10 p-2">
   173→            <Input
   174→              value={q}
   175→              onChange={(e) => setQ(e.target.value.slice(0, 60))}
   176→              placeholder="Filter tools…"
   177→              aria-label={`Filter tools for ${label}`}
   178→              className="h-9 border-white/10 bg-transparent text-sm focus-visible:border-ember/50 focus-visible:ring-0"
   179→            />
   180→          </div>
   181→          <div className="max-h-64 overflow-y-auto p-1.5">
   182→            {filtered.map((r) => (
   183→              <button
   184→                key={r.slug}
   185→                type="button"
   186→                onClick={() => {
   187→                  onPick(r.slug);
   188→                  setOpen(false);
   189→                }}
   190→                className="flex w-full items-center gap-2.5 rounded-lg px-2 py-2 text-left transition-colors hover:bg-ember/10"
   191→              >
   192→                <span
   193→                  aria-hidden
   194→                  className={cn(
   195→                    "grid size-7 shrink-0 place-items-center rounded-md bg-gradient-to-br text-sm",
   196→                    r.gradient
   197→                  )}
   198→                >
   199→                  {r.emoji}
   200→                </span>
   201→                <span className="min-w-0 flex-1">
   202→                  <span className="block truncate text-sm font-semibold text-white/90">{r.name}</span>
   203→                  <span className="block truncate text-xs text-white/40">{r.tagline}</span>
   204→                </span>
   205→                <span className="shrink-0 font-mono text-xs tabular-nums text-ember">▲{r.votes}</span>
   206→              </button>
   207→            ))}
   208→            {filtered.length === 0 && (
   209→              <p className="px-3 py-6 text-center text-xs text-white/40">
   210→                No tools match “{q}”.
   211→              </p>
   212→            )}
   213→          </div>
   214→        </PopoverContent>
   215→      </Popover>
   216→      {current && (
   217→        <Button
   218→          type="button"
   219→          variant="outline"
   220→          size="icon"
   221→          aria-label={`Clear ${label}`}
   222→          onClick={onClear}
   223→          className="bg-white/[0.03] hover:bg-white/[0.08] size-11 shrink-0 border-white/10 text-white/50 hover:border-ember/40 hover:text-white"
   224→        >
   225→          <X className="size-4" aria-hidden />
   226→        </Button>
   227→      )}
   228→    </div>
   229→  );
   230→}

   284→export function CompareFullPage() {
   285→  const compareOpen = useExplorer((s) => s.compareOpen);
   286→  const compare = useExplorer((s) => s.compare);
   287→  const closeCompare = useExplorer((s) => s.closeCompare);
   288→  const openCompare = useExplorer((s) => s.openCompare);
   289→  const addCompare = useExplorer((s) => s.addCompare);
   290→  const removeCompare = useExplorer((s) => s.removeCompare);
   291→  const openCategory = useExplorer((s) => s.openCategory);
   292→
   293→  const [rows, setRows] = useState<DirectoryRow[]>([]);
   294→  const [directoryLoaded, setDirectoryLoaded] = useState(false);
   295→  const [data, setData] = useState<CompareResponse | null>(null);
   296→  /** Slug pair the stored head-to-head data was fetched for — guards staleness. */
   297→  const [dataPair, setDataPair] = useState<string[]>([]);
   298→  const [popular, setPopular] = useState<PopularPair[]>([]);
   299→  const [loading, setLoading] = useState(false);
   300→  const [loadErr, setLoadErr] = useState(false);
   301→
   302→  const a = compare[0] ?? null;
   303→  const b = compare[1] ?? null;
   304→
   305→  // Directory list powers both pickers (client-side filtering).
   306→  useEffect(() => {
   307→    if (!compareOpen || directoryLoaded) return;
   308→    let alive = true;
   309→    fetch("/api/tools?limit=60&sort=votes")
   310→      .then(async (res) => {
   311→        if (!res.ok) throw new Error("failed");
   312→        return (await res.json()) as { rows: DirectoryRow[] };
   313→      })
   314→      .then((json) => {
   315→        if (!alive) return;
   316→        setRows(json.rows ?? []);
   317→        setDirectoryLoaded(true);
   318→      })
   319→      .catch(() => {
   320→        /* pickers stay empty; the page still renders chips/slugs */
   321→        if (alive) setDirectoryLoaded(true);
   322→      });
   323→    return () => {
   324→      alive = false;
   325→    };
   326→  }, [compareOpen, directoryLoaded]);
   327→
   328→  // Head-to-head data (only when both slugs are chosen).
   329→  const loadPair = useCallback(async (aSlug: string, bSlug: string) => {
   330→    setLoading(true);
   331→    setLoadErr(false);
   332→    try {
   333→      const res = await fetch(
   334→        `/api/compare?a=${encodeURIComponent(aSlug)}&b=${encodeURIComponent(bSlug)}`
   335→      );
   336→      if (!res.ok) throw new Error("failed");
   337→      const json = (await res.json()) as CompareResponse;
   338→      setData(json);
   339→      setDataPair([aSlug, bSlug]);
   340→      setPopular(json.popular ?? []);
   341→    } catch {
   342→      setLoadErr(true);
   343→      setData(null);
   344→      setDataPair([]);
   345→    } finally {
   346→      setLoading(false);
   347→    }
   348→  }, []);
   349→
   350→  useEffect(() => {
   351→    if (!compareOpen || !a || !b) return;
   352→    // Stale responses are harmless: rendering is guarded by dataPair.
   353→    void loadPair(a, b);
   354→  }, [compareOpen, a, b, loadPair]);
   355→
   356→  // Popular strip for the picker view (when no head-to-head yet).
   357→  useEffect(() => {
   358→    if (!compareOpen || (a && b)) return;
   359→    let alive = true;
   360→    fetch("/api/compare?popular=1")
   361→      .then(async (res) => {
   362→        if (!res.ok) throw new Error("failed");
   363→        return (await res.json()) as { popular: PopularPair[] };
   364→      })
   365→      .then((json) => {
   366→        if (alive) setPopular(json.popular ?? []);
   367→      })
   368→      .catch(() => {
   369→        /* strip is optional */
   370→      });
   371→    return () => {
   372→      alive = false;
   373→    };
   374→  }, [compareOpen, a, b]);
   375→
   376→  const rowOf = useCallback(
   377→    (slug: string | null) => rows.find((r) => r.slug === slug) ?? null,
   378→    [rows]
   379→  );
   380→
   381→  /** Head-to-head data matches the currently selected pair. */
   382→  const dataValid =
   383→    !!data && dataPair.length === 2 && dataPair[0] === a && dataPair[1] === b;
   384→
   385→  const pickA = useCallback(
   386→    (slug: string) => {
   387→      if (b) openCompare(slug, b);
   388→      else {
   389→        if (a) removeCompare(a);
   390→        addCompare(slug);
   391→      }
   392→    },
   393→    [a, b, openCompare, removeCompare, addCompare]
   394→  );
   395→
   396→  const pickB = useCallback(
   397→    (slug: string) => {
   398→      if (a) openCompare(a, slug);
   399→      else addCompare(slug);
   400→    },
   401→    [a, openCompare, addCompare]
   402→  );
   403→
   404→  const shareUrl =
   405→    typeof window !== "undefined"
   406→      ? `${window.location.origin}/?compare=${encodeURIComponent(a ?? "")},${encodeURIComponent(b ?? "")}`
   407→      : `/?compare=${encodeURIComponent(a ?? "")},${encodeURIComponent(b ?? "")}`;
   408→
   409→  const verdict = useMemo(() => {
   410→    if (!dataValid) return null;
   411→    const diff = Math.abs(data!.a.votes - data!.b.votes);
   412→    if (diff === 0) return "DEAD EVEN — SPLIT THE ROOM";
   413→    const winner = data!.a.votes > data!.b.votes ? data!.a.name : data!.b.name;
   414→    return `COMMUNITY LEANS ${winner.toUpperCase()} · ${diff} VOTE${diff === 1 ? "" : "S"}`;
   415→  }, [data, dataValid]);
   416→
   417→  if (!compareOpen) return null;
   418→
   419→  const rowA = dataValid ? data!.a : null;
   420→  const rowB = dataValid ? data!.b : null;
   421→  const hasPair = !!a && !!b;
   422→
   423→  return (
   424→    <FullPageShell
   425→      kicker="Side-by-side"
   426→      breadcrumb={[{ label: "Home" }, { label: "Compare" }]}
   427→      onClose={closeCompare}
   428→      shareUrl={hasPair ? shareUrl : undefined}
   429→      wide
   430→      ariaLabel="Compare tools side by side"
   431→    >
   432→      <div className="space-y-8">
   433→        {/* Pickers */}
   434→        <section aria-label="Choose tools to compare" className="space-y-3">
   435→          <div className="grid gap-3 sm:grid-cols-2">
   436→            <ToolPicker
   437→              label="TOOL A"
   438→              current={rowOf(a)}
   439→              excludeSlug={b}
   440→              rows={rows}
   441→              onPick={pickA}
   442→              onClear={() => a && removeCompare(a)}
   443→            />
   444→            <ToolPicker
   445→              label="TOOL B"
   446→              current={rowOf(b)}
   447→              excludeSlug={a}
   448→              rows={rows}
   449→              onPick={pickB}
   450→              onClear={() => b && removeCompare(b)}
   451→            />
   452→          </div>
   453→          {!hasPair && (
   454→            <p className="font-mono text-[11px] uppercase tracking-[0.2em] text-white/40">
   455→              {a || b
   456→                ? "Pick the second tool to unlock the side-by-side."
   457→                : "Pick two tools to compare — or start from a popular matchup below."}
   458→            </p>
   459→          )}
   460→        </section>
   461→
   462→        {/* Loading */}
   463→        {hasPair && loading && <PageSkeleton />}
   464→
   465→        {/* Load failure — pickers stay usable */}
   466→        {hasPair && !loading && (loadErr || !rowA || !rowB) && (
   467→          <p className="rounded-xl border border-red-500/25 bg-red-500/[0.04] p-4 text-sm text-red-300">
   468→            Couldn&apos;t load the comparison for this pair — try different tools.
   469→          </p>
   470→        )}
   471→
   472→        {/* Comparison grid */}
   473→        {hasPair && !loading && rowA && rowB && (
   474→          <section aria-label="Comparison" className="space-y-6">
   475→            {/* Header cards */}
   476→            <div className="grid gap-4 sm:grid-cols-2">
   477→              {[rowA, rowB].map((r, i) => (
   478→                <div
   479→                  key={r.slug}
   480→                  className="flex items-center gap-4 rounded-xl border border-white/10 bg-white/[0.02] p-4 transition-colors hover:border-ember/40"
   481→                >
   482→                  <span
   483→                    aria-hidden
   484→                    className={cn(
   485→                      "grid size-14 shrink-0 place-items-center rounded-2xl bg-gradient-to-br text-3xl",
   486→                      r.gradient
   487→                    )}
   488→                  >
   489→                    {r.emoji}
   490→                  </span>
   491→                  <div className="min-w-0 flex-1">
   492→                    <div className="flex items-center gap-2">
   493→                      <h2 className="truncate text-xl font-black text-white">{r.name}</h2>
   494→                      <span className="shrink-0 rounded-full border border-white/15 px-1.5 py-px font-mono text-[9px] tracking-wider text-white/50">
   495→                        {i === 0 ? "TOOL A" : "TOOL B"}
   496→                      </span>
   497→                    </div>
   498→                    <p className="mt-0.5 truncate text-sm text-white/55">{r.tagline}</p>
   499→                  </div>
   500→                  <span className="shrink-0 rounded-full bg-ember/10 px-3 py-1.5 font-mono text-sm font-black tabular-nums text-ember">
   501→                    ▲ {r.votes}
   502→                  </span>
   503→                </div>
   504→              ))}
   505→            </div>
   506→
   507→            {/* Auto-verdict */}
   508→            {verdict && (
   509→              <p className="text-center font-mono text-[11px] uppercase tracking-[0.25em] text-white/50" aria-live="polite">
   510→                {verdict}
   511→              </p>
   512→            )}
   513→
   514→            {/* Metric table */}
   515→            <div className="rounded-xl border border-white/10 bg-white/[0.02] px-4 py-1 sm:px-5">
   516→              <MetricRow
   517→                label="Votes"
   518→                a={<span className="font-mono font-bold tabular-nums text-ember">{rowA.votes}</span>}
   519→                b={<span className="font-mono font-bold tabular-nums text-ember">{rowB.votes}</span>}
   520→              />
   521→              <MetricRow
   522→                label="Rating"
   523→                a={
   524→                  rowA.rating != null ? (
   525→                    <span className="tabular-nums">
   526→                      {rowA.rating.overall.toFixed(1)}{" "}
   527→                      <span className="font-mono text-[10px] text-white/40">({rowA.rating.count})</span>
   528→                    </span>
   529→                  ) : (
   530→                    <span className="font-mono text-xs text-white/40">
   531→                      LOCKED ({rowA.reviewCount})
   532→                    </span>
   533→                  )
   534→                }
   535→                b={
   536→                  rowB.rating != null ? (
   537→                    <span className="tabular-nums">
   538→                      {rowB.rating.overall.toFixed(1)}{" "}
   539→                      <span className="font-mono text-[10px] text-white/40">({rowB.rating.count})</span>
   540→                    </span>
   541→                  ) : (
   542→                    <span className="font-mono text-xs text-white/40">
   543→                      LOCKED ({rowB.reviewCount})
   544→                    </span>
   545→                  )
   546→                }
   547→              />
   548→              {rowA.rating?.ease != null && rowB.rating?.ease != null && (
   549→                <MetricRow label="Ease" a={<MiniBar value={rowA.rating.ease} />} b={<MiniBar value={rowB.rating.ease} />} />
   550→              )}
   551→              {rowA.rating?.power != null && rowB.rating?.power != null && (
   552→                <MetricRow label="Power" a={<MiniBar value={rowA.rating.power} />} b={<MiniBar value={rowB.rating.power} />} />
   553→              )}
