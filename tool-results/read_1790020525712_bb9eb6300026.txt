   135→function fmtDate(iso: string | null): string {
   136→  if (!iso) return "—";
   137→  return new Date(iso).toLocaleDateString("en-US", {
   138→    month: "short",
   139→    day: "numeric",
   140→    year: "numeric",
   141→    timeZone: "UTC",
   142→  });
   143→}
   144→
   145→function fmtDay(iso: string | null): string {
   146→  if (!iso) return "—";
   147→  return new Date(iso).toLocaleDateString("en-US", {
   148→    month: "short",
   149→    day: "numeric",
   150→    timeZone: "UTC",
   151→  });
   152→}
   153→
   154→function relTime(iso: string): string {
   155→  const s = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000);
   156→  if (s < 60) return "just now";
   157→  const m = Math.floor(s / 60);
   158→  if (m < 60) return `${m}m ago`;
   159→  const h = Math.floor(m / 60);
   160→  if (h < 24) return `${h}h ago`;
   161→  return `${Math.floor(h / 24)}d ago`;
   162→}

   308→type StartedClaim = {

   311→  token: string;

   316→const DEFAULT_CLAIM_STEPS = [
   317→  "Add the meta tag below to your site's <head> section.",
   318→  "Deploy or save the change so the tag is live on your homepage.",
   319→  "Hit VERIFY NOW — we fetch your page and look for the tag.",

   322→function ClaimPanel({
   323→  slug,
   324→  maker,
   325→  claim,
   326→  sessionStatus,
   327→  onRefresh,
   328→}: {
   329→  slug: string;
   330→  maker: string;
   331→  claim: StartedClaim | null;
   332→  sessionStatus: "loading" | "authenticated" | "unauthenticated";
   333→  onRefresh: () => void;
   334→}) {
   335→  const { toast } = useToast();
   336→  const [starting, setStarting] = useState(false);
   337→  const [verifying, setVerifying] = useState(false);
   338→  const [copied, setCopied] = useState(false);
   339→  const [failReason, setFailReason] = useState<string | null>(null);
   340→  const claimRef = useRef(claim);
   341→  claimRef.current = claim;
   342→
   343→  const metaTag = claim
   344→    ? `<meta name="prother-verify" content="${claim.token}" />`
   345→    : "";
   346→
   347→  const startClaim = useCallback(async () => {
   348→    setStarting(true);
   349→    setFailReason(null);
   350→    try {
   351→      const res = await fetch("/api/claims", {
   352→        method: "POST",
   353→        headers: { "Content-Type": "application/json" },
   354→        body: JSON.stringify({ toolSlug: slug }),
   355→      });
   356→      const data = (await res.json().catch(() => ({}))) as {
   357→        claim?: StartedClaim;
   358→        verified?: boolean;
   359→        error?: string;
   360→      };
   361→      if (res.status === 409) {
   362→        toast({ title: "This listing is already claimed." });
   363→        onRefresh();
   364→        return;
   365→      }
   366→      if (!res.ok || !data.claim) {
   367→        toast({
   368→          title: "Could not start the claim",
   369→          description: data.error ?? "Please try again.",
   370→          variant: "destructive",
   371→        });
   372→        return;
   373→      }
   374→      toast({ title: "Claim started — add the meta tag to your site." });
   375→      onRefresh();
   376→    } catch {
   377→      toast({
   378→        title: "Could not start the claim",
   379→        description: "Please try again.",
   380→        variant: "destructive",
   381→      });
   382→    } finally {
   383→      setStarting(false);
   384→    }
   385→  }, [slug, toast, onRefresh]);
   386→
   387→  const onVerify = useCallback(async () => {
   388→    const c = claimRef.current;
   389→    if (!c) return;
   390→    setVerifying(true);
   391→    setFailReason(null);
   392→    try {
   393→      const res = await fetch(`/api/claims/${encodeURIComponent(c.id)}/verify`, {
   394→        method: "POST",
   395→      });
   396→      const data = (await res.json().catch(() => ({}))) as {
   397→        verified?: boolean;
   398→        reason?: string;
   399→        claim?: StartedClaim;
   400→      };
   401→      if (data.verified) {
   402→        toast({
   403→          title: "Ownership verified ✓",
   404→          description: "This listing is yours.",
   405→        });
   406→        onRefresh();
   407→      } else {
   408→        setFailReason(data.reason ?? "We couldn't find the meta tag on your site yet.");
   409→      }
   410→    } catch {
   411→      setFailReason("Verification request failed — please try again.");
   412→    } finally {
   413→      setVerifying(false);
   414→    }
   415→  }, [toast, onRefresh]);
   416→
   417→  const onCopyTag = useCallback(async () => {
   418→    try {
   419→      await navigator.clipboard.writeText(metaTag);
   420→      setCopied(true);
   421→      window.setTimeout(() => setCopied(false), 1600);
   422→    } catch {
   423→      toast({ title: "Could not copy — select the tag manually.", variant: "destructive" });
   424→    }
   425→  }, [metaTag, toast]);
   426→
   427→  // Anonymous: sign-in gate.
   428→  if (sessionStatus === "unauthenticated") {
   429→    return (
   430→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border border-ember/30 bg-ember/[0.05] p-4">
   431→        <p className="text-sm font-semibold text-white/85">
   432→          Sign in to prove ownership of this listing.
   433→        </p>
   434→        <Button
   435→          type="button"
   436→          onClick={openAuth}
   437→          className="h-10 bg-ember font-mono text-xs font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   438→        >
   439→          SIGN IN TO CLAIM →
   440→        </Button>
   441→      </div>
   442→    );
   443→  }
   444→
   445→  // No claim yet (authenticated): start one.
   446→  if (!claim) {
   447→    return (
   448→      <div className="rounded-xl border border-ember/30 bg-ember/[0.05] p-4">
   449→        <p className="text-sm text-white/70">
   450→          Start a claim — we&apos;ll generate a verification meta tag for{" "}
   451→          <span className="text-white">{maker}</span> to place on the product
   452→          site.
   453→        </p>
   454→        <Button
   455→          type="button"
   456→          onClick={() => void startClaim()}
   457→          disabled={starting}
   458→          className="mt-3 h-10 bg-ember font-mono text-xs font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   459→        >
   460→          {starting ? (
   461→            <>
   462→              <Loader2 className="size-3.5 animate-spin" aria-hidden /> STARTING…
   463→            </>
   464→          ) : (
   465→            "START CLAIM →"
   466→          )}
   467→        </Button>
   468→      </div>
   469→    );
   470→  }
   471→
   472→  // Claim in progress / verified / failed.
   473→  const verified = claim.status === "verified" || !!claim.verifiedAt;
   474→  return (
   475→    <div
   476→      className={cn(
   477→        "rounded-xl border p-4",
   478→        verified ? "border-mint/40 bg-mint/[0.05]" : "border-ember/30 bg-ember/[0.05]"
   479→      )}
   480→      aria-live="polite"
   481→    >
   482→      <div className="flex flex-wrap items-center gap-2">
   483→        <p className={MONO}>OWNERSHIP CLAIM</p>
   484→        <span
   485→          className={cn(
   486→            "rounded-full border px-2 py-0.5 font-mono text-[9px] tracking-wider",
   487→            verified
   488→              ? "border-mint/40 bg-mint/10 text-mint"
   489→              : claim.status === "failed"
   490→                ? "border-red-500/40 bg-red-500/10 text-red-400"
   491→                : "border-white/20 bg-white/5 text-white/70"
   492→          )}
   493→        >
   494→          {verified ? "VERIFIED" : claim.status.toUpperCase()}
   495→        </span>
   496→      </div>
   497→
   498→      {verified ? (
   499→        <p className="mt-2 text-sm text-mint">
   500→          ✓ Verified — this listing is yours.
   501→        </p>
   502→      ) : (
   503→        <>
   504→          <ol className="mt-3 space-y-1.5">
   505→            {DEFAULT_CLAIM_STEPS.map((step, i) => (
   506→              <li key={i} className="flex gap-2 text-xs leading-relaxed text-white/65">
   507→                <span className="font-mono text-ember">{i + 1}.</span>
   508→                {step}
   509→              </li>
   510→            ))}
   511→          </ol>
   512→
   513→          <div className="mt-3 flex items-start gap-2">
   514→            <code className="min-w-0 flex-1 break-all rounded-lg border border-white/10 bg-black/40 p-2.5 font-mono text-[11px] text-white/80">
   515→              {metaTag}
   516→            </code>
   517→            <Button
   518→              type="button"
   519→              variant="outline"
   520→              size="icon"
   521→              aria-label="Copy the verification meta tag"
   522→              onClick={() => void onCopyTag()}
   523→              className={cn(
   524→                "bg-white/[0.03] hover:bg-white/[0.08] size-9 shrink-0 border-white/10",
   525→                copied ? "text-mint" : "text-white/70 hover:border-ember/40 hover:text-ember"
   526→              )}
   527→            >
   528→              {copied ? <Check className="size-4" aria-hidden /> : <Copy className="size-4" aria-hidden />}
   529→            </Button>
   530→          </div>
   531→
   532→          {failReason && (
   533→            <p className="mt-3 rounded-lg border border-red-500/25 bg-red-500/[0.06] p-2.5 text-xs text-red-300">
   534→              {failReason}
   535→            </p>
   536→          )}
   537→
   538→          <div className="mt-3 flex flex-wrap items-center gap-3">
   539→            <Button
   540→              type="button"
   541→              onClick={() => void onVerify()}
   542→              disabled={verifying}
   543→              className="h-10 bg-ember font-mono text-xs font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   544→            >
   545→              {verifying ? (
   546→                <>
   547→                  <Loader2 className="size-3.5 animate-spin" aria-hidden /> VERIFYING…
   548→                </>
   549→              ) : (
   550→                "VERIFY NOW"
   551→              )}
   552→            </Button>
   553→            <p className="font-mono text-[10px] uppercase tracking-[0.2em] text-white/35">
   554→              False claims fail by design · disputes → editor arbitration
   555→            </p>
   556→          </div>
   557→        </>
   558→      )}
   559→    </div>
   560→  );
   561→}
