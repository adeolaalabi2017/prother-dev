   298→function QueueTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   299→  const { toast } = useToast();
   300→  const [subs, setSubs] = useState<PendingSub[] | null>(null);
   301→  const [rejecting, setRejecting] = useState<string | null>(null);
   302→  const [failed, setFailed] = useState<string[]>([]);
   303→  const [busy, setBusy] = useState<string | null>(null);
   304→
   305→  const load = useCallback(() => {
   306→    adminFetch(apiKey, "/api/editor/queue")
   307→      .then((r) => r.json())
   308→      .then((d: { pending?: PendingSub[] }) => setSubs(d.pending ?? []))
   309→      .catch(() => setSubs([]));
   310→  }, [apiKey]);
   311→
   312→  useEffect(load, [load]);
   313→
   314→  const decide = useCallback(
   315→    async (id: string, decision: "approve" | "reject") => {
   316→      setBusy(id);
   317→      try {
   318→        const res = await adminFetch(apiKey, "/api/editor/decision", {
   319→          method: "POST",
   320→          body: JSON.stringify(
   321→            decision === "approve"
   322→              ? { decision, id }
   323→              : { decision, id, failedStandards: failed, note: "" }
   324→          ),
   325→        });
   326→        const d = (await res.json()) as { ok?: boolean; error?: string };
   327→        if (!res.ok || !d.ok) {
   328→          toast({ title: d.error ?? "Decision failed", variant: "destructive" });
   329→          return;
   330→        }
   331→        toast({
   332→          title: decision === "approve" ? "Approved & scheduled" : "Rejected",
   333→          description:
   334→            decision === "approve"
   335→              ? "Launches tomorrow at 00:00 UTC."
   336→              : "Maker email names the failed standards.",
   337→        });
   338→        setRejecting(null);
   339→        setFailed([]);
   340→        load();
   341→        onChanged();
   342→      } catch {
   343→        toast({ title: "Network error", variant: "destructive" });
   344→      } finally {
   345→        setBusy(null);
   346→      }
   347→    },
   348→    [apiKey, failed, load, onChanged, toast]
   349→  );
   350→
   351→  if (subs === null) return <Spinner />;
   352→
   353→  return (
   354→    <div className="space-y-3">
   355→      {subs.length === 0 && (
   356→        <div className="rounded-xl border border-dashed border-white/15 bg-white/[0.02] p-10 text-center">
   357→          <CheckCircle2 className="mx-auto size-8 text-emerald-400" aria-hidden />
   358→          <p className="mt-3 font-mono text-sm text-white/60">
   359→            Queue clear — nothing waiting.
   360→          </p>
   361→        </div>
   362→      )}
   363→      {subs.map((s) => (
   364→        <div key={s.id} className="rounded-xl border border-white/10 bg-white/[0.02] p-4">
   365→          <div className="flex items-center gap-3">
   366→            <div
   367→              aria-hidden
   368→              className={cn(
   369→                "flex size-10 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br text-lg",
   370→                s.logoGradient
   371→              )}
   372→            >
   373→              {s.logoEmoji}
   374→            </div>
   375→            <div className="min-w-0 flex-1">
   376→              <p className="truncate text-sm font-bold text-white">{s.name}</p>
   377→              <p className="truncate text-xs text-white/50">{s.tagline}</p>
   378→            </div>
   379→            <span className="hidden shrink-0 font-mono text-[10px] text-white/40 sm:block">
   380→              {s.domain} · {s.ageH}h
   381→            </span>
   382→          </div>
   383→
   384→          {rejecting === s.id ? (
   385→            <div className="mt-3 space-y-2.5 rounded-lg border border-red-400/25 bg-red-400/[0.05] p-3">
   386→              <p className="font-mono text-[10px] tracking-widest text-red-300">
   387→                REJECTIONS MUST CITE FAILED STANDARD(S)
   388→              </p>
   389→              <div className="flex flex-wrap gap-1.5">
   390→                {STANDARD_DEFS.map((sd) => {
   391→                  const on = failed.includes(sd.id);
   392→                  return (
   393→                    <label
   394→                      key={sd.id}
   395→                      className={cn(
   396→                        "flex cursor-pointer items-center gap-1 rounded-full border px-2 py-0.5 font-mono text-[10px]",
   397→                        on ? "border-red-400 bg-red-400/15 text-red-200" : "border-white/15 text-white/60"
   398→                      )}
   399→                    >
   400→                      <Checkbox
   401→                        checked={on}
   402→                        onCheckedChange={(v) =>
   403→                          setFailed((p) => (v ? [...p, sd.id] : p.filter((x) => x !== sd.id)))
   404→                        }
   405→                        className="size-3 border-white/30 data-[state=checked]:border-red-400 data-[state=checked]:bg-red-400"
   406→                      />
   407→                      {sd.id}
   408→                    </label>
   409→                  );
   410→                })}
   411→              </div>
   412→              <div className="flex gap-2">
   413→                <Button
   414→                  size="sm"
   415→                  disabled={failed.length === 0 || busy === s.id}
   416→                  onClick={() => void decide(s.id, "reject")}
   417→                  className="rounded-lg bg-red-500 text-white shadow-none hover:bg-red-400 disabled:opacity-40"
   418→                >
   419→                  Reject ({failed.length})
   420→                </Button>
   421→                <Button
   422→                  size="sm"
   423→                  variant="ghost"
   424→                  onClick={() => setRejecting(null)}
   425→                  className="rounded-lg text-white/60 hover:text-white"
   426→                >
   427→                  Back
   428→                </Button>
   429→              </div>
   430→            </div>
   431→          ) : (
   432→            <div className="mt-3 flex gap-2">
   433→              <Button
   434→                size="sm"
   435→                disabled={busy === s.id}
   436→                onClick={() => void decide(s.id, "approve")}
   437→                className="rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   438→              >
   439→                {busy === s.id ? (
   440→                  <Loader2 className="size-3.5 animate-spin" aria-hidden />
   441→                ) : (
   442→                  <CalendarClock className="size-3.5" aria-hidden />
   443→                )}
   444→                Approve → tomorrow
   445→              </Button>
   446→              <Button
   447→                size="sm"
   448→                variant="outline"
   449→                onClick={() => setRejecting(s.id)}
   450→                className="rounded-lg border-red-400/30 text-red-400 hover:bg-red-400/10 hover:text-red-300"
   451→              >
   452→                Reject…
   453→              </Button>
   454→              <a
   455→                href={s.email ? `mailto:${s.email}` : "#"}
   456→                className="ml-auto hidden items-center gap-1 font-mono text-[10px] text-white/40 hover:text-white/70 sm:flex"
   457→              >
   458→                <Mail className="size-3" aria-hidden /> maker
   459→              </a>
   460→            </div>
   461→          )}
   462→        </div>
   463→      ))}
   464→    </div>
   465→  );
   466→}

   470→function CalendarTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   471→  const { toast } = useToast();
   472→  const [data, setData] = useState<ScheduleData | null>(null);
   473→  const [dates, setDates] = useState<Record<string, string>>({});
   474→  const [busy, setBusy] = useState<string | null>(null);
   475→
   476→  const load = useCallback(() => {
   477→    adminFetch(apiKey, "/api/admin/schedule")
   478→      .then((r) => r.json())
   479→      .then((d: ScheduleData) => setData(d))
   480→      .catch(() => toast({ title: "Calendar unavailable", variant: "destructive" }));
   481→  }, [apiKey, toast]);
   482→
   483→  useEffect(load, [load]);
   484→
   485→  const schedule = useCallback(
   486→    async (toolId: string, date: string | null, name: string) => {
   487→      setBusy(toolId);
   488→      try {
   489→        const res = await adminFetch(apiKey, "/api/admin/schedule", {
   490→          method: "POST",
   491→          body: JSON.stringify({ toolId, date }),
   492→        });
   493→        const d = (await res.json()) as { ok?: boolean; error?: string };
   494→        if (!res.ok || !d.ok) {
   495→          toast({ title: d.error ?? "Scheduling failed", variant: "destructive" });
   496→          return;
   497→        }
   498→        toast({ title: date ? `${name} → ${date}` : `${name} unscheduled` });
   499→        load();
   500→        onChanged();
   501→      } catch {
   502→        toast({ title: "Network error", variant: "destructive" });
   503→      } finally {
   504→        setBusy(null);
   505→      }
   506→    },
   507→    [apiKey, load, onChanged, toast]
   508→  );
   509→
   510→  if (!data) return <Spinner />;
   511→
   512→  const chip = (n: number) => {
   513→    if (n < data.floor) return { emoji: "🔴", label: "below floor" };
   514→    if (n >= 11) return { emoji: "🟡", label: "11–15" };
   515→    return { emoji: "🟢", label: `${data.floor}–10` };
   516→  };
   517→
   518→  return (
   519→    <div className="space-y-5">
   520→      <p className="font-mono text-[10px] tracking-wider text-white/35 uppercase">
   521→        Floor ≥{data.floor} · Cap ≤{data.cap} · today locked (live ranking)
   522→      </p>
   523→
   524→      {/* 14-day grid */}
   525→      <div className="max-h-72 space-y-1.5 overflow-y-auto pr-1">
   526→        {data.days.map((d) => {
   527→          const c = chip(d.count);
   528→          return (
   529→            <div
   530→              key={d.date}
   531→              className="rounded-xl border border-white/10 bg-white/[0.02] px-3 py-2.5"
   532→            >
   533→              <div className="flex items-center gap-2.5">
   534→                <span className="w-20 shrink-0 font-mono text-[11px] text-white/70">
   535→                  {d.weekday} {d.date.slice(5)}
   536→                </span>
   537→                <span
   538→                  title={c.label}
   539→                  className="shrink-0 rounded-full border border-white/10 bg-white/5 px-2 py-0.5 font-mono text-[10px] text-white/60"
   540→                >
   541→                  {d.locked ? "🔒" : c.emoji} {d.count}
   542→                </span>
   543→                <div className="flex min-w-0 flex-1 flex-wrap gap-1">
   544→                  {d.tools.map((t) => (
   545→                    <span
   546→                      key={t.id}
   547→                      className={cn(
   548→                        "group inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[10px]",
   549→                        t.scheduled
   550→                          ? "border-ember/40 bg-ember/10 text-ember"
   551→                          : "border-white/15 bg-white/5 text-white/70"
   552→                      )}
   553→                    >
   554→                      <span aria-hidden>{t.logoEmoji}</span>
   555→                      {t.name}
   556→                      {t.editorsPick && <Star className="size-3 text-ember" aria-label="Editor's Pick" />}
   557→                      {!d.locked && (
   558→                        <button
   559→                          type="button"
   560→                          onClick={() => void schedule(t.id, null, t.name)}
   561→                          aria-label={`Unschedule ${t.name}`}
   562→                          className="ml-0.5 text-white/30 transition-colors hover:text-red-400"
   563→                        >
   564→                          <X className="size-3" aria-hidden />
   565→                        </button>
   566→                      )}
   567→                    </span>
   568→                  ))}
   569→                  {d.tools.length === 0 && (
   570→                    <span className="font-mono text-[10px] text-white/25">empty</span>
   571→                  )}
   572→                </div>
   573→              </div>
   574→            </div>
   575→          );
   576→        })}
   577→      </div>
   578→
   579→      {/* unscheduled pool */}
   580→      <div>
   581→        <p className={labelCx}>Unscheduled pool ({data.pool.length})</p>
   582→        <div className="mt-2 max-h-72 space-y-1.5 overflow-y-auto pr-1">
   583→          {data.pool.length === 0 && (
   584→            <p className="py-6 text-center text-sm text-white/30">
   585→              Pool empty — every listing is on the calendar.
   586→            </p>
   587→          )}
   588→          {data.pool.map((t) => (
   589→            <div
   590→              key={t.id}
   591→              className="flex flex-wrap items-center gap-2.5 rounded-xl border border-white/10 bg-white/[0.02] px-3 py-2"
   592→            >

  1051→    setBusy(true);
  1052→    try {
  1053→      const payload = {
  1054→        title: f.title,
  1055→        slug: f.slug || undefined,
  1056→        excerpt: f.excerpt,
  1057→        // Only send the body when rewritten — a metadata-only edit of an
  1058→        // existing post keeps the stored markdown server-side.
  1059→        ...(f.body.trim().length >= 50 ? { body: f.body } : {}),
  1060→        category: f.category,

  1068→        keywords: f.keywords || undefined,
  1069→      };

  1071→        apiKey,
  1072→        isNew ? "/api/admin/posts" : `/api/admin/posts/${f.id}`,
  1073→        { method: isNew ? "POST" : "PATCH", body: JSON.stringify(payload) }
  1074→      );
  1075→      const d = (await res.json()) as { ok?: boolean; error?: string; slug?: string };

  1077→        toast({ title: d.error ?? "Save failed", variant: "destructive" });
  1078→        return;
  1079→      }

  1081→        title: publishOverride === "published" ? "Published" : "Saved",
  1082→        description: d.slug ? `/journal/${d.slug}` : undefined,
  1083→      });
  1084→      onDone(publishOverride === "published" ? d.slug : undefined);
  1085→    } catch {
  1086→      toast({ title: "Network error", variant: "destructive" });
  1087→    } finally {
  1088→      setBusy(false);
  1089→    }
  1090→  };

  1102→      setBusy(false);
  1103→    }

  1107→  const seoDesc = f.seoDescription || f.excerpt;

  1114→        </Field>
  1115→        <Field label="Slug (auto from title if blank)">
  1116→          <Input value={f.slug} onChange={(e) => setF({ ...f, slug: e.target.value })} className={cn(inputCx, "font-mono")} placeholder="how-to-launch-an-ai-tool" />
  1117→        </Field>
  1118→      </div>

  1122→        <p className="text-[10px] text-white/30">{f.excerpt.length}/200</p>

  1662→type Delta = { text: string; dir: "up" | "down" | "flat"; title: string };

  1664→/** Honest delta vs yesterday — no fabrication when the base is zero. */
  1665→function computeDelta(cur: number, prev: number): Delta {
  1666→  if (prev === 0 && cur === 0)
  1667→    return { text: "±0", dir: "flat", title: "No change vs yesterday" };
  1668→  if (prev === 0)
  1669→    return { text: "▲ new", dir: "up", title: `${cur} today · none yesterday` };
  1670→  const pct = Math.round(((cur - prev) / prev) * 100);
  1671→  if (pct === 0)
  1672→    return { text: "±0%", dir: "flat", title: `${cur} today vs ${prev} yesterday` };
  1673→  return pct > 0
  1674→    ? { text: `▲ ${pct}%`, dir: "up", title: `${cur} today vs ${prev} yesterday` }
  1675→    : { text: `▼ ${Math.abs(pct)}%`, dir: "down", title: `${cur} today vs ${prev} yesterday` };
  1676→}
