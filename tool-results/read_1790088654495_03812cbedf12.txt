   164→function PendingCard({

   173→  const { toast } = useToast();
   174→  const [busy, setBusy] = useState<"approve" | "reject" | null>(null);
   175→  const [rejecting, setRejecting] = useState(false);
   176→  const [failed, setFailed] = useState<string[]>([]);

   180→  const category = CATEGORIES.find((c) => c.slug === sub.categorySlug);

   182→    PRICING_MODELS.find((p) => p.value === sub.pricingModel)?.label ??
   183→    sub.pricingModel;
   184→  const tags = sub.tags ? sub.tags.split("|").filter(Boolean) : [];
   185→
   186→  const decide = useCallback(
   187→    async (decision: "approve" | "reject") => {
   188→      setBusy(decision);
   189→      try {
   190→        const res = await fetch("/api/editor/decision", {
   191→          method: "POST",
   192→          headers: {
   193→            "Content-Type": "application/json",
   194→            "x-editor-key": sessionStorage.getItem(KEY_STORAGE) ?? "",
   195→          },
   196→          body: JSON.stringify(
   197→            decision === "approve"
   198→              ? { decision, id: sub.id }
   199→              : { decision, id: sub.id, failedStandards: failed, note }
   200→          ),
   201→        });
   202→        const data = (await res.json()) as {
   203→          ok?: boolean;
   204→          error?: string;
   205→          slug?: string;
   206→          reviewNote?: string;
   207→        };

   209→          toast({
   210→            title: data.error ?? "Decision failed",
   211→            variant: "destructive",
   212→          });
   213→          return;
   214→        }
   215→        if (decision === "approve" && data.slug) {
   216→          toast({
   217→            title: `${sub.name} scheduled`,
   218→            description: `Added to tomorrow's launch day as /tool/${data.slug}.`,
   219→          });
   220→          onApproved(data.slug);
   221→        } else {
   222→          toast({ title: `${sub.name} rejected`, description: data.reviewNote });
   223→          onRejected(sub.id);
   224→        }

   226→        toast({
   227→          title: "Decision failed",
   228→          description: "Network error — try again.",
   229→          variant: "destructive",
   230→        });
   231→      } finally {
   232→        setBusy(null);
   233→      }
   234→    },
   235→    [sub, failed, note, toast, onApproved, onRejected]
   236→  );

   247→        <div
   248→          aria-hidden
   249→          className={cn(
   250→            "flex size-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br text-lg shadow-inner",
   251→            sub.logoGradient
   252→          )}
   253→        >
   254→          {sub.logoEmoji}
   255→        </div>
   256→        <div className="min-w-0 flex-1">
   257→          <p className="flex items-center gap-2 font-bold text-white">
   258→            <span className="truncate">{sub.name}</span>
   259→            {sub.isOwner ? (
   260→              <span className="inline-flex shrink-0 items-center gap-0.5 rounded-full border border-emerald-500/30 bg-emerald-500/10 px-1.5 py-px font-mono text-[9px] text-emerald-400">
   261→                <BadgeCheck className="size-2.5" aria-hidden /> OWNER
   262→              </span>
   263→            ) : (
   264→              <span className="shrink-0 rounded-full border border-white/15 px-1.5 py-px font-mono text-[9px] text-white/50">
   265→                3RD PARTY
   266→              </span>
   267→            )}
   268→          </p>
   269→          <p className="truncate text-sm text-white/60">{sub.tagline}</p>
   270→        </div>
   271→        <div className="hidden shrink-0 text-right sm:block">
   272→          <p className="font-mono text-[10px] text-white/40">{sub.domain}</p>

   274→            <Clock className="size-3" aria-hidden /> {sub.ageH}h in queue
   275→          </p>
   276→        </div>
   277→        <ChevronDown
   278→          className={cn(
   279→            "size-4 shrink-0 text-white/40 transition-transform",
   280→            expanded && "rotate-180"

   282→          aria-hidden

   300→                    LISTING

   486→const CLAIM_STATUS_STYLE: Record<EditorClaim["status"], string> = {
   487→  pending: "border-amber-400/30 bg-amber-400/10 text-amber-300",

   491→
   492→function ClaimCard({ claim, onDone }: { claim: EditorClaim; onDone: () => void }) {
   493→  const { toast } = useToast();
   494→  const [busy, setBusy] = useState<"verify" | "dismiss" | null>(null);
   495→  const [dismissing, setDismissing] = useState(false);
   496→  const [note, setNote] = useState("");
   497→
   498→  const decide = useCallback(
   499→    async (action: "verify" | "dismiss") => {
   500→      setBusy(action);
   501→      try {
   502→        const res = await fetch("/api/editor/arbitrate", {
   503→          method: "POST",
   504→          headers: {
   505→            "Content-Type": "application/json",
   506→            "x-editor-key": sessionStorage.getItem(KEY_STORAGE) ?? "",
   507→          },
   508→          body: JSON.stringify(
   509→            action === "verify"
   510→              ? { type: "claim", id: claim.id, action }
   511→              : { type: "claim", id: claim.id, action, note: note || undefined }
   512→          ),
   513→        });
   514→        const data = (await res.json()) as { ok?: boolean; error?: string };
   515→        if (!res.ok || !data.ok) {
   516→          toast({ title: data.error ?? "Arbitration failed", variant: "destructive" });
   517→          return;
   518→        }
   519→        toast({
   520→          title:
   521→            action === "verify"
   522→              ? `${claim.toolName} verified → ${claim.userName}`
   523→              : `Claim by ${claim.userName} dismissed`,
   524→          description:
   525→            action === "verify"
   526→              ? "Listing ownership transferred, badge goes live."
   527→              : "Claimant sees DISPUTED — they can re-claim with proof.",
   528→        });
   529→        onDone();
   530→      } catch {
   531→        toast({
   532→          title: "Arbitration failed",
   533→          description: "Network error — try again.",
   534→          variant: "destructive",
   535→        });
   536→      } finally {
   537→        setBusy(null);
   538→      }
   539→    },
   540→    [claim, note, toast, onDone]
   541→  );
   542→
   543→  return (
   544→    <div className="rounded-xl border border-white/10 bg-white/[0.02] p-4">
   545→      <div className="flex items-start gap-3">
   546→        <div
   547→          aria-hidden
   548→          className="flex size-11 shrink-0 items-center justify-center rounded-xl bg-white/5 text-lg"
   549→        >
   550→          {claim.toolEmoji}
   551→        </div>
   552→        <div className="min-w-0 flex-1">
   553→          <p className="flex flex-wrap items-center gap-2 font-bold text-white">
   554→            <span className="truncate">{claim.toolName}</span>
   555→            <span
   556→              className={cn(
   557→                "rounded-full border px-1.5 py-px font-mono text-[9px] uppercase",
   558→                CLAIM_STATUS_STYLE[claim.status]
   559→              )}
   560→            >
   561→              {claim.status}
   562→            </span>
   563→            <span className="rounded-full border border-white/15 px-1.5 py-px font-mono text-[9px] text-white/50">
   564→              {claim.method === "email_domain" ? "EMAIL DOMAIN" : "META TAG"}
   565→            </span>
   566→          </p>
   567→          <p className="truncate text-sm text-white/60">

   678→function FilteredReviewCard({
   679→  review,
   680→  onDone,
   681→}: {
   682→  review: EditorReview;
   683→  onDone: () => void;
   684→}) {
   685→  const { toast } = useToast();
   686→  const [busy, setBusy] = useState<"publish" | "spam" | null>(null);
   687→  const [confirmSpam, setConfirmSpam] = useState(false);
   688→
   689→  const decide = useCallback(
   690→    async (action: "publish" | "spam") => {
   691→      setBusy(action);
   692→      try {
   693→        const res = await fetch("/api/editor/arbitrate", {
   694→          method: "POST",
   695→          headers: {
   696→            "Content-Type": "application/json",
   697→            "x-editor-key": sessionStorage.getItem(KEY_STORAGE) ?? "",
   698→          },
   699→          body: JSON.stringify({ type: "review", id: review.id, action }),
   700→        });
   701→        const data = (await res.json()) as { ok?: boolean; error?: string };
   702→        if (!res.ok || !data.ok) {
   703→          toast({ title: data.error ?? "Moderation failed", variant: "destructive" });
   704→          return;
   705→        }
   706→        toast({
   707→          title:
   708→            action === "publish"
   709→              ? `Review on ${review.toolName} published`
   710→              : `Spam removed from ${review.toolName}`,
   711→          description:
   712→            action === "publish"
   713→              ? "It now counts toward the tool's rating aggregate."
   714→              : "The review was deleted permanently.",
   715→        });
   716→        onDone();
   717→      } catch {
   718→        toast({
   719→          title: "Moderation failed",
   720→          description: "Network error — try again.",
   721→          variant: "destructive",
   722→        });
   723→      } finally {
   724→        setBusy(null);
   725→      }
   726→    },
   727→    [review, toast, onDone]
   728→  );
   729→
   730→  return (
   731→    <div className="rounded-xl border border-amber-400/20 bg-amber-400/[0.04] p-4">
   732→      <div className="flex items-start gap-3">
   733→        <div
   734→          aria-hidden
   735→          className="flex size-11 shrink-0 items-center justify-center rounded-xl bg-white/5 text-lg"
   736→        >
   737→          {review.toolEmoji}
   738→        </div>
   739→        <div className="min-w-0 flex-1">
   740→          <p className="flex flex-wrap items-center gap-2 font-bold text-white">
   741→            <span className="truncate">{review.toolName}</span>
   742→            <span className="rounded-full border border-amber-400/30 bg-amber-400/10 px-1.5 py-px font-mono text-[9px] text-amber-300">
   743→              HELD &lt;48H ACCOUNT
   744→            </span>
   745→          </p>
   746→          <p className="font-mono text-xs text-white/50">
   747→            by {review.author} · waiting {review.ageH}h · /{review.toolSlug}
   748→          </p>
   749→        </div>
   750→      </div>
   751→
   752→      <div className="mt-3 flex flex-wrap gap-x-4 gap-y-1.5">
   753→        <StarRow label="EASE" value={review.ease} />
   754→        <StarRow label="POWER" value={review.power} />
   755→        <StarRow label="VALUE" value={review.value} />
   756→      </div>
   757→      <p className="mt-2.5 whitespace-pre-line rounded-lg border border-white/10 bg-black/30 p-3 text-sm leading-relaxed text-white/75">
   758→        {review.body}
   759→      </p>
   760→
   761→      <div className="mt-3 flex flex-wrap items-center gap-2.5">
   762→        <Button
   763→          type="button"
   764→          disabled={busy !== null}
   765→          onClick={() => decide("publish")}
   766→          className="h-9 rounded-lg bg-emerald-500 font-semibold text-black shadow-none hover:bg-emerald-400 disabled:opacity-50"
   767→        >
   768→          {busy === "publish" ? (
   769→            <Loader2 className="size-4 animate-spin" aria-hidden />
   770→          ) : (
   771→            <CheckCircle2 className="size-4" aria-hidden />
   772→          )}
   773→          Publish
   774→        </Button>
   775→        {!confirmSpam ? (
   776→          <Button
   777→            type="button"
   778→            variant="outline"
   779→            disabled={busy !== null}
   780→            onClick={() => setConfirmSpam(true)}
   781→            className="h-9 rounded-lg border-red-400/30 bg-transparent text-red-400 hover:bg-red-400/10 hover:text-red-300"
   782→          >
   783→            <XCircle className="size-4" aria-hidden />
   784→            Spam…
   785→          </Button>
   786→        ) : (
   787→          <>
   788→            <Button
   789→              type="button"
   790→              disabled={busy !== null}
   791→              onClick={() => decide("spam")}
   792→              className="h-9 rounded-lg bg-red-500 font-semibold text-white shadow-none hover:bg-red-400 disabled:opacity-40"
   793→            >
   794→              {busy === "spam" ? (
   795→                <Loader2 className="size-4 animate-spin" aria-hidden />
   796→              ) : (
   797→                <XCircle className="size-4" aria-hidden />
   798→              )}
   799→              Delete permanently
   800→            </Button>
   801→            <Button
   802→              type="button"
   803→              variant="ghost"
   804→              onClick={() => setConfirmSpam(false)}
   805→              className="h-9 rounded-lg text-white/60 hover:bg-white/5 hover:text-white"
   806→            >
   807→              Back
   808→            </Button>
   809→          </>
   810→        )}
   811→      </div>
   812→    </div>
   813→  );

   826→    "submissions"
   827→  );
   828→
   829→  const load = useCallback(async () => {
   830→    setLoading(true);
   831→    try {
   832→      const res = await fetch("/api/editor/queue", {
   833→        headers: { "x-editor-key": sessionStorage.getItem(KEY_STORAGE) ?? "" },
   834→      });
   835→      if (res.status === 401) {
   836→        sessionStorage.removeItem(KEY_STORAGE);
   837→        setAuthed(false);
   838→        return;

   853→    (key: string) => {

   898→      >

   900→        <DialogDescription className="sr-only">
   901→          Moderate pending tool submissions: approve to schedule for tomorrow,
   902→          or reject citing failed listing standards.
   903→        </DialogDescription>

   906→          <Gate onUnlock={unlock} />

   916→                  Review queue
   917→                </h2>
