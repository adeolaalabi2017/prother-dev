    20→function formatCountdown(totalSec: number): string {
    21→  const s = Math.max(0, totalSec);
    22→  const hh = String(Math.floor(s / 3600)).padStart(2, "0");
    23→  const mm = String(Math.floor((s % 3600) / 60)).padStart(2, "0");
    24→  const ss = String(s % 60).padStart(2, "0");
    25→  return `${hh}:${mm}:${ss}`;
    26→}

    28→function pricingLabel(row: FeedRow): string {
    29→  const { model, price } = row.pricing;
    30→  switch (model) {
    31→    case "free":
    32→      return "Free";
    33→    case "freemium":
    34→      return `Freemium ${price ?? ""}`.trim();
    35→    case "paid":
    36→      return `Paid ${price ?? ""}`.trim();
    37→    case "open_source":
    38→      return "Open Source";
    39→    default:
    40→      return "Free";
    41→  }
    42→}

   335→export function LaunchFeed() {
   336→  const { toast } = useToast();
   337→  const { feed, loading, error, refresh } = useFeed();
   338→  const openTool = useExplorer((s) => s.openTool);
   339→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   340→  const categoryFilter = useExplorer((s) => s.categoryFilter);
   341→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   342→  // ?tab= persistence — shareable feed states (new | top | tomorrow | yesterday).
   343→  // Read AFTER mount only: reading window.location during the initial render makes
   344→  // the server HTML ("new") and the hydrated client (?tab=top) diverge, which
   345→  // React reports as a hydration mismatch. Post-mount sync keeps SSR markup valid.
   346→  const [tab, setTab] = useState<Tab>("new");
   347→  useEffect(() => {
   348→    const t = new URLSearchParams(window.location.search).get("tab");
   349→    if (t === "top" || t === "tomorrow" || t === "yesterday") setTab(t);
   350→  }, []);
   351→
   352→  // Archive day browser: null = yesterday (served from the main feed payload),
   353→  // otherwise an ISO day from the strip — fetched once, then cached.
   354→  const [archiveDate, setArchiveDate] = useState<string | null>(null);
   355→  const [dayCache, setDayCache] = useState<Record<string, DayArchiveResponse>>({});
   356→  const [dayLoading, setDayLoading] = useState(false);
   357→
   358→  // Fetch (and cache) a selected archive day's final standings.
   359→  useEffect(() => {
   360→    if (!archiveDate || dayCache[archiveDate]) return;
   361→    let alive = true;
   362→    setDayLoading(true);
   363→    fetch(`/api/feed/day?date=${encodeURIComponent(archiveDate)}`)
   364→      .then(async (res) => {
   365→        if (!res.ok) throw new Error("Failed to load this archive day");
   366→        return (await res.json()) as DayArchiveResponse;
   367→      })
   368→      .then((data) => {
   369→        if (alive) setDayCache((c) => ({ ...c, [data.date]: data }));
   370→      })
   371→      .catch(() => {
   372→        if (alive) setArchiveDate(null); // gracefully fall back to yesterday
   373→      })
   374→      .finally(() => {
   375→        if (alive) setDayLoading(false);
   376→      });
   377→    return () => {
   378→      alive = false;
   379→    };
   380→  }, [archiveDate, dayCache]);
   381→
   382→  // UTC rollover shifts the whole archive window — drop cached days + selection.
   383→  useEffect(() => {
   384→    setDayCache({});
   385→    setArchiveDate(null);
   386→  }, [feed?.date]);
   387→
   388→  // Keep the selected tab in the URL (replaceState — no history spam).
   389→  const setTabSync = useCallback((next: Tab) => {
   390→    setTab(next);
   391→    if (typeof window === "undefined") return;
   392→    const url = new URL(window.location.href);
   393→    const prev = url.searchParams.get("tab");
   394→    if (next === "new") {
   395→      if (prev) {
   396→        url.searchParams.delete("tab");
   397→        window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   398→      }
   399→    } else if (prev !== next) {
   400→      url.searchParams.set("tab", next);
   401→      window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   402→    }
   403→  }, []);
   404→  const [now, setNow] = useState<number>(() => Date.now());
   405→  const [voteState, setVoteState] = useState<VoteState>({});
   406→
   407→  // Countdown deadline — computed from the feed at render (no effect-synced state).
   408→  const deadlineRef = useRef<{ date: string | null; at: number }>({ date: null, at: 0 });
   409→  if (feed && deadlineRef.current.date !== feed.date) {
   410→    deadlineRef.current = { date: feed.date, at: Date.now() + feed.resetsInSec * 1000 };
   411→  }
   412→  const deadline = deadlineRef.current.at;
   413→
   414→  // Tick the clock every second (setState inside the interval callback is fine).
   415→  useEffect(() => {
   416→    const iv = setInterval(() => setNow(Date.now()), 1000);
   417→    return () => clearInterval(iv);
   418→  }, []);
   419→
   420→  // Votes cast inside the tool detail modal sync back into the feed list.
   421→  useEffect(() => {
   422→    const onModalVote = (e: Event) => {
   423→      const d = (e as CustomEvent).detail as {
   424→        launchId: string;
   425→        votes: number;
   426→        voted: boolean;
   427→      };
   428→      setVoteState((prev) => ({
   429→        ...prev,
   430→        [d.launchId]: { votes: d.votes, voted: d.voted },
   431→      }));
   432→    };
   433→    window.addEventListener("prother:vote", onModalVote);
   434→    return () => window.removeEventListener("prother:vote", onModalVote);
   435→  }, []);
   436→
   437→  const remaining = feed ? Math.max(0, Math.round((deadline - now) / 1000)) : null;
   438→
   439→  // UTC-midnight rollover: refetch the feed exactly once when the countdown hits zero.
   440→  const rolledRef = useRef(false);
   441→  useEffect(() => {
   442→    if (remaining !== null && remaining > 0) {
   443→      rolledRef.current = false;
   444→    } else if (remaining === 0 && !rolledRef.current) {
   445→      rolledRef.current = true;
   446→      void refresh();
   447→    }
   448→  }, [remaining, refresh]);
   449→
   450→  const onVote = useCallback(
   451→    async (row: FeedRow) => {
   452→      const current = voteState[row.launchId] ?? { votes: row.votes, voted: row.voted };
   453→      const nextVoted = !current.voted;
   454→      const nextVotes = current.votes + (nextVoted ? 1 : -1);
   455→      setVoteState((prev) => ({
   456→        ...prev,
   457→        [row.launchId]: { votes: nextVotes, voted: nextVoted },
   458→      }));
   459→      try {
   460→        const res = await fetch("/api/vote", {
   461→          method: "POST",
   462→          headers: { "Content-Type": "application/json" },
   463→          body: JSON.stringify({ launchId: row.launchId, voterKey: getVoterKey() }),
   464→        });
   465→        const data = (await res.json()) as { voted: boolean; votes: number };
   466→        setVoteState((prev) => ({
   467→          ...prev,
   468→          [row.launchId]: { votes: data.votes, voted: data.voted },
   469→        }));
   470→      } catch {
   471→        setVoteState((prev) => ({ ...prev, [row.launchId]: current }));
   472→        toast({ title: "Vote failed", description: "Please try again.", variant: "destructive" });
   473→      }
   474→    },
   475→    [voteState, toast]
   476→  );
   477→
   478→  const rows = useMemo<FeedRow[]>(() => {
   479→    if (!feed) return [];
   480→    const base =
   481→      tab === "top"
   482→        ? feed.top ?? []
   483→        : tab === "yesterday"
   484→          ? archiveDate
   485→            ? dayCache[archiveDate]?.rows ?? []
   486→            : feed.yesterday ?? []
   487→          : feed.new ?? [];
   488→    if (!categoryFilter) return base;
   489→    return base.filter((r) => r.category?.slug === categoryFilter);
   490→  }, [feed, tab, categoryFilter, archiveDate, dayCache]);
   491→
   492→  const tomorrowRows = useMemo(() => {
   493→    if (!feed) return [];
   494→    if (!categoryFilter) return feed.tomorrow ?? [];
   495→    return (feed.tomorrow ?? []).filter((t) => t.category?.slug === categoryFilter);
   496→  }, [feed, categoryFilter]);
   497→
   498→  const activeCategory = useMemo(
   499→    () => CATEGORIES.find((c) => c.slug === categoryFilter) ?? null,
   500→    [categoryFilter]
   501→  );
   502→
   503→  const dayLabel = feed?.dayLabel ?? "Today";
   504→  const todayCount = feed?.todayCount ?? 0;
   505→  const isTomorrow = tab === "tomorrow";
   506→  const isYesterday = tab === "yesterday";
   507→  const weekDays = feed?.weekDays ?? [];
   508→  const archiveLabel = archiveDate
   509→    ? dayCache[archiveDate]?.label ?? feed?.yesterdayLabel ?? "Archive"
   510→    : feed?.yesterdayLabel ?? "Yesterday";
   511→
   512→  const tabs: { key: Tab; label: string; short: string }[] = [
   513→    { key: "new", label: "New", short: "New" },
   514→    { key: "top", label: "Top Today", short: "Top" },
   515→    { key: "tomorrow", label: `Tomorrow (${feed?.tomorrow?.length ?? 0})`, short: `Tmrw (${feed?.tomorrow?.length ?? 0})` },
   516→    { key: "yesterday", label: `Archive (${feed?.yesterday?.length ?? 0})`, short: `Arch. (${feed?.yesterday?.length ?? 0})` },
   517→  ];
   518→
   519→  return (
   520→    <section id="feed" className="bg-ink py-24">
   521→      <div className="mx-auto max-w-6xl px-4 sm:px-6">
   522→        {/* Header */}
   523→        <motion.div
   524→          initial={{ opacity: 0, y: 20 }}
   525→          whileInView={{ opacity: 1, y: 0 }}
   526→          viewport={{ once: true, margin: "-80px" }}
   527→          transition={{ duration: 0.55, ease: "easeOut" }}
