    21→function formatCountdown(totalSec: number): string {
    22→  const s = Math.max(0, totalSec);
    23→  const hh = String(Math.floor(s / 3600)).padStart(2, "0");
    24→  const mm = String(Math.floor((s % 3600) / 60)).padStart(2, "0");
    25→  const ss = String(s % 60).padStart(2, "0");
    26→  return `${hh}:${mm}:${ss}`;
    27→}

    29→function pricingLabel(row: FeedRow): string {
    30→  const { model, price } = row.pricing;
    31→  switch (model) {
    32→    case "free":
    33→      return "Free";
    34→    case "freemium":
    35→      return `Freemium ${price ?? ""}`.trim();
    36→    case "paid":
    37→      return `Paid ${price ?? ""}`.trim();
    38→    case "open_source":
    39→      return "Open Source";
    40→    default:
    41→      return "Free";
    42→  }

   399→type PromoAd = {
   400→  id: string;

   402→  body: string;

   408→type ServedPromo = { ad: PromoAd; clickHref: string };
   409→
   410→/**
   411→ * Sponsored row in the TODAY list — visually a sibling of FeedRowItem but
   412→ * unmistakably an ad: mono PROMOTED chip (ember outline), "by {advertiser} ·
   413→ * Sponsored" byline, and the whole card is ONE link through /api/ads/click
   414→ * (target=_blank, nofollow) so CTR is real. min-h matches the reserved slot
   415→ * rendered while the serve request is in flight → no layout shift.
   416→ */
   417→function PromotedFeedRow({ promo }: { promo: ServedPromo | null | undefined }) {
   418→  // MRC viewability (Task 28) — one ping per served creative, ≥50% on screen
   419→  // ≥1s. Hooks before the early returns so the order is stable across the
   420→  // loading/loaded/empty phases; the skeleton never counts (active=false).
   421→  const boxRef = useRef<HTMLAnchorElement | null>(null);
   422→  const adId = promo?.ad.id ?? null;
   423→  useViewable(boxRef, adId !== null, () => {
   424→    if (!adId) return;
   425→    fetch("/api/ads/viewable", {
   426→      method: "POST",
   427→      headers: { "Content-Type": "application/json" },
   428→      body: JSON.stringify({ id: adId }),
   429→      keepalive: true,
   430→    }).catch(() => {});
   431→  });
   432→
   433→  if (promo === undefined) {
   434→    // Serve request in flight — reserve the row's space (render-when-loaded,
   435→    // reserved space) so the card taking its place never jumps the list.
   436→    return (
   437→      <div
   438→        aria-hidden
   439→        className="min-h-[120px] rounded-xl border border-white/[0.05] bg-white/[0.01]"
   440→      />
   441→    );
   442→  }
   443→  if (promo === null) return null; // nothing eligible — render nothing
   444→
   445→  const { ad, clickHref } = promo;
   446→  return (
   447→    <a
   448→      ref={boxRef}
   449→      href={clickHref}
   450→      target="_blank"
   451→      rel="noopener noreferrer nofollow"
   452→      aria-label={`Sponsored: ${ad.headline} — opens the advertiser's page in a new tab`}
   453→      className="group relative flex min-h-[120px] items-center gap-3 overflow-hidden rounded-xl border border-white/10 bg-white/[0.02] p-4 transition-colors hover:border-ember/40 hover:bg-white/5 focus-visible:outline-2 focus-visible:outline-ember/60 sm:gap-4"
   454→    >
   455→      <Logo emoji={ad.emoji} gradient={ad.gradient} size="md" />
   456→      <div className="min-w-0 flex-1">
   457→        <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   458→          <span className="inline-flex shrink-0 items-center rounded-full border border-ember/50 px-1.5 py-0.5 font-mono text-[9px] uppercase tracking-[0.2em] text-ember">
   459→            Promoted
   460→          </span>
   461→          <h3 className="text-lg leading-snug font-bold text-white">{ad.headline}</h3>
   462→        </div>
   463→        {ad.body && (
   464→          <p className="mt-0.5 line-clamp-2 text-sm text-white/70">{ad.body}</p>
   465→        )}
   466→        <p className="mt-2 flex flex-wrap items-center gap-1.5 font-mono text-[11px] text-white/35">
   467→          <span>by {ad.advertiser}</span>
   468→          <span aria-hidden>·</span>
   469→          <span>Sponsored</span>
   470→          <ArrowUpRight
   471→            className="size-3.5 transition-colors group-hover:text-ember"
   472→            aria-hidden
   473→          />
   474→        </p>
   475→      </div>
   476→    </a>
   477→  );
   478→}

   480→export function LaunchFeed() {
   481→  const { toast } = useToast();
   482→  const { feed, loading, error, refresh } = useFeed();
   483→  const openTool = useExplorer((s) => s.openTool);
   484→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   485→  const categoryFilter = useExplorer((s) => s.categoryFilter);
   486→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   487→  // ?tab= persistence — shareable feed states (new | top | tomorrow | yesterday).
   488→  // Read AFTER mount only: reading window.location during the initial render makes
   489→  // the server HTML ("new") and the hydrated client (?tab=top) diverge, which
   490→  // React reports as a hydration mismatch. Post-mount sync keeps SSR markup valid.
   491→  const [tab, setTab] = useState<Tab>("new");
   492→  useEffect(() => {
   493→    const t = new URLSearchParams(window.location.search).get("tab");
   494→    if (t === "top" || t === "tomorrow" || t === "yesterday") setTab(t);
   495→  }, []);
   496→
   497→  // ── Promoted row (Task 23) ──────────────────────────────────────────
   498→  // ONE impression-counting serve per visit, lazily fired the first time a
   499→  // TODAY list (new/top) is shown — never for Tomorrow/Archive viewers.
   500→  // undefined = request in flight · null = nothing eligible.
   501→  const [promo, setPromo] = useState<ServedPromo | null | undefined>(undefined);
   502→  const promoRequestedRef = useRef(false);
   503→  useEffect(() => {
   504→    if (promoRequestedRef.current) return;
   505→    // The URL is the authority on mount — a deep link straight to
   506→    // ?tab=yesterday must not spend an impression on an archive view.
   507→    const urlTab = new URLSearchParams(window.location.search).get("tab");
   508→    const effective =
   509→      urlTab === "top" || urlTab === "tomorrow" || urlTab === "yesterday"
   510→        ? urlTab
   511→        : tab;
   512→    if (effective !== "new" && effective !== "top") return;
   513→    promoRequestedRef.current = true;
   514→    let alive = true;
   515→    fetch("/api/ads/serve?placement=feed_row", { cache: "no-store" })
   516→      .then((r) => (r.ok ? r.json() : Promise.resolve({ ad: null })))
   517→      .then((j: { ad: PromoAd | null; clickHref?: string }) => {
   518→        if (!alive) return;
   519→        setPromo(j.ad && j.clickHref ? { ad: j.ad, clickHref: j.clickHref } : null);
