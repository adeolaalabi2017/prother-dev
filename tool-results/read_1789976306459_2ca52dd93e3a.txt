    20→function formatCountdown(totalSec: number): string {
    21→  const s = Math.max(0, totalSec);
    22→  const hh = String(Math.floor(s / 3600)).padStart(2, "0");
    23→  const mm = String(Math.floor((s % 3600) / 60)).padStart(2, "0");
    24→  const ss = String(s % 60).padStart(2, "0");
    25→  return `${hh}:${mm}:${ss}`;
    26→}

    28→function pricingLabel(row: FeedRow): string {
    29→  const { model, price } = row.pricing;
    30→  switch (model) {
    31→    case "free":
    32→      return "Free";
    33→    case "freemium":
    34→      return `Freemium ${price ?? ""}`.trim();
    35→    case "paid":
    36→      return `Paid ${price ?? ""}`.trim();
    37→    case "open_source":
    38→      return "Open Source";
    39→    default:
    40→      return "Free";
    41→  }
    42→}

   333→export function LaunchFeed() {
   334→  const { toast } = useToast();
   335→  const { feed, loading, error, refresh } = useFeed();
   336→  const openTool = useExplorer((s) => s.openTool);
   337→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   338→  const categoryFilter = useExplorer((s) => s.categoryFilter);
   339→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   340→  // ?tab= persistence — shareable feed states (new | top | tomorrow | yesterday).
   341→  // Read AFTER mount only: reading window.location during the initial render makes
   342→  // the server HTML ("new") and the hydrated client (?tab=top) diverge, which
   343→  // React reports as a hydration mismatch. Post-mount sync keeps SSR markup valid.
   344→  const [tab, setTab] = useState<Tab>("new");
   345→  useEffect(() => {
   346→    const t = new URLSearchParams(window.location.search).get("tab");
   347→    if (t === "top" || t === "tomorrow" || t === "yesterday") setTab(t);
   348→  }, []);
   349→
   350→  // Archive day browser: null = yesterday (served from the main feed payload),
   351→  // otherwise an ISO day from the strip — fetched once, then cached.
   352→  const [archiveDate, setArchiveDate] = useState<string | null>(null);
   353→  const [dayCache, setDayCache] = useState<Record<string, DayArchiveResponse>>({});
   354→  const [dayLoading, setDayLoading] = useState(false);
   355→
   356→  // Fetch (and cache) a selected archive day's final standings.
   357→  useEffect(() => {
   358→    if (!archiveDate || dayCache[archiveDate]) return;
   359→    let alive = true;
   360→    setDayLoading(true);
   361→    fetch(`/api/feed/day?date=${encodeURIComponent(archiveDate)}`)
   362→      .then(async (res) => {
   363→        if (!res.ok) throw new Error("Failed to load this archive day");
   364→        return (await res.json()) as DayArchiveResponse;
   365→      })
   366→      .then((data) => {
   367→        if (alive) setDayCache((c) => ({ ...c, [data.date]: data }));
   368→      })
   369→      .catch(() => {
   370→        if (alive) setArchiveDate(null); // gracefully fall back to yesterday
   371→      })
   372→      .finally(() => {
   373→        if (alive) setDayLoading(false);
   374→      });
   375→    return () => {
   376→      alive = false;
   377→    };
   378→  }, [archiveDate, dayCache]);
   379→
   380→  // UTC rollover shifts the whole archive window — drop cached days + selection.
   381→  useEffect(() => {
   382→    setDayCache({});
   383→    setArchiveDate(null);
   384→  }, [feed?.date]);
   385→
   386→  // Keep the selected tab in the URL (replaceState — no history spam).
   387→  const setTabSync = useCallback((next: Tab) => {
   388→    setTab(next);
   389→    if (typeof window === "undefined") return;
   390→    const url = new URL(window.location.href);
   391→    const prev = url.searchParams.get("tab");
   392→    if (next === "new") {
   393→      if (prev) {
   394→        url.searchParams.delete("tab");
   395→        window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   396→      }
   397→    } else if (prev !== next) {
   398→      url.searchParams.set("tab", next);
   399→      window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   400→    }
   401→  }, []);
   402→  const [now, setNow] = useState<number>(() => Date.now());
   403→  const [voteState, setVoteState] = useState<VoteState>({});
   404→
   405→  // Countdown deadline — computed from the feed at render (no effect-synced state).
   406→  const deadlineRef = useRef<{ date: string | null; at: number }>({ date: null, at: 0 });
   407→  if (feed && deadlineRef.current.date !== feed.date) {
   408→    deadlineRef.current = { date: feed.date, at: Date.now() + feed.resetsInSec * 1000 };
   409→  }
   410→  const deadline = deadlineRef.current.at;
   411→
   412→  // Tick the clock every second (setState inside the interval callback is fine).
   413→  useEffect(() => {
   414→    const iv = setInterval(() => setNow(Date.now()), 1000);
   415→    return () => clearInterval(iv);
   416→  }, []);
   417→
   418→  // Votes cast inside the tool detail modal sync back into the feed list.
   419→  useEffect(() => {
   420→    const onModalVote = (e: Event) => {
   421→      const d = (e as CustomEvent).detail as {
   422→        launchId: string;
   423→        votes: number;
   424→        voted: boolean;
   425→      };
   426→      setVoteState((prev) => ({
   427→        ...prev,
   428→        [d.launchId]: { votes: d.votes, voted: d.voted },
   429→      }));
   430→    };
   431→    window.addEventListener("prother:vote", onModalVote);
   432→    return () => window.removeEventListener("prother:vote", onModalVote);
   433→  }, []);
   434→
   435→  const remaining = feed ? Math.max(0, Math.round((deadline - now) / 1000)) : null;
   436→
   437→  // UTC-midnight rollover: refetch the feed exactly once when the countdown hits zero.
   438→  const rolledRef = useRef(false);
   439→  useEffect(() => {
   440→    if (remaining !== null && remaining > 0) {
   441→      rolledRef.current = false;
   442→    } else if (remaining === 0 && !rolledRef.current) {
   443→      rolledRef.current = true;
   444→      void refresh();
   445→    }
   446→  }, [remaining, refresh]);
   447→
   448→  const onVote = useCallback(
   449→    async (row: FeedRow) => {
   450→      const current = voteState[row.launchId] ?? { votes: row.votes, voted: row.voted };
   451→      const nextVoted = !current.voted;
   452→      const nextVotes = current.votes + (nextVoted ? 1 : -1);
   453→      setVoteState((prev) => ({
   454→        ...prev,
   455→        [row.launchId]: { votes: nextVotes, voted: nextVoted },
   456→      }));
   457→      try {
   458→        const res = await fetch("/api/vote", {
   459→          method: "POST",
   460→          headers: { "Content-Type": "application/json" },
   461→          body: JSON.stringify({ launchId: row.launchId, voterKey: getVoterKey() }),
   462→        });
   463→        const data = (await res.json()) as { voted: boolean; votes: number };
   464→        setVoteState((prev) => ({
   465→          ...prev,
   466→          [row.launchId]: { votes: data.votes, voted: data.voted },
   467→        }));
   468→      } catch {
   469→        setVoteState((prev) => ({ ...prev, [row.launchId]: current }));
   470→        toast({ title: "Vote failed", description: "Please try again.", variant: "destructive" });
   471→      }
   472→    },
   473→    [voteState, toast]
   474→  );
   475→
   476→  const rows = useMemo<FeedRow[]>(() => {
   477→    if (!feed) return [];
   478→    const base =
   479→      tab === "top"
   480→        ? feed.top ?? []
   481→        : tab === "yesterday"
   482→          ? archiveDate
   483→            ? dayCache[archiveDate]?.rows ?? []
   484→            : feed.yesterday ?? []
   485→          : feed.new ?? [];
   486→    if (!categoryFilter) return base;
   487→    return base.filter((r) => r.category?.slug === categoryFilter);
   488→  }, [feed, tab, categoryFilter, archiveDate, dayCache]);
   489→
   490→  const tomorrowRows = useMemo(() => {
   491→    if (!feed) return [];
   492→    if (!categoryFilter) return feed.tomorrow ?? [];
   493→    return (feed.tomorrow ?? []).filter((t) => t.category?.slug === categoryFilter);
   494→  }, [feed, categoryFilter]);
   495→
   496→  const activeCategory = useMemo(
   497→    () => CATEGORIES.find((c) => c.slug === categoryFilter) ?? null,
   498→    [categoryFilter]
   499→  );
   500→
   501→  const dayLabel = feed?.dayLabel ?? "Today";
   502→  const todayCount = feed?.todayCount ?? 0;
   503→  const isTomorrow = tab === "tomorrow";
   504→  const isYesterday = tab === "yesterday";
   505→  const weekDays = feed?.weekDays ?? [];
   506→  const archiveLabel = archiveDate
   507→    ? dayCache[archiveDate]?.label ?? feed?.yesterdayLabel ?? "Archive"
   508→    : feed?.yesterdayLabel ?? "Yesterday";
   509→
   510→  const tabs: { key: Tab; label: string; short: string }[] = [
   511→    { key: "new", label: "New", short: "New" },
   512→    { key: "top", label: "Top Today", short: "Top" },
   513→    { key: "tomorrow", label: `Tomorrow (${feed?.tomorrow?.length ?? 0})`, short: `Tmrw (${feed?.tomorrow?.length ?? 0})` },
   514→    { key: "yesterday", label: `Archive (${feed?.yesterday?.length ?? 0})`, short: `Arch. (${feed?.yesterday?.length ?? 0})` },
   515→  ];
   516→
   517→  return (
   518→    <section id="feed" className="bg-ink py-24">
   519→      <div className="mx-auto max-w-6xl px-4 sm:px-6">
   520→        {/* Header */}
   521→        <motion.div
   522→          initial={{ opacity: 0, y: 20 }}
   523→          whileInView={{ opacity: 1, y: 0 }}
   524→          viewport={{ once: true, margin: "-80px" }}
   525→          transition={{ duration: 0.55, ease: "easeOut" }}
   526→          className="flex flex-wrap items-end justify-between gap-4"
