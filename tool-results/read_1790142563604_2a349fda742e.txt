    72→function ToolPicker({
    73→  label,
    74→  current,
    75→  excludeSlug,
    76→  rows,
    77→  onPick,
    78→  onClear,
    79→}: {
    80→  label: string;
    81→  current: DirectoryRow | null;
    82→  excludeSlug: string | null;
    83→  rows: DirectoryRow[];
    84→  onPick: (slug: string) => void;
    85→  onClear: () => void;
    86→}) {
    87→  const [open, setOpen] = useState(false);
    88→  const [q, setQ] = useState("");
    89→
    90→  const filtered = useMemo(() => {
    91→    const needle = q.trim().toLowerCase();
    92→    return rows
    93→      .filter((r) => r.slug !== excludeSlug)
    94→      .filter(
    95→        (r) =>
    96→          needle === "" ||
    97→          r.name.toLowerCase().includes(needle) ||
    98→          r.slug.includes(needle) ||
    99→          r.tagline.toLowerCase().includes(needle)
   100→      )
   101→      .slice(0, 40);
   102→  }, [rows, excludeSlug, q]);
   103→
   104→  return (
   105→    <div className="flex min-w-0 items-center gap-3">
   106→      <span className={cn(MONO, "w-14 shrink-0")}>{label}</span>
   107→      <Popover
   108→        open={open}
   109→        onOpenChange={(next) => {
   110→          setOpen(next);
   111→          if (next) setQ("");
   112→        }}
   113→      >
   114→        <PopoverTrigger asChild>
   115→          <button
   116→            type="button"
   117→            aria-label={current ? `Change tool ${label}` : `Pick tool for ${label}`}
   118→            className={cn(
   119→              "flex h-11 min-w-0 flex-1 items-center gap-2.5 rounded-xl border px-3 text-left transition-colors",
   120→              current
   121→                ? "border-white/10 bg-white/[0.02] hover:border-ember/40"
   122→                : "border-dashed border-white/20 text-white/60 hover:border-ember/40 hover:text-white/70"
   123→            )}
   124→          >
   125→            {current ? (
   126→              <>
   127→                <span
   128→                  aria-hidden
   129→                  className={cn(
   130→                    "grid size-7 shrink-0 place-items-center rounded-md bg-gradient-to-br text-sm",
   131→                    current.gradient
   132→                  )}
   133→                >
   134→                  {current.emoji}
   135→                </span>
   136→                <span className="min-w-0 flex-1 truncate text-sm font-semibold text-white/90">
   137→                  {current.name}
   138→                </span>
   139→              </>
   140→            ) : (
   141→              <span className="flex items-center gap-2 text-sm">
   142→                <Search className="size-4" aria-hidden /> Pick a tool…
   143→              </span>
   144→            )}
   145→          </button>
   146→        </PopoverTrigger>
   147→        <PopoverContent
   148→          align="start"
   149→          sideOffset={8}
   150→          className="w-80 border-white/10 bg-coal p-0 text-white"
   151→          aria-label={`Choose ${label}`}
   152→        >
   153→          <div className="border-b border-white/10 p-2">
   154→            <Input
   155→              value={q}
   156→              onChange={(e) => setQ(e.target.value.slice(0, 60))}
   157→              placeholder="Filter tools…"
   158→              aria-label={`Filter tools for ${label}`}
   159→              className="h-9 border-white/10 bg-transparent text-sm focus-visible:border-ember/50 focus-visible:ring-0"
   160→            />
   161→          </div>
   162→          <div className="max-h-64 overflow-y-auto p-1.5">
   163→            {filtered.map((r) => (
   164→              <button
   165→                key={r.slug}
   166→                type="button"
   167→                onClick={() => {
   168→                  onPick(r.slug);
   169→                  setOpen(false);
   170→                }}
   171→                className="flex w-full items-center gap-2.5 rounded-lg px-2 py-2 text-left transition-colors hover:bg-ember/10"
   172→              >
   173→                <span
   174→                  aria-hidden
   175→                  className={cn(
   176→                    "grid size-7 shrink-0 place-items-center rounded-md bg-gradient-to-br text-sm",
   177→                    r.gradient
   178→                  )}
   179→                >
   180→                  {r.emoji}
   181→                </span>
   182→                <span className="min-w-0 flex-1">
   183→                  <span className="block truncate text-sm font-semibold text-white/90">{r.name}</span>
   184→                  <span className="block truncate text-xs text-white/60">{r.tagline}</span>
   185→                </span>
   186→              </button>
   187→            ))}
   188→            {filtered.length === 0 && (
   189→              <p className="px-3 py-6 text-center text-xs text-white/60">
   190→                No tools match “{q}”.
   191→              </p>
   192→            )}
   193→          </div>
   194→        </PopoverContent>
   195→      </Popover>
   196→      {current && (
   197→        <Button
   198→          type="button"
   199→          variant="outline"
   200→          size="icon"
   201→          aria-label={`Clear ${label}`}
   202→          onClick={onClear}
   203→          className="bg-white/[0.03] hover:bg-white/[0.08] size-11 shrink-0 border-white/10 text-white/50 hover:border-ember/40 hover:text-white"
   204→        >
   205→          <X className="size-4" aria-hidden />
   206→        </Button>
   207→      )}
   208→    </div>
   209→  );
   210→}

   262→// ── Main page ─────────────────────────────────────────────────────────────
   263→
   264→export function CompareFullPage() {
   265→  const compareOpen = useExplorer((s) => s.compareOpen);
   266→  const compare = useExplorer((s) => s.compare);
   267→  const closeCompare = useExplorer((s) => s.closeCompare);
   268→  const openCompare = useExplorer((s) => s.openCompare);
   269→  const addCompare = useExplorer((s) => s.addCompare);
   270→  const removeCompare = useExplorer((s) => s.removeCompare);
   271→  const openCategory = useExplorer((s) => s.openCategory);
   272→
   273→  const [rows, setRows] = useState<DirectoryRow[]>([]);
   274→  const [directoryLoaded, setDirectoryLoaded] = useState(false);
   275→  const [data, setData] = useState<CompareResponse | null>(null);
   276→  /** Slug pair the stored head-to-head data was fetched for — guards staleness. */
   277→  const [dataPair, setDataPair] = useState<string[]>([]);
   278→  const [popular, setPopular] = useState<PopularPair[]>([]);
   279→  const [loading, setLoading] = useState(false);
   280→  const [loadErr, setLoadErr] = useState(false);
   281→
   282→  const a = compare[0] ?? null;
   283→  const b = compare[1] ?? null;
   284→
   285→  // Directory list powers both pickers (client-side filtering).
   286→  useEffect(() => {
   287→    if (!compareOpen || directoryLoaded) return;
   288→    let alive = true;
   289→    fetch("/api/tools?pageSize=60")
   290→      .then(async (res) => {
   291→        if (!res.ok) throw new Error("failed");
   292→        return (await res.json()) as { rows: DirectoryRow[] };
   293→      })
   294→      .then((json) => {
   295→        if (!alive) return;
   296→        setRows(json.rows ?? []);
   297→        setDirectoryLoaded(true);
   298→      })
   299→      .catch(() => {
   300→        /* pickers stay empty; the page still renders chips/slugs */
   301→        if (alive) setDirectoryLoaded(true);
   302→      });
   303→    return () => {
   304→      alive = false;
   305→    };
   306→  }, [compareOpen, directoryLoaded]);
   307→
   308→  // Head-to-head data (only when both slugs are chosen).
   309→  const loadPair = useCallback(async (aSlug: string, bSlug: string) => {
   310→    setLoading(true);
   311→    setLoadErr(false);
   312→    try {
   313→      const res = await fetch(
   314→        `/api/compare?a=${encodeURIComponent(aSlug)}&b=${encodeURIComponent(bSlug)}`
   315→      );
   316→      if (!res.ok) throw new Error("failed");
   317→      const json = (await res.json()) as CompareResponse;
   318→      setData(json);
   319→      setDataPair([aSlug, bSlug]);
   320→      setPopular(json.popular ?? []);
   321→    } catch {
   322→      setLoadErr(true);
   323→      setData(null);
   324→      setDataPair([]);
   325→    } finally {
   326→      setLoading(false);
   327→    }
   328→  }, []);
   329→
   330→  useEffect(() => {
   331→    if (!compareOpen || !a || !b) return;
   332→    // Stale responses are harmless: rendering is guarded by dataPair.
   333→    void loadPair(a, b);
   334→  }, [compareOpen, a, b, loadPair]);
   335→
   336→  // Popular strip for the picker view (when no head-to-head yet).
   337→  useEffect(() => {
   338→    if (!compareOpen || (a && b)) return;
   339→    let alive = true;
   340→    fetch("/api/compare?popular=1")
   341→      .then(async (res) => {
   342→        if (!res.ok) throw new Error("failed");
   343→        return (await res.json()) as { popular: PopularPair[] };
   344→      })
   345→      .then((json) => {
   346→        if (alive) setPopular(json.popular ?? []);
   347→      })
   348→      .catch(() => {
   349→        /* strip is optional */
   350→      });
   351→    return () => {
   352→      alive = false;
   353→    };
   354→  }, [compareOpen, a, b]);
   355→
   356→  const rowOf = useCallback(
   357→    (slug: string | null) => rows.find((r) => r.slug === slug) ?? null,
   358→    [rows]
   359→  );
   360→
   361→  /** Head-to-head data matches the currently selected pair. */
   362→  const dataValid =
   363→    !!data && dataPair.length === 2 && dataPair[0] === a && dataPair[1] === b;
   364→
   365→  const pickA = useCallback(
   366→    (slug: string) => {
   367→      if (b) openCompare(slug, b);
   368→      else {
   369→        if (a) removeCompare(a);
   370→        addCompare(slug);
   371→      }
   372→    },
   373→    [a, b, openCompare, removeCompare, addCompare]
   374→  );
   375→
   376→  const pickB = useCallback(
   377→    (slug: string) => {
   378→      if (a) openCompare(a, slug);
   379→      else addCompare(slug);
   380→    },
   381→    [a, openCompare, addCompare]
   382→  );
   383→
   384→  const shareUrl =
   385→    typeof window !== "undefined"
   386→      ? `${window.location.origin}/?compare=${encodeURIComponent(a ?? "")},${encodeURIComponent(b ?? "")}`
   387→      : `/?compare=${encodeURIComponent(a ?? "")},${encodeURIComponent(b ?? "")}`;
   388→
   389→  const verdict = useMemo(() => {
   390→    if (!dataValid) return null;
   391→    const a = data!.a.rating;
   392→    const b = data!.b.rating;
   393→    // No vote winner — the honest verdict is the review aggregate (when both
   394→    // tools have one). Otherwise no banner; the table below does the talking.
   395→    if (!a || !b) return null;
   396→    const diff = Math.round((a.overall - b.overall) * 10) / 10;
   397→    if (diff === 0) return "EVEN ON REVIEW RATING";
   398→    const winner = diff > 0 ? data!.a.name : data!.b.name;
   399→    return `${winner.toUpperCase()} LEADS · BY REVIEW RATING (+${Math.abs(diff).toFixed(1)})`;
   400→  }, [data, dataValid]);
   401→
   402→  if (!compareOpen) return null;
   403→
   404→  const rowA = dataValid ? data!.a : null;
   405→  const rowB = dataValid ? data!.b : null;
   406→  const hasPair = !!a && !!b;
   407→
   408→  return (
   409→    <FullPageShell
   410→      kicker="Side-by-side"
   411→      breadcrumb={[{ label: "Home" }, { label: "Compare" }]}
   412→      onClose={closeCompare}
   413→      shareUrl={hasPair ? shareUrl : undefined}
   414→      wide
   415→      ariaLabel="Compare tools side by side"
   416→    >
   417→      <div className="space-y-8">
   418→        {/* Pickers */}
   419→        <section aria-label="Choose tools to compare" className="space-y-3">
   420→          <div className="grid gap-3 sm:grid-cols-2">
   421→            <ToolPicker
   422→              label="TOOL A"
   423→              current={rowOf(a)}
   424→              excludeSlug={b}
   425→              rows={rows}
   426→              onPick={pickA}
   427→              onClear={() => a && removeCompare(a)}
   428→            />
   429→            <ToolPicker
   430→              label="TOOL B"
   431→              current={rowOf(b)}
   432→              excludeSlug={a}
   433→              rows={rows}
   434→              onPick={pickB}
   435→              onClear={() => b && removeCompare(b)}
   436→            />
   437→          </div>
   438→          {!hasPair && (
   439→            <p className="font-mono text-xs uppercase tracking-[0.2em] text-white/60">
   440→              {a || b
   441→                ? "Pick the second tool to unlock the side-by-side."
   442→                : "Pick two tools to compare, or start from a popular matchup below."}
   443→            </p>
   444→          )}
   445→        </section>
   446→
   447→        {/* Loading */}
   448→        {hasPair && loading && <PageSkeleton />}
   449→
   450→        {/* Load failure — pickers stay usable */}
   451→        {hasPair && !loading && (loadErr || !rowA || !rowB) && (
   452→          <p className="rounded-xl border border-red-500/25 bg-red-500/[0.04] p-4 text-sm text-red-300">
   453→            Couldn&apos;t load the comparison for this pair. Try different tools.
   454→          </p>
   455→        )}
   456→
   457→        {/* Comparison grid */}
   458→        {hasPair && !loading && rowA && rowB && (
   459→          <section aria-label="Comparison" className="space-y-6">
   460→            {/* Header cards */}
   461→            <div className="grid gap-4 sm:grid-cols-2">
   462→              {[rowA, rowB].map((r, i) => (
   463→                <div
   464→                  key={r.slug}
   465→                  className="flex items-center gap-4 rounded-xl border border-white/10 bg-white/[0.02] p-4 transition-colors hover:border-ember/40"
   466→                >
   467→                  <span
   468→                    aria-hidden
   469→                    className={cn(
   470→                      "grid size-14 shrink-0 place-items-center rounded-2xl bg-gradient-to-br text-3xl",
   471→                      r.gradient
   472→                    )}
   473→                  >
   474→                    {r.emoji}
   475→                  </span>
   476→                  <div className="min-w-0 flex-1">
   477→                    <div className="flex items-center gap-2">
   478→                      <h2 className="truncate text-xl font-black text-white">{r.name}</h2>
   479→                      <span className="shrink-0 rounded-full border border-white/15 px-1.5 py-px font-mono text-xs tracking-wider text-white/50">
   480→                        {i === 0 ? "TOOL A" : "TOOL B"}
   481→                      </span>
   482→                    </div>
   483→                    <p className="mt-0.5 truncate text-sm text-white/55">{r.tagline}</p>
   484→                  </div>
   485→                  <span className="shrink-0 rounded-full bg-ember/10 px-3 py-1.5 font-mono text-sm font-black tabular-nums text-ember">
   486→                    {r.rating != null ? (
   487→                      <>★ {r.rating.overall.toFixed(1)}/5</>
   488→                    ) : (
   489→                      <>{r.reviewCount} reviews</>
   490→                    )}
   491→                  </span>
   492→                </div>
   493→              ))}
   494→            </div>
   495→
   496→            {/* Auto-verdict */}
   497→            {verdict && (
   498→              <p className="text-center font-mono text-xs uppercase tracking-[0.25em] text-white/50" aria-live="polite">
   499→                {verdict}
   500→              </p>
   501→            )}
   502→
   503→            {/* Metric table */}
   504→            <div className="rounded-xl border border-white/10 bg-white/[0.02] px-4 py-1 sm:px-5">
   505→              <MetricRow
   506→                label="Rating"
   507→                a={
   508→                  rowA.rating != null ? (
   509→                    <span className="tabular-nums">
   510→                      {rowA.rating.overall.toFixed(1)}{" "}
   511→                      <span className="font-mono text-xs text-white/60">({rowA.rating.count})</span>
   512→                    </span>
   513→                  ) : (
   514→                    <span className="font-mono text-xs text-white/60">
   515→                      UNRATED · {rowA.reviewCount} reviews
   516→                    </span>
   517→                  )
   518→                }
   519→                b={
   520→                  rowB.rating != null ? (
   521→                    <span className="tabular-nums">
   522→                      {rowB.rating.overall.toFixed(1)}{" "}
   523→                      <span className="font-mono text-xs text-white/60">({rowB.rating.count})</span>
   524→                    </span>
   525→                  ) : (
   526→                    <span className="font-mono text-xs text-white/60">
   527→                      UNRATED · {rowB.reviewCount} reviews
   528→                    </span>
   529→                  )
   530→                }
   531→              />
   532→              {rowA.rating?.ease != null && rowB.rating?.ease != null && (
   533→                <MetricRow label="Ease" a={<MiniBar value={rowA.rating.ease} />} b={<MiniBar value={rowB.rating.ease} />} />
   534→              )}
   535→              {rowA.rating?.power != null && rowB.rating?.power != null && (
   536→                <MetricRow label="Power" a={<MiniBar value={rowA.rating.power} />} b={<MiniBar value={rowB.rating.power} />} />
   537→              )}
   538→              {rowA.rating?.value != null && rowB.rating?.value != null && (
   539→                <MetricRow label="Value" a={<MiniBar value={rowA.rating.value} />} b={<MiniBar value={rowB.rating.value} />} />
   540→              )}
   541→              <MetricRow
   542→                label="Pricing"
   543→                a={
   544→                  <span>
   545→                    <span className="font-semibold text-white">
   546→                      {PRICING_LABEL[rowA.pricing.model] ?? "Free"}
   547→                    </span>
   548→                    {rowA.pricing.price && (
   549→                      <span className="ml-1.5 font-mono text-xs text-ember">{rowA.pricing.price}</span>
   550→                    )}
   551→                  </span>
   552→                }
   553→                b={
   554→                  <span>
   555→                    <span className="font-semibold text-white">
   556→                      {PRICING_LABEL[rowB.pricing.model] ?? "Free"}
   557→                    </span>
