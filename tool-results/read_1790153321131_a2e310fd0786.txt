   120→function fmtDay(iso: string | null): string {
   121→  if (!iso) return "—";
   122→  return new Date(iso).toLocaleDateString("en-US", {
   123→    month: "short",
   124→    day: "numeric",
   125→    timeZone: "UTC",
   126→  });
   127→}
   128→
   129→function relTime(iso: string): string {
   130→  const s = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000);
   131→  if (s < 60) return "just now";
   132→  const m = Math.floor(s / 60);
   133→  if (m < 60) return `${m}m ago`;
   134→  const h = Math.floor(m / 60);
   135→  if (h < 24) return `${h}h ago`;
   136→  return `${Math.floor(h / 24)}d ago`;
   137→}

   286→  "Add the meta tag below to your site's <head> section.",

   291→function ClaimPanel({
   292→  slug,
   293→  maker,
   294→  claim,
   295→  sessionStatus,
   296→  onRefresh,
   297→}: {
   298→  slug: string;
   299→  maker: string;
   300→  claim: StartedClaim | null;
   301→  sessionStatus: "loading" | "authenticated" | "unauthenticated";
   302→  onRefresh: () => void;
   303→}) {
   304→  const { toast } = useToast();
   305→  const [starting, setStarting] = useState(false);
   306→  const [verifying, setVerifying] = useState(false);
   307→  const [copied, setCopied] = useState(false);
   308→  const [failReason, setFailReason] = useState<string | null>(null);
   309→  const claimRef = useRef(claim);
   310→  claimRef.current = claim;
   311→
   312→  const metaTag = claim
   313→    ? `<meta name="prother-verify" content="${claim.token}" />`
   314→    : "";
   315→
   316→  const startClaim = useCallback(async () => {
   317→    setStarting(true);
   318→    setFailReason(null);
   319→    try {
   320→      const res = await fetch("/api/claims", {
   321→        method: "POST",
   322→        headers: { "Content-Type": "application/json" },
   323→        body: JSON.stringify({ toolSlug: slug }),
   324→      });
   325→      const data = (await res.json().catch(() => ({}))) as {
   326→        claim?: StartedClaim;
   327→        verified?: boolean;
   328→        error?: string;
   329→      };
   330→      if (res.status === 409) {
   331→        toast({ title: "This listing is already claimed." });
   332→        onRefresh();
   333→        return;
   334→      }
   335→      if (!res.ok || !data.claim) {
   336→        toast({
   337→          title: "Could not start the claim",
   338→          description: data.error ?? "Please try again.",
   339→          variant: "destructive",
   340→        });
   341→        return;
   342→      }
   343→      toast({ title: "Claim started: add the meta tag to your site." });
   344→      onRefresh();
   345→    } catch {
   346→      toast({
   347→        title: "Could not start the claim",
   348→        description: "Please try again.",
   349→        variant: "destructive",
   350→      });
   351→    } finally {
   352→      setStarting(false);
   353→    }
   354→  }, [slug, toast, onRefresh]);
   355→
   356→  const onVerify = useCallback(async () => {
   357→    const c = claimRef.current;
   358→    if (!c) return;
   359→    setVerifying(true);
   360→    setFailReason(null);
   361→    try {
   362→      const res = await fetch(`/api/claims/${encodeURIComponent(c.id)}/verify`, {
   363→        method: "POST",
   364→      });
   365→      const data = (await res.json().catch(() => ({}))) as {
   366→        verified?: boolean;
   367→        reason?: string;
   368→        claim?: StartedClaim;
   369→      };
   370→      if (data.verified) {
   371→        toast({
   372→          title: "Ownership verified ✓",
   373→          description: "This listing is yours.",
   374→        });
   375→        onRefresh();
   376→      } else {
   377→        setFailReason(data.reason ?? "We couldn't find the meta tag on your site yet.");
   378→      }
   379→    } catch {
   380→      setFailReason("Verification request failed. Please try again.");
   381→    } finally {
   382→      setVerifying(false);
   383→    }
   384→  }, [toast, onRefresh]);
   385→
   386→  const onCopyTag = useCallback(async () => {
   387→    try {
   388→      await navigator.clipboard.writeText(metaTag);
   389→      setCopied(true);
   390→      window.setTimeout(() => setCopied(false), 1600);
   391→    } catch {
   392→      toast({ title: "Could not copy. Select the tag manually.", variant: "destructive" });
   393→    }
   394→  }, [metaTag, toast]);
   395→
   396→  // Anonymous: sign-in gate.
   397→  if (sessionStatus === "unauthenticated") {
   398→    return (
   399→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border border-ember/30 bg-ember/[0.05] p-4">
   400→        <p className="text-sm font-semibold text-white/85">
   401→          Sign in to prove ownership of this listing.
   402→        </p>
   403→        <Button
   404→          type="button"
   405→          onClick={openAuth}
   406→          className="h-10 bg-ember font-mono text-sm font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   407→        >
   408→          SIGN IN TO CLAIM →
   409→        </Button>
   410→      </div>
   411→    );
   412→  }
   413→
   414→  // No claim yet (authenticated): start one.
   415→  if (!claim) {
   416→    return (
   417→      <div className="rounded-xl border border-ember/30 bg-ember/[0.05] p-4">
   418→        <p className="text-sm text-white/70">
   419→          Start a claim: we&apos;ll generate a verification meta tag for{" "}
   420→          <span className="text-white">{maker}</span> to place on the product
   421→          site.
   422→        </p>
   423→        <Button
   424→          type="button"
   425→          onClick={() => void startClaim()}
   426→          disabled={starting}
   427→          className="mt-3 h-10 bg-ember font-mono text-sm font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   428→        >
   429→          {starting ? (
   430→            <>
   431→              <Loader2 className="size-3.5 animate-spin" aria-hidden /> STARTING…
   432→            </>
   433→          ) : (
   434→            "START CLAIM →"
   435→          )}
   436→        </Button>
   437→      </div>
   438→    );
   439→  }
   440→
   441→  // Claim in progress / verified / failed.
   442→  const verified = claim.status === "verified" || !!claim.verifiedAt;
   443→  return (
   444→    <div
   445→      className={cn(
   446→        "rounded-xl border p-4",
   447→        verified ? "border-mint/40 bg-mint/[0.05]" : "border-ember/30 bg-ember/[0.05]"
   448→      )}
   449→      aria-live="polite"
   450→    >
   451→      <div className="flex flex-wrap items-center gap-2">
   452→        <p className={MONO}>OWNERSHIP CLAIM</p>
   453→        <span
   454→          className={cn(
   455→            "rounded-full border px-2 py-0.5 font-mono text-xs tracking-wider",
   456→            verified
   457→              ? "border-mint/40 bg-mint/10 text-mint"
   458→              : claim.status === "failed"
   459→                ? "border-red-500/40 bg-red-500/10 text-red-400"
   460→                : "border-white/20 bg-white/5 text-white/70"
   461→          )}
   462→        >
   463→          {verified ? "VERIFIED" : claim.status.toUpperCase()}
   464→        </span>
   465→      </div>
   466→
   467→      {verified ? (
   468→        <p className="mt-2 text-sm text-mint">
   469→          ✓ Verified: this listing is yours.
   470→        </p>
   471→      ) : (
   472→        <>
   473→          <ol className="mt-3 space-y-1.5">
   474→            {DEFAULT_CLAIM_STEPS.map((step, i) => (
   475→              <li key={i} className="flex gap-2 text-xs leading-relaxed text-white/65">
   476→                <span className="font-mono text-ember">{i + 1}.</span>
   477→                {step}
   478→              </li>
   479→            ))}
   480→          </ol>
   481→
   482→          <div className="mt-3 flex items-start gap-2">
   483→            <code className="min-w-0 flex-1 break-all rounded-lg border border-white/10 bg-black/40 p-2.5 font-mono text-xs text-white/80">
   484→              {metaTag}
   485→            </code>
   486→            <Button
   487→              type="button"
   488→              variant="outline"
   489→              size="icon"
   490→              aria-label="Copy the verification meta tag"
   491→              onClick={() => void onCopyTag()}
   492→              className={cn(
   493→                "bg-white/[0.03] hover:bg-white/[0.08] size-9 shrink-0 border-white/10",
   494→                copied ? "text-mint" : "text-white/70 hover:border-ember/40 hover:text-ember"
   495→              )}
   496→            >
   497→              {copied ? <Check className="size-4" aria-hidden /> : <Copy className="size-4" aria-hidden />}
   498→            </Button>
   499→          </div>
   500→
   501→          {failReason && (
   502→            <p className="mt-3 rounded-lg border border-red-500/25 bg-red-500/[0.06] p-2.5 text-xs text-red-300">
   503→              {failReason}
   504→            </p>
   505→          )}
   506→
   507→          <div className="mt-3 flex flex-wrap items-center gap-3">
   508→            <Button
   509→              type="button"
   510→              onClick={() => void onVerify()}
   511→              disabled={verifying}
   512→              className="h-10 bg-ember font-mono text-sm font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot"
   513→            >
   514→              {verifying ? (
   515→                <>
   516→                  <Loader2 className="size-3.5 animate-spin" aria-hidden /> VERIFYING…
   517→                </>
   518→              ) : (
   519→                "VERIFY NOW"
   520→              )}
   521→            </Button>
   522→            <p className="font-mono text-xs uppercase tracking-[0.2em] text-white/55">
   523→              False claims fail by design · disputes → editor arbitration
   524→            </p>
   525→          </div>
   526→        </>
   527→      )}
   528→    </div>
   529→  );
   530→}

   763→        </p>

   771→            (data?.reviews.length ?? 0) > 4 && "max-h-96 overflow-y-auto pr-1"

   774→        >
   775→          {data!.reviews.map((r) => (

   777→              <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   778→                <span className="font-mono text-xs font-semibold text-white/90">

   780→                </span>
   781→                {r.status === "filtered" && r.mine && (

   824→      ) : data?.reason === "maker" || isMaker ? (

   826→          Makers can&apos;t review their own product
   827→        </p>

   838→              id="review-body"

   845→            <div className="mt-1 flex items-center justify-between">
   846→              <span className="font-mono text-xs text-white/55">

   849→              <span className={cn("font-mono text-xs tabular-nums", counterTone)} aria-live="polite">
   850→                {body.length}/{REVIEW_BODY_MAX}
   851→              </span>
   852→            </div>
   853→          </div>
   854→          {errors.length > 0 && (
   855→            <ul className="space-y-1 rounded-lg border border-red-500/25 bg-red-500/[0.06] p-3" aria-live="polite">

   858→                  {e}
   859→                </li>
   860→              ))}
   861→            </ul>
   862→          )}
   863→          <Button
   864→            type="button"
   865→            onClick={() => void submit()}
   866→            disabled={submitting}
   867→            className="h-11 w-full bg-ember font-mono text-sm font-black tracking-wider text-[#0A0A0A] hover:bg-ember-hot sm:w-auto"
   868→          >

   872→              </>
   873→            ) : isUpdate ? (
   874→              "UPDATE REVIEW"
   875→            ) : (
   876→              "POST REVIEW"
   877→            )}
   878→          </Button>
   879→        </div>
   880→      ) : null}
   881→    </section>
   882→  );
   883→}

   885→function slugToName(slug: string): string {
   886→  return slug
   887→    .split("-")
   888→    .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
   889→    .join(" ");

   910→function avatarGradient(author: string): string {

  1143→function SavePopover({
  1144→  slug,
  1145→  toolName,
  1146→  savedIn,
  1147→  onSavedChange,
  1148→  children,
  1149→}: {
  1150→  slug: string;
  1151→  toolName: string;
  1152→  savedIn: { slug: string; name: string }[];
  1153→  onSavedChange: (saved: { slug: string; name: string }[]) => void;
  1154→  children: React.ReactNode;
  1155→}) {
  1156→  const { toast } = useToast();
  1157→  const { status } = useSession();
  1158→  const [open, setOpen] = useState(false);
  1159→  const [colls, setColls] = useState<CollectionRow[] | null>(null);

  1162→  const [creating, setCreating] = useState(false);
  1163→
  1164→  const savedSet = useMemo(() => new Set(savedIn.map((s) => s.slug)), [savedIn]);
  1165→
  1166→  const onOpenChange = useCallback(
  1167→    (next: boolean) => {
  1168→      if (next && status === "unauthenticated") {
  1169→        openAuth();
  1170→        toast({ title: "Sign in to save tools to collections." });
  1171→        return;
  1172→      }
  1173→      setOpen(next);
  1174→      if (next) {
  1175→        setColls(null);
  1176→        fetch("/api/collections")
  1177→          .then(async (res) => {
  1178→            if (!res.ok) throw new Error("failed");
  1179→            return (await res.json()) as { mine: CollectionRow[] };
  1180→          })
  1181→          .then((data) => setColls(data.mine ?? []))
  1182→          .catch(() => setColls([]));
  1183→      }
  1184→    },
  1185→    [status, toast]
  1186→  );
  1187→
  1188→  const toggle = useCallback(
  1189→    async (c: CollectionRow) => {
  1190→      setBusySlug(c.slug);
  1191→      try {
  1192→        const res = await fetch(`/api/collections/${encodeURIComponent(c.slug)}/items`, {
  1193→          method: "POST",
  1194→          headers: { "Content-Type": "application/json" },
  1195→          body: JSON.stringify({ toolSlug: slug }),
  1196→        });
  1197→        if (res.status === 401) {
  1198→          openAuth();
  1199→          return;
  1200→        }
  1201→        const data = (await res.json().catch(() => ({}))) as { inCollection?: boolean };
  1202→        const nowIn = !!data.inCollection;
  1203→        const next = nowIn
  1204→          ? [...savedIn, { slug: c.slug, name: c.name }]
  1205→          : savedIn.filter((s) => s.slug !== c.slug);
  1206→        onSavedChange(next);
  1207→        toast({
  1208→          title: nowIn ? `Saved to ${c.name}` : `Removed from ${c.name}`,
  1209→        });
  1210→      } catch {
  1211→        toast({ title: "Could not update the collection", variant: "destructive" });
  1212→      } finally {
  1213→        setBusySlug(null);
  1214→      }
  1215→    },
  1216→    [slug, savedIn, onSavedChange, toast]
  1217→  );
  1218→
  1219→  const create = useCallback(async () => {
  1220→    const nameTrimmed = newName.trim();
  1221→    if (nameTrimmed.length < 2 || creating) return;
  1222→    setCreating(true);
  1223→    try {
  1224→      const res = await fetch("/api/collections", {
  1225→        method: "POST",
  1226→        headers: { "Content-Type": "application/json" },
  1227→        body: JSON.stringify({ name: nameTrimmed }),
  1228→      });
  1229→      if (res.status === 401) {
  1230→        openAuth();
  1231→        return;
  1232→      }
  1233→      const data = (await res.json().catch(() => ({}))) as { collection?: CollectionRow };
  1234→      if (!res.ok || !data.collection) {
  1235→        toast({ title: "Could not create the collection", variant: "destructive" });
  1236→        return;
  1237→      }
  1238→      setColls((prev) => [data.collection!, ...(prev ?? [])]);
  1239→      setNewName("");

  1241→      const c = data.collection;
  1242→      try {
  1243→        const itemRes = await fetch(`/api/collections/${encodeURIComponent(c.slug)}/items`, {
  1244→          method: "POST",
  1245→          headers: { "Content-Type": "application/json" },
  1246→          body: JSON.stringify({ toolSlug: slug }),
  1247→        });
  1248→        if (itemRes.ok) {
  1249→          onSavedChange([...savedIn, { slug: c.slug, name: c.name }]);
  1250→        }
  1251→      } catch {
  1252→        /* collection exists; saving is best-effort */
  1253→      }
  1254→      toast({ title: `Collection "${c.name}" created` });
  1255→    } catch {
  1256→      toast({ title: "Could not create the collection", variant: "destructive" });
  1257→    } finally {
  1258→      setCreating(false);
  1259→    }
  1260→  }, [newName, creating, slug, savedIn, onSavedChange, toast]);
  1261→
  1262→  return (
  1263→    <Popover open={open} onOpenChange={onOpenChange}>
  1264→      <PopoverTrigger asChild>{children}</PopoverTrigger>
  1265→      <PopoverContent
  1266→        align="end"
  1267→        sideOffset={8}
  1268→        className="w-72 border-white/10 bg-coal p-3 text-white"
  1269→        aria-label={`Save ${toolName} to a collection`}
  1270→      >
  1271→        <p className={MONO}>Save to collection</p>
  1272→        <div className="mt-2 max-h-64 space-y-1 overflow-y-auto">
  1273→          {colls === null && (
  1274→            <>
  1275→              <Skeleton className="h-10 w-full rounded-lg bg-white/5" />
  1276→              <Skeleton className="h-10 w-5/6 rounded-lg bg-white/5" />
  1277→            </>
  1278→          )}
  1279→          {colls !== null && colls.length === 0 && (
  1280→            <p className="px-1 py-2 text-xs text-white/60">
  1281→              No collections yet. Create your first one below.
  1282→            </p>
  1283→          )}
  1284→          {colls?.map((c) => {
  1285→            const saved = savedSet.has(c.slug);
  1286→            return (
  1287→              <button
  1288→                key={c.id}
  1289→                type="button"
  1290→                onClick={() => void toggle(c)}
  1291→                disabled={busySlug === c.slug}
  1292→                aria-pressed={saved}
  1293→                className="flex w-full items-center gap-2.5 rounded-lg px-2 py-2.5 text-left transition-colors hover:bg-white/5"
  1294→              >
  1295→                <span className="min-w-0 flex-1 truncate text-sm text-white/85">{c.name}</span>
  1296→                <span className="font-mono text-xs text-white/55">{c.itemCount}</span>
  1297→                <span
  1298→                  aria-hidden
  1299→                  className={cn(
  1300→                    "flex size-5 items-center justify-center rounded border",
  1301→                    saved ? "border-ember bg-ember/15 text-ember" : "border-white/20 text-transparent"
  1302→                  )}
  1303→                >
  1304→                  <Check className="size-3" />
  1305→                </span>
  1306→              </button>
  1307→            );
  1308→          })}
  1309→        </div>
  1310→        <div className="mt-2 flex items-center gap-2 border-t border-white/10 pt-2.5">
  1311→          <label htmlFor="new-collection-name" className="sr-only">
  1312→            New collection name

  1314→          <Input
  1315→            id="new-collection-name"
  1316→            value={newName}
  1317→            onChange={(e) => setNewName(e.target.value.slice(0, 40))}
  1318→            onKeyDown={(e) => {
  1319→              if (e.key === "Enter") {
  1320→                e.preventDefault();
  1321→                void create();
  1322→              }
  1323→            }}
  1324→            placeholder="+ New collection"
  1325→            className="h-9 border-white/10 bg-transparent text-sm focus-visible:border-ember/50 focus-visible:ring-0"
  1326→          />
  1327→          <Button
  1328→            type="button"
  1329→            size="icon"
  1330→            variant="outline"
  1331→            aria-label="Create collection"
  1332→            onClick={() => void create()}
  1333→            disabled={newName.trim().length < 2 || creating}

  1337→          </Button>
  1338→        </div>
