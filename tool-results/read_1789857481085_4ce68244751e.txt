    52→function fmtDate(iso: string | null): string {
    53→  if (!iso) return "—";
    54→  return new Date(iso).toLocaleDateString("en-US", {
    55→    month: "short",
    56→    day: "numeric",
    57→    timeZone: "UTC",
    58→  });
    59→}

    98→function DetailBody({ slug, onClose }: { slug: string; onClose: () => void }) {
    99→  const { toast } = useToast();
   100→  const openTool = useExplorer((s) => s.openTool);
   101→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   102→  const [detail, setDetail] = useState<ToolDetailResponse | null>(null);
   103→  const [loading, setLoading] = useState(true);
   104→  const [error, setError] = useState<string | null>(null);
   105→  const [vote, setVote] = useState<{ votes: number; voted: boolean } | null>(
   106→    null,
   107→  );
   108→  const [copied, setCopied] = useState(false);
   109→
   110→  useEffect(() => {
   111→    let alive = true;
   112→    fetch(`/api/tools/${slug}?vk=${encodeURIComponent(getVoterKey())}`)
   113→      .then(async (res) => {
   114→        if (!res.ok)
   115→          throw new Error(
   116→            res.status === 404 ? "Tool not found" : "Failed to load tool",
   117→          );
   118→        return (await res.json()) as ToolDetailResponse;
   119→      })
   120→      .then((data) => {
   121→        if (alive) {
   122→          setDetail(data);
   123→          setVote({ votes: data.votes, voted: data.voted });
   124→        }
   125→      })
   126→      .catch((err: unknown) => {
   127→        if (alive)
   128→          setError(err instanceof Error ? err.message : "Failed to load tool");
   129→      })
   130→      .finally(() => {
   131→        if (alive) setLoading(false);
   132→      });
   133→    return () => {
   134→      alive = false;
   135→    };
   136→  }, [slug]);
   137→
   138→  const onVote = useCallback(async () => {
   139→    if (!detail?.launchId || !vote) return;
   140→    const nextVoted = !vote.voted;
   141→    setVote({ votes: vote.votes + (nextVoted ? 1 : -1), voted: nextVoted });
   142→    try {
   143→      const res = await fetch("/api/vote", {
   144→        method: "POST",
   145→        headers: { "Content-Type": "application/json" },
   146→        body: JSON.stringify({
   147→          launchId: detail.launchId,
   148→          voterKey: getVoterKey(),
   149→        }),
   150→      });
   151→      const data = (await res.json()) as { voted: boolean; votes: number };
   152→      setVote({ votes: data.votes, voted: data.voted });
   153→      // Sync the feed list behind the modal.
   154→      window.dispatchEvent(
   155→        new CustomEvent("prother:vote", {
   156→          detail: { launchId: detail.launchId, votes: data.votes, voted: data.voted },
   157→        }),
   158→      );
   159→    } catch {
   160→      setVote(vote);
   161→      toast({
   162→        title: "Vote failed",
   163→        description: "Please try again.",
   164→        variant: "destructive",
   165→      });
   166→    }
   167→  }, [detail, vote, toast]);
   168→
   169→  const passedCount = detail?.standards.filter((s) => s.passed).length ?? 0;
   170→
   171→  const shareUrl =
   172→    typeof window !== "undefined"
   173→      ? `${window.location.origin}/${toolHash(slug)}`
   174→      : "";
   175→  const shareText = detail
   176→    ? `🚀 ${detail.name} — ${detail.tagline} is on Prother`
   177→    : "";
   178→
   179→  const onCopyLink = useCallback(async () => {
   180→    try {
   181→      await navigator.clipboard.writeText(shareUrl);
   182→      setCopied(true);
   183→      window.setTimeout(() => setCopied(false), 1600);
   184→    } catch {
   185→      toast({
   186→        title: "Could not copy the link",
   187→        description: shareUrl,
   188→        variant: "destructive",
   189→      });
   190→    }
   191→  }, [shareUrl, toast]);
   192→
   193→  // Cross-link: close the modal, filter the feed to this category, scroll there.
   194→  const browseCategory = useCallback(() => {
   195→    if (!detail) return;
   196→    const cat = detail.category.slug;
   197→    onClose();
   198→    window.setTimeout(() => {
   199→      setCategoryFilter(cat);
   200→      document
   201→        .querySelector("#feed")
   202→        ?.scrollIntoView({ behavior: "smooth" });
   203→    }, 80);
   204→  }, [detail, onClose, setCategoryFilter]);
   205→
   206→  return (
   207→    <>
   208→      {loading && <DetailSkeleton />}
   209→
   210→      {!loading && error && (
   211→        <div className="p-10 text-center">
   212→          <p className="text-sm text-white/60">{error}</p>
   213→          <button
   214→            type="button"
   215→            onClick={onClose}
   216→            className="mt-4 font-mono text-xs text-ember hover:underline"
   217→          >
   218→            ← Close
   219→          </button>
   220→        </div>
   221→      )}
   222→
   223→      {!loading && !error && detail && vote && (
   224→        <>
   225→          {/* Header with gradient banner */}
   226→          <div className="relative">
   227→            <div
   228→              aria-hidden
   229→              className={cn(
   230→                "h-20 w-full bg-gradient-to-br opacity-30",
   231→                detail.gradient,
   232→              )}
   233→            />
   234→            <div className="flex items-end gap-4 px-6 pb-4">
   235→              <div
   236→                aria-hidden
   237→                className={cn(
   238→                  "-mt-9 flex size-16 shrink-0 items-center justify-center rounded-2xl border-4 border-coal bg-gradient-to-br text-2xl shadow-xl",
   239→                  detail.gradient,
   240→                )}
   241→              >
   242→                {detail.emoji}
   243→              </div>
   244→              <div className="min-w-0 pb-1">
   245→                <div className="flex flex-wrap items-center gap-2">
   246→                  <h2 className="text-2xl font-black tracking-tight">
   247→                    {detail.name}
   248→                  </h2>
   249→                  {detail.badges.editorsPick && (
   250→                    <span className="rounded-full border border-ember/30 bg-ember/15 px-2 py-0.5 font-mono text-[10px] text-ember">
   251→                      ⭐ EDITOR&apos;S PICK
   252→                    </span>
   253→                  )}
   254→                  {detail.badges.curated && (
   255→                    <span className="rounded-full border border-yellow-500/30 bg-yellow-500/10 px-2 py-0.5 font-mono text-[10px] text-yellow-500">
   256→                      CURATED
   257→                    </span>
   258→                  )}
   259→                  {detail.badges.relaunch && (
   260→                    <span className="rounded-full border border-white/15 bg-white/5 px-2 py-0.5 font-mono text-[10px] text-white/60">
   261→                      🔁 RE-LAUNCH

   312→                >

   319→                  </svg>

   322→              <button

   325→                aria-label="Copy link to this tool"

   327→                  "inline-flex size-10 items-center justify-center rounded-lg border border-white/10 transition-colors",

   338→              </button>

   341→                target="_blank"

   345→              >
   346→                <Share2 className="size-4" aria-hidden />

   349→                <button

   356→                    "inline-flex h-10 items-center gap-2 rounded-lg border px-4 font-mono text-sm font-semibold tabular-nums transition active:scale-95",

   368→                </button>

   370→            </div>
   371→
   372→            {/* Meta grid */}

   801→// ── Mounted-once explorer + ⌘K shortcut ──────────────────────────────────
   802→
   803→export function ToolExplorer() {
   804→  const setSearch = useExplorer((s) => s.setSearch);
   805→  const openTool = useExplorer((s) => s.openTool);
   806→
   807→  // ⌘K / ctrl+K opens the command palette.
   808→  useEffect(() => {
   809→    const onKey = (e: KeyboardEvent) => {
   810→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   811→        e.preventDefault();
   812→        setSearch(true);
   813→      }
   814→    };
   815→    window.addEventListener("keydown", onKey);
   816→    return () => window.removeEventListener("keydown", onKey);
   817→  }, [setSearch]);
   818→
   819→  // Shareable deep links: /#tool=<slug> opens the detail modal on load —
   820→  // and keeps working when the hash is navigated to *after* load
   821→  // (same-document navigation, pasted links, back/forward).
   822→  // Clearing the hash (e.g. browser Back) closes the modal again.
   823→  useEffect(() => {
   824→    const applyHash = (initial = false) => {
   825→      const m = window.location.hash.match(/^#tool=([^&]+)$/);
   826→      if (m?.[1]) openTool(decodeURIComponent(m[1]));
   827→      else if (!initial) useExplorer.getState().closeTool();
   828→    };
   829→    applyHash(true);
   830→    const onHash = () => applyHash(false);
   831→    window.addEventListener("hashchange", onHash);
   832→    return () => window.removeEventListener("hashchange", onHash);
   833→  }, [openTool]);
   834→
   835→  // Restore a shared/filtered category from the URL on load (?cat=<slug>)
   836→  // and jump straight to the feed — a shared ?cat= link is an intent to browse.
   837→  useEffect(() => {
   838→    const cat = new URLSearchParams(window.location.search).get("cat");
   839→    if (cat && CATEGORIES.some((c) => c.slug === cat)) {
   840→      useExplorer.getState().setCategoryFilter(cat);
   841→      // Wait a beat for the feed to render before smooth-scrolling to it.
   842→      const t = window.setTimeout(() => {
   843→        document
   844→          .querySelector("#feed")
   845→          ?.scrollIntoView({ behavior: "smooth" });
   846→      }, 500);
   847→      return () => window.clearTimeout(t);
   848→    }
   849→  }, []);
   850→
   851→  return (
   852→    <>
   853→      <CommandPalette />
   854→      <ToolDetailDialog />
   855→    </>
   856→  );
   857→}
