    51→function fmtDate(iso: string | null): string {
    52→  if (!iso) return "—";
    53→  return new Date(iso).toLocaleDateString("en-US", {
    54→    month: "short",
    55→    day: "numeric",
    56→    timeZone: "UTC",
    57→  });
    58→}

    97→function DetailBody({ slug, onClose }: { slug: string; onClose: () => void }) {
    98→  const { toast } = useToast();
    99→  const [detail, setDetail] = useState<ToolDetailResponse | null>(null);
   100→  const [loading, setLoading] = useState(true);
   101→  const [error, setError] = useState<string | null>(null);
   102→  const [vote, setVote] = useState<{ votes: number; voted: boolean } | null>(
   103→    null,
   104→  );
   105→  const [copied, setCopied] = useState(false);
   106→
   107→  useEffect(() => {
   108→    let alive = true;
   109→    fetch(`/api/tools/${slug}?vk=${encodeURIComponent(getVoterKey())}`)
   110→      .then(async (res) => {
   111→        if (!res.ok)
   112→          throw new Error(
   113→            res.status === 404 ? "Tool not found" : "Failed to load tool",
   114→          );
   115→        return (await res.json()) as ToolDetailResponse;
   116→      })
   117→      .then((data) => {
   118→        if (alive) {
   119→          setDetail(data);
   120→          setVote({ votes: data.votes, voted: data.voted });
   121→        }
   122→      })
   123→      .catch((err: unknown) => {
   124→        if (alive)
   125→          setError(err instanceof Error ? err.message : "Failed to load tool");
   126→      })
   127→      .finally(() => {
   128→        if (alive) setLoading(false);
   129→      });
   130→    return () => {
   131→      alive = false;
   132→    };
   133→  }, [slug]);
   134→
   135→  const onVote = useCallback(async () => {
   136→    if (!detail?.launchId || !vote) return;
   137→    const nextVoted = !vote.voted;
   138→    setVote({ votes: vote.votes + (nextVoted ? 1 : -1), voted: nextVoted });
   139→    try {
   140→      const res = await fetch("/api/vote", {
   141→        method: "POST",
   142→        headers: { "Content-Type": "application/json" },
   143→        body: JSON.stringify({
   144→          launchId: detail.launchId,
   145→          voterKey: getVoterKey(),
   146→        }),
   147→      });
   148→      const data = (await res.json()) as { voted: boolean; votes: number };
   149→      setVote({ votes: data.votes, voted: data.voted });
   150→      // Sync the feed list behind the modal.
   151→      window.dispatchEvent(
   152→        new CustomEvent("prother:vote", {
   153→          detail: { launchId: detail.launchId, votes: data.votes, voted: data.voted },
   154→        }),
   155→      );
   156→    } catch {
   157→      setVote(vote);
   158→      toast({
   159→        title: "Vote failed",
   160→        description: "Please try again.",
   161→        variant: "destructive",
   162→      });
   163→    }
   164→  }, [detail, vote, toast]);
   165→
   166→  const passedCount = detail?.standards.filter((s) => s.passed).length ?? 0;
   167→
   168→  const shareUrl =
   169→    typeof window !== "undefined"
   170→      ? `${window.location.origin}/${toolHash(slug)}`
   171→      : "";
   172→  const shareText = detail
   173→    ? `🚀 ${detail.name} — ${detail.tagline} is on Prother`
   174→    : "";
   175→
   176→  const onCopyLink = useCallback(async () => {
   177→    try {
   178→      await navigator.clipboard.writeText(shareUrl);
   179→      setCopied(true);
   180→      window.setTimeout(() => setCopied(false), 1600);
   181→    } catch {
   182→      toast({
   183→        title: "Could not copy the link",
   184→        description: shareUrl,
   185→        variant: "destructive",
   186→      });
   187→    }
   188→  }, [shareUrl, toast]);
   189→
   190→  return (

   195→        <div className="p-10 text-center">
   196→          <p className="text-sm text-white/60">{error}</p>

   198→            type="button"
   199→            onClick={onClose}

   201→          >
   202→            ← Close
   203→          </button>
   204→        </div>

   520→function CommandPalette() {
   521→  const { feed } = useFeed();
   522→  const searchOpen = useExplorer((s) => s.searchOpen);
   523→  const setSearch = useExplorer((s) => s.setSearch);
   524→  const openTool = useExplorer((s) => s.openTool);
   525→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   526→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   527→
   528→  const pickTool = useCallback(
   529→    (slug: string) => {
   530→      setSearch(false);
   531→      // Let the command dialog finish closing before opening the detail modal.
   532→      window.setTimeout(() => openTool(slug), 80);
   533→    },
   534→    [setSearch, openTool],
   535→  );
   536→
   537→  const goTo = useCallback(
   538→    (hash: string) => {
   539→      setSearch(false);
   540→      window.setTimeout(() => {
   541→        document.querySelector(hash)?.scrollIntoView({ behavior: "smooth" });
   542→      }, 80);
   543→    },
   544→    [setSearch],
   545→  );
   546→
   547→  const filterCategory = useCallback(
   548→    (slug: string) => {
   549→      setCategoryFilter(slug);
   550→      goTo("#feed");
   551→    },
   552→    [setCategoryFilter, goTo],
   553→  );
   554→
   555→  const seen = new Set<string>();
   556→  const todayTools = (feed?.top ?? []).filter((r) => {
   557→    if (seen.has(r.slug)) return false;
   558→    seen.add(r.slug);
   559→    return true;
   560→  });
   561→
   562→  return (
   563→    <CommandDialog
   564→      open={searchOpen}
   565→      onOpenChange={setSearch}

   717→// ── Mounted-once explorer + ⌘K shortcut ──────────────────────────────────

   719→export function ToolExplorer() {
   720→  const setSearch = useExplorer((s) => s.setSearch);
   721→  const openTool = useExplorer((s) => s.openTool);
   722→
   723→  // ⌘K / ctrl+K opens the command palette.
   724→  useEffect(() => {
   725→    const onKey = (e: KeyboardEvent) => {
   726→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   727→        e.preventDefault();
   728→        setSearch(true);
   729→      }
   730→    };
   731→    window.addEventListener("keydown", onKey);
   732→    return () => window.removeEventListener("keydown", onKey);
   733→  }, [setSearch]);
   734→
   735→  // Shareable deep links: /#tool=<slug> opens the detail modal on load —
   736→  // and keeps working when the hash is navigated to *after* load
   737→  // (same-document navigation, pasted links, back/forward).
   738→  // Clearing the hash (e.g. browser Back) closes the modal again.
   739→  useEffect(() => {
   740→    const applyHash = (initial = false) => {
   741→      const m = window.location.hash.match(/^#tool=([^&]+)$/);
   742→      if (m?.[1]) openTool(decodeURIComponent(m[1]));
   743→      else if (!initial) useExplorer.getState().closeTool();
   744→    };
   745→    applyHash(true);
   746→    const onHash = () => applyHash(false);
   747→    window.addEventListener("hashchange", onHash);
   748→    return () => window.removeEventListener("hashchange", onHash);
   749→  }, [openTool]);
   750→
   751→  // Restore a shared/filtered category from the URL on load (?cat=<slug>).
   752→  useEffect(() => {
   753→    const cat = new URLSearchParams(window.location.search).get("cat");
   754→    if (cat && CATEGORIES.some((c) => c.slug === cat)) {
   755→      useExplorer.getState().setCategoryFilter(cat);
   756→    }
   757→  }, []);
   758→
   759→  return (
   760→    <>
   761→      <CommandPalette />
   762→      <ToolDetailDialog />
   763→    </>
   764→  );
   765→}
