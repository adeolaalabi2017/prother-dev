   100→   100→function avatarGradient(author: string): string {
   101→   101→  let h = 0;
   102→   102→  for (let i = 0; i < author.length; i++) h = (h * 31 + author.charCodeAt(i)) >>> 0;
   103→   103→  return AVATAR_GRADIENTS[h % AVATAR_GRADIENTS.length];
   104→   104→}

   106→   106→function relTime(iso: string): string {
   107→   107→  const s = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000);
   108→   108→  if (s < 60) return "just now";
   109→   109→  const m = Math.floor(s / 60);
   110→   110→  if (m < 60) return `${m}m ago`;
   111→   111→  const h = Math.floor(m / 60);
   112→   112→  if (h < 24) return `${h}h ago`;
   113→   113→  return `${Math.floor(h / 24)}d ago`;
   114→   114→}

   449→   449→        new CustomEvent("prother:vote", {
   450→   450→          detail: { launchId: detail.launchId, votes: data.votes, voted: data.voted },
   451→   451→        }),
   452→   452→      );
   453→   453→    } catch {

   456→   456→        title: "Vote failed",
   457→   457→        description: "Please try again.",
   458→   458→        variant: "destructive",
   459→   459→      });
   460→   460→    }
   461→   461→  }, [detail, vote, toast]);
   462→   462→
   463→   463→  const passedCount = detail?.standards.filter((s) => s.passed).length ?? 0;
   464→   464→
   465→   465→  const shareUrl =
   466→   466→    typeof window !== "undefined"

  1107→  1107→
  1108→  1108→export function ToolExplorer() {
  1109→  1109→  const setSearch = useExplorer((s) => s.setSearch);
  1110→  1110→  const openTool = useExplorer((s) => s.openTool);
  1111→  1111→
  1112→  1112→  // ⌘K / ctrl+K opens the command palette.
  1113→  1113→  useEffect(() => {
  1114→  1114→    const onKey = (e: KeyboardEvent) => {
  1115→  1115→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
  1116→  1116→        e.preventDefault();
  1117→  1117→        setSearch(true);
  1118→  1118→      }
  1119→  1119→    };
  1120→  1120→    window.addEventListener("keydown", onKey);
  1121→  1121→    return () => window.removeEventListener("keydown", onKey);
  1122→  1122→  }, [setSearch]);
  1123→  1123→
  1124→  1124→  // Shareable deep links open the detail modal:
  1125→  1125→  //  - /#tool=<slug>     hash form (works on load AND same-document nav)
  1126→  1126→  //  - /?tool=<slug>     canonical form (server-reachable → real OG unfurl;
  1127→  1127→  //                      copied by the modal's Copy-link action)
  1128→  1128→  // Clearing the hash (e.g. browser Back) closes the modal again.
  1129→  1129→  useEffect(() => {
  1130→  1130→    const applyHash = (initial = false) => {
  1131→  1131→      const m = window.location.hash.match(/^#tool=([^&]+)$/);
  1132→  1132→      if (m?.[1]) openTool(decodeURIComponent(m[1]));
  1133→  1133→      else if (!initial) useExplorer.getState().closeTool();
  1134→  1134→    };
  1135→  1135→    applyHash(true);
  1136→  1136→    const onHash = () => applyHash(false);
  1137→  1137→    window.addEventListener("hashchange", onHash);
  1138→  1138→    return () => window.removeEventListener("hashchange", onHash);
  1139→  1139→  }, [openTool]);
  1140→  1140→
  1141→  1141→  // Canonical ?tool= form: open the modal once at mount, then normalize the
  1142→  1142→  // URL to the hash form (keeps ?cat=/?tab=, drops the query param) so the
  1143→  1143→  // address bar matches what in-app navigation produces.
  1144→  1144→  useEffect(() => {
  1145→  1145→    const t = new URLSearchParams(window.location.search).get("tool");
  1146→  1146→    if (!t) return;
  1147→  1147→    openTool(t);
  1148→  1148→    const url = new URL(window.location.href);
  1149→  1149→    url.searchParams.delete("tool");
  1150→  1150→    window.history.replaceState(null, "", url.pathname + url.search + url.hash);
  1151→  1151→  }, [openTool]);
  1152→  1152→
  1153→  1153→  // Restore a shared/filtered category from the URL on load (?cat=<slug>)
  1154→  1154→  // and jump straight to the feed — a shared ?cat= link is an intent to browse.
  1155→  1155→  useEffect(() => {
  1156→  1156→    const cat = new URLSearchParams(window.location.search).get("cat");
  1157→  1157→    if (cat && CATEGORIES.some((c) => c.slug === cat)) {
  1158→  1158→      useExplorer.getState().setCategoryFilter(cat);
  1159→  1159→      // Wait a beat for the feed to render before smooth-scrolling to it.
  1160→  1160→      const t = window.setTimeout(() => {
  1161→  1161→        document
  1162→  1162→          .querySelector("#feed")
  1163→  1163→          ?.scrollIntoView({ behavior: "smooth" });
  1164→  1164→      }, 500);
  1165→  1165→      return () => window.clearTimeout(t);
  1166→  1166→    }
  1167→  1167→  }, []);
  1168→  1168→
  1169→  1169→  return (
  1170→  1170→    <>
  1171→  1171→      <CommandPalette />
  1172→  1172→      <ToolDetailDialog />
  1173→  1173→    </>
  1174→  1174→  );
  1175→  1175→}
