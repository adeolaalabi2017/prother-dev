    20→    20→function formatCountdown(totalSec: number): string {
    21→    21→  const s = Math.max(0, totalSec);
    22→    22→  const hh = String(Math.floor(s / 3600)).padStart(2, "0");
    23→    23→  const mm = String(Math.floor((s % 3600) / 60)).padStart(2, "0");
    24→    24→  const ss = String(s % 60).padStart(2, "0");
    25→    25→  return `${hh}:${mm}:${ss}`;
    26→    26→}

   335→   335→export function LaunchFeed() {
   336→   336→  const { toast } = useToast();
   337→   337→  const { feed, loading, error, refresh } = useFeed();
   338→   338→  const openTool = useExplorer((s) => s.openTool);
   339→   339→  const setSubmitOpen = useExplorer((s) => s.setSubmitOpen);
   340→   340→  const categoryFilter = useExplorer((s) => s.categoryFilter);
   341→   341→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   342→   342→  // ?tab= persistence — shareable feed states (new | top | tomorrow | yesterday).
   343→   343→  // Read AFTER mount only: reading window.location during the initial render makes
   344→   344→  // the server HTML ("new") and the hydrated client (?tab=top) diverge, which
   345→   345→  // React reports as a hydration mismatch. Post-mount sync keeps SSR markup valid.
   346→   346→  const [tab, setTab] = useState<Tab>("new");
   347→   347→  useEffect(() => {
   348→   348→    const t = new URLSearchParams(window.location.search).get("tab");
   349→   349→    if (t === "top" || t === "tomorrow" || t === "yesterday") setTab(t);
   350→   350→  }, []);
   351→   351→
   352→   352→  // Archive day browser: null = yesterday (served from the main feed payload),
   353→   353→  // otherwise an ISO day from the strip — fetched once, then cached.
   354→   354→  const [archiveDate, setArchiveDate] = useState<string | null>(null);
   355→   355→  const [dayCache, setDayCache] = useState<Record<string, DayArchiveResponse>>({});
   356→   356→  const [dayLoading, setDayLoading] = useState(false);
   357→   357→
   358→   358→  // Fetch (and cache) a selected archive day's final standings.
   359→   359→  useEffect(() => {
   360→   360→    if (!archiveDate || dayCache[archiveDate]) return;
   361→   361→    let alive = true;
   362→   362→    setDayLoading(true);
   363→   363→    fetch(`/api/feed/day?date=${encodeURIComponent(archiveDate)}`)
   364→   364→      .then(async (res) => {
   365→   365→        if (!res.ok) throw new Error("Failed to load this archive day");
   366→   366→        return (await res.json()) as DayArchiveResponse;
   367→   367→      })
   368→   368→      .then((data) => {
   369→   369→        if (alive) setDayCache((c) => ({ ...c, [data.date]: data }));
   370→   370→      })
   371→   371→      .catch(() => {
   372→   372→        if (alive) setArchiveDate(null); // gracefully fall back to yesterday
   373→   373→      })
   374→   374→      .finally(() => {
   375→   375→        if (alive) setDayLoading(false);
   376→   376→      });
   377→   377→    return () => {
   378→   378→      alive = false;
   379→   379→    };
   380→   380→  }, [archiveDate, dayCache]);
   381→   381→
   382→   382→  // UTC rollover shifts the whole archive window — drop cached days + selection.
   383→   383→  useEffect(() => {
   384→   384→    setDayCache({});
   385→   385→    setArchiveDate(null);
   386→   386→  }, [feed?.date]);
   387→   387→
   388→   388→  // Keep the selected tab in the URL (replaceState — no history spam).
   389→   389→  const setTabSync = useCallback((next: Tab) => {
   390→   390→    setTab(next);
   391→   391→    if (typeof window === "undefined") return;
   392→   392→    const url = new URL(window.location.href);
   393→   393→    const prev = url.searchParams.get("tab");
   394→   394→    if (next === "new") {
   395→   395→      if (prev) {
   396→   396→        url.searchParams.delete("tab");
   397→   397→        window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   398→   398→      }
   399→   399→    } else if (prev !== next) {
   400→   400→      url.searchParams.set("tab", next);
   401→   401→      window.history.replaceState(null, "", url.pathname + url.search + url.hash);
   402→   402→    }
   403→   403→  }, []);
   404→   404→  const [now, setNow] = useState<number>(() => Date.now());
   405→   405→  const [voteState, setVoteState] = useState<VoteState>({});
   406→   406→
   407→   407→  // Countdown deadline — computed from the feed at render (no effect-synced state).
   408→   408→  const deadlineRef = useRef<{ date: string | null; at: number }>({ date: null, at: 0 });
   409→   409→  if (feed && deadlineRef.current.date !== feed.date) {
   410→   410→    deadlineRef.current = { date: feed.date, at: Date.now() + feed.resetsInSec * 1000 };
   411→   411→  }
   412→   412→  const deadline = deadlineRef.current.at;
   413→   413→
   414→   414→  // Tick the clock every second (setState inside the interval callback is fine).
   415→   415→  useEffect(() => {
   416→   416→    const iv = setInterval(() => setNow(Date.now()), 1000);
   417→   417→    return () => clearInterval(iv);
   418→   418→  }, []);
   419→   419→
   420→   420→  // Votes cast inside the tool detail modal sync back into the feed list.
   421→   421→  useEffect(() => {
   422→   422→    const onModalVote = (e: Event) => {
   423→   423→      const d = (e as CustomEvent).detail as {
   424→   424→        launchId: string;
   425→   425→        votes: number;
   426→   426→        voted: boolean;
   427→   427→      };
   428→   428→      setVoteState((prev) => ({
   429→   429→        ...prev,
   430→   430→        [d.launchId]: { votes: d.votes, voted: d.voted },
   431→   431→      }));
   432→   432→    };
   433→   433→    window.addEventListener("prother:vote", onModalVote);
   434→   434→    return () => window.removeEventListener("prother:vote", onModalVote);
   435→   435→  }, []);
   436→   436→
   437→   437→  const remaining = feed ? Math.max(0, Math.round((deadline - now) / 1000)) : null;
   438→   438→
   439→   439→  // UTC-midnight rollover: refetch the feed exactly once when the countdown hits zero.
   440→   440→  const rolledRef = useRef(false);
   441→   441→  useEffect(() => {
   442→   442→    if (remaining !== null && remaining > 0) {
   443→   443→      rolledRef.current = false;
   444→   444→    } else if (remaining === 0 && !rolledRef.current) {
