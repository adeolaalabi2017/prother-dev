    36→/**
    37→ * Admin — API integrations module (Task 34-c), rendered inside Settings.
    38→ * Stored credentials (AI providers, payment, storage/CDN, email, analytics)
    39→ * backed by GET/PUT/DELETE /api/admin/integrations.
    40→ *
    41→ * Secret handling: the server masks secret-looking fields in every response
    42→ * (sentinel + last 4 chars), and a write that echoes the mask keeps the
    43→ * stored value. This UI therefore never renders a real secret: masked values
    44→ * render as empty password inputs ("Stored: keep or replace"), and a blank
    45→ * secret field is sent back as the mask so saving never destroys a key.

   170→type FormState = {
   171→  key: string;
   172→  name: string;
   173→  category: string;
   174→  enabled: boolean;
   175→  config: Record<string, string>;
   176→  /** Field order: template fields + stored keys + custom rows. */
   177→  fields: string[];
   178→  /** Masked stored values echoed back on save (blank input keeps them). */
   179→  masked: Record<string, string>;
   180→  notes: string;
   181→  /** Templates lock the key; custom rows let the editor pick one. */
   182→  keyLocked: boolean;

   184→
   185→function formFromTemplate(tpl: Template): FormState {
   186→  return {
   187→    key: tpl.key === "custom" ? "" : tpl.key,
   188→    name: tpl.key === "custom" ? "" : tpl.name,
   189→    category: tpl.category,
   190→    enabled: true,
   191→    config: Object.fromEntries(tpl.fields.map((f) => [f.name, f.value ?? ""])),
   192→    fields: tpl.fields.map((f) => f.name),
   193→    masked: {},
   194→    notes: "",
   195→    keyLocked: tpl.key !== "custom",
   196→  };
   197→}
   198→
   199→function formFromRow(row: IntegrationRow): FormState {
   200→  const config: Record<string, string> = {};
   201→  const masked: Record<string, string> = {};
   202→  for (const [field, value] of Object.entries(row.config)) {
   203→    if (isMaskedValue(value)) {
   204→      masked[field] = value;
   205→      config[field] = "";
   206→    } else {
   207→      config[field] = value;
   208→    }
   209→  }
   210→  return {
   211→    key: row.key,
   212→    name: row.name,
   213→    category: row.category,
   214→    enabled: row.enabled,
   215→    config,
   216→    fields: Object.keys(row.config),
   217→    masked,
   218→    notes: row.notes,
   219→    keyLocked: true,
   220→  };
   221→}
   222→
   223→function fieldIsSecret(field: string, tpl?: Template): boolean {
   224→  return tpl?.fields.find((f) => f.name === field)?.secret === true || looksSecret(field);
   225→}

   227→function IntegrationForm({
   228→  apiKey,
   229→  state,
   230→  onStateChange,
   231→  onDone,
   232→}: {
   233→  apiKey: string;
   234→  state: FormState;
   235→  onStateChange: (next: FormState) => void;
   236→  onDone: (rows: IntegrationRow[]) => void;
   237→}) {
   238→  const { toast } = useToast();
   239→  const [busy, setBusy] = useState(false);
   240→  const [newField, setNewField] = useState("");
   241→
   242→  const set = (patch: Partial<FormState>) => onStateChange({ ...state, ...patch });
   243→  const setField = (field: string, value: string) =>
   244→    set({ config: { ...state.config, [field]: value } });
   245→
   246→  const addField = () => {
   247→    const name = newField.trim().slice(0, 60);
   248→    if (!name || state.fields.includes(name)) return;
   249→    set({
   250→      fields: [...state.fields, name],
   251→      config: { ...state.config, [name]: "" },
   252→    });
   253→    setNewField("");
   254→  };
   255→
   256→  const removeField = (field: string) => {
   257→    const config = { ...state.config };
   258→    delete config[field];
   259→    set({
   260→      fields: state.fields.filter((f) => f !== field),
   261→      config,
   262→      masked: Object.fromEntries(
   263→        Object.entries(state.masked).filter(([k]) => k !== field)
   264→      ),
   265→    });
   266→  };
   267→
   268→  const save = async () => {
   269→    setBusy(true);
   270→    try {
   271→      // Only send filled fields + echoed masks: an emptied non-secret drops
   272→      // from the config (whole-object replace), a blank secret echoes its
   273→      // mask so the stored value survives the round trip.
   274→      const config: Record<string, string> = {};
   275→      for (const field of state.fields) {
   276→        const v = (state.config[field] ?? "").trim();
   277→        if (v) config[field] = v;
   278→        else if (state.masked[field]) config[field] = state.masked[field];
   279→      }
   280→      const res = await adminFetch(apiKey, "/api/admin/integrations", {
   281→        method: "PUT",
   282→        body: JSON.stringify({
   283→          key: state.key.trim().toLowerCase(),
   284→          name: state.name.trim(),
   285→          category: state.category,
   286→          enabled: state.enabled,
   287→          config,
   288→          notes: state.notes.trim(),
   289→        }),
   290→      });
   291→      const d = (await res.json()) as {
   292→        integrations?: IntegrationRow[];
   293→        error?: string;
   294→      };
   295→      if (!res.ok || !d.integrations) {
   296→        toast({ title: d.error ?? "Save failed", variant: "destructive" });
   297→        return;
   298→      }
   299→      toast({
   300→        title: "Integration saved",
   301→        description: `${state.name.trim()} (${state.category}) stored server-side.`,
   302→      });
   303→      onDone(d.integrations);
   304→    } catch {
   305→      toast({ title: "Network error", variant: "destructive" });
   306→    } finally {
   307→      setBusy(false);
   308→    }
   309→  };
   310→
   311→  const valid =
   312→    KEY_RE.test(state.key.trim().toLowerCase()) && state.name.trim().length >= 2;
   313→
   314→  return (
   315→    <div className="space-y-3 rounded-xl border border-ember/25 bg-ember/[0.04] p-3.5">
   316→      <div className="grid gap-3 sm:grid-cols-2">
   317→        <div className="space-y-1.5">
   318→          <Label className={labelCx}>Key</Label>
   319→          <Input
   320→            value={state.key}
   321→            readOnly={state.keyLocked}
   322→            onChange={(e) =>
   323→              set({ key: e.target.value.toLowerCase().replace(/[^a-z0-9-]/g, "-") })
   324→            }
   325→            placeholder="openai"
   326→            aria-label="Integration key"
   327→            className={cn(inputCx, "font-mono", state.keyLocked && "opacity-70")}
   328→          />
   329→          <p className="text-xs text-white/55">
   330→            {state.keyLocked
   331→              ? "Fixed by the template. Server code reads config via this key."
   332→              : "2 to 41 chars: a-z, 0-9, dashes. Server code reads config via this key."}
   333→          </p>
   334→        </div>
   335→        <div className="space-y-1.5">
   336→          <Label className={labelCx}>Display name</Label>
   337→          <Input
   338→            value={state.name}
   339→            onChange={(e) => set({ name: e.target.value })}
   340→            placeholder="OpenAI"
   341→            aria-label="Integration display name"
   342→            className={inputCx}
   343→          />
   344→        </div>
   345→        <div className="space-y-1.5">
   346→          <Label className={labelCx}>Category</Label>
   347→          <Select
   348→            value={state.category}
   349→            onValueChange={(v) => set({ category: v })}
   350→          >
   351→            <SelectTrigger aria-label="Integration category" className={inputCx}>
   352→              <SelectValue />
   353→            </SelectTrigger>
   354→            <SelectContent className="border-white/10 bg-coal text-white">
   355→              {CATEGORIES.map((c) => (
   356→                <SelectItem key={c.value} value={c.value}>
   357→                  {c.label}
   358→                </SelectItem>
   359→              ))}
   360→            </SelectContent>
   361→          </Select>
   362→        </div>
   363→        <div className="space-y-1.5">
   364→          <Label className={labelCx}>Status</Label>
   365→          <label className="flex min-h-11 items-center gap-2 text-sm text-white/70">
   366→            <Switch
   367→              checked={state.enabled}
   368→              onCheckedChange={(v) => set({ enabled: v })}
   369→              className="data-[state=checked]:bg-ember"
   370→              aria-label="Integration enabled"
   371→            />
   372→            {state.enabled ? "Enabled" : "Disabled"}
   373→          </label>
   374→        </div>
   375→      </div>
   376→
   377→      <div className="space-y-2.5">
   378→        <p className={labelCx}>Configured fields</p>
   379→        {state.fields.map((field) => {
   380→          const secret = fieldIsSecret(field);
   381→          const wasMasked = Boolean(state.masked[field]);
   382→          return (
   383→            <div key={field} className="flex items-end gap-2">
   384→              <div className="w-40 shrink-0 pb-2.5">
   385→                <span
   386→                  className="block truncate font-mono text-xs text-white/70"
   387→                  title={field}
   388→                >
   389→                  {field}
   390→                </span>
   391→                {secret && (
   392→                  <span className="font-mono text-xs tracking-wider text-ember uppercase">
   393→                    secret
   394→                  </span>
   395→                )}
   396→              </div>
   397→              <div className="min-w-0 flex-1">
   398→                <Input
   399→                  type={secret ? "password" : "text"}
   400→                  value={state.config[field] ?? ""}
   401→                  onChange={(e) => setField(field, e.target.value)}
   402→                  placeholder={
   403→                    wasMasked
   404→                      ? "Stored: keep or replace"
   405→                      : secret
   406→                        ? "Value is stored server-side"
   407→                        : field
   408→                  }
   409→                  autoComplete="off"
   410→                  aria-label={`Value for ${field}`}
   411→                  className={cn(inputCx, secret && "font-mono")}
   412→                />
   413→              </div>
   414→              <button
   415→                type="button"
   416→                aria-label={`Remove field ${field}`}
   417→                onClick={() => removeField(field)}
   418→                className="mb-0.5 rounded-lg p-2.5 text-white/60 transition-colors hover:text-red-400"
   419→              >
   420→                <X className="size-3.5" aria-hidden />
   421→              </button>
   422→            </div>
   423→          );
   424→        })}
   425→        {state.fields.length === 0 && (
   426→          <p className="text-xs text-white/55">
   427→            No fields yet. Add the keys this integration needs below.
   428→          </p>
   429→        )}
   430→        <div className="flex items-center gap-2 pt-1">
   431→          <Input
   432→            value={newField}
   433→            onChange={(e) => setNewField(e.target.value)}
   434→            placeholder="Custom field name"
   435→            aria-label="Custom field name"
   436→            maxLength={60}
   437→            className={cn(inputCx, "w-44")}
   438→          />
   439→          <Button
   440→            size="sm"
   441→            variant="outline"
   442→            onClick={addField}
   443→            disabled={!newField.trim()}
   444→            className="rounded-lg border-white/15 text-white/70 hover:bg-white/5"
   445→          >
   446→            <Plus className="size-3.5" aria-hidden />
   447→            Add field
   448→          </Button>
   449→        </div>
   450→      </div>
   451→
   452→      <div className="space-y-1.5">
   453→        <Label className={labelCx}>Notes</Label>
   454→        <Textarea
   455→          value={state.notes}
   456→          rows={2}
   457→          onChange={(e) => set({ notes: e.target.value })}
   458→          placeholder="Where this integration is used, rotation dates…"
   459→          aria-label="Integration notes"
   460→          className={inputCx}
   461→        />
   462→      </div>
   463→
   464→      <div className="flex flex-wrap items-center gap-2 border-t border-white/10 pt-3">
   465→        <Button
   466→          disabled={busy || !valid}
   467→          onClick={() => void save()}
   468→          className="rounded-lg bg-ember font-semibold text-coal shadow-none hover:bg-ember-hot dark:text-coal"
   469→        >
   470→          {busy ? (
   471→            <Loader2 className="size-4 animate-spin" aria-hidden />
   472→          ) : (
   473→            <Save className="size-4" aria-hidden />
   474→          )}
   475→          Save integration
   476→        </Button>
   477→        <p className="text-xs text-white/55">
   478→          Secrets never leave the server verbatim: they are returned masked
   479→          (last 4 characters only). Leave a secret blank to keep the stored
   480→          value.
   481→        </p>
   482→      </div>
   483→    </div>
   484→  );
   485→}

   487→export function IntegrationsModule({ apiKey }: { apiKey: string }) {
   488→  const { toast } = useToast();
   489→  const [rows, setRows] = useState<IntegrationRow[] | null>(null);
   490→  const [error, setError] = useState(false);
   491→  const [adding, setAdding] = useState(false);
   492→  const [tplKey, setTplKey] = useState("");
   493→  const [draft, setDraft] = useState<FormState | null>(null);
   494→  const [editingKey, setEditingKey] = useState<string | null>(null);
   495→  const [editState, setEditState] = useState<FormState | null>(null);
   496→  const [confirmDelete, setConfirmDelete] = useState<string | null>(null);
   497→  const [busyKey, setBusyKey] = useState<string | null>(null);
   498→
   499→  const load = useCallback(() => {
   500→    adminFetch(apiKey, "/api/admin/integrations")
   501→      .then((r) => {
   502→        if (!r.ok) throw new Error(String(r.status));
   503→        return r.json() as Promise<{ integrations: IntegrationRow[] }>;
   504→      })
   505→      .then((d) => {
   506→        setRows(d.integrations);
   507→        setError(false);
   508→      })
   509→      .catch(() => setError(true));
   510→  }, [apiKey]);
   511→
   512→  useEffect(load, [load]);
   513→
   514→  const retry = useCallback(() => {
   515→    setRows(null);
   516→    setError(false);
   517→    load();
   518→  }, [load]);
   519→
   520→  const pickTemplate = (key: string) => {
   521→    setTplKey(key);
   522→    const tpl = TEMPLATES.find((t) => t.key === key);
   523→    setDraft(tpl ? formFromTemplate(tpl) : null);
   524→  };
   525→
   526→  const toggleEnabled = useCallback(
   527→    async (row: IntegrationRow) => {
   528→      setBusyKey(row.key);
   529→      try {
   530→        // Masked config echoes back: the server keeps stored secrets.
   531→        const res = await adminFetch(apiKey, "/api/admin/integrations", {
   532→          method: "PUT",
   533→          body: JSON.stringify({
   534→            key: row.key,
   535→            name: row.name,
   536→            category: row.category,
   537→            enabled: !row.enabled,
   538→            config: row.config,
   539→            notes: row.notes,
   540→          }),
   541→        });
   542→        const d = (await res.json()) as {
   543→          integrations?: IntegrationRow[];
   544→          error?: string;
   545→        };
   546→        if (!res.ok || !d.integrations) {
   547→          toast({ title: d.error ?? "Update failed", variant: "destructive" });
   548→          return;
   549→        }
   550→        setRows(d.integrations);
   551→        toast({
   552→          title: `${row.name} ${!row.enabled ? "enabled" : "disabled"}`,
   553→          description: !row.enabled
   554→            ? "Server code can read its credentials again."
   555→            : "Disabled integrations are ignored server-side.",
   556→        });
   557→      } catch {
   558→        toast({ title: "Network error", variant: "destructive" });
   559→      } finally {
   560→        setBusyKey(null);
   561→      }
   562→    },
   563→    [apiKey, toast]
   564→  );
   565→
   566→  const deleteRow = useCallback(
   567→    async (row: IntegrationRow) => {
   568→      setBusyKey(row.key);
   569→      try {
   570→        const res = await adminFetch(
   571→          apiKey,
   572→          `/api/admin/integrations?key=${encodeURIComponent(row.key)}`,
   573→          { method: "DELETE" }
   574→        );
   575→        const d = (await res.json()) as {
   576→          integrations?: IntegrationRow[];
   577→          error?: string;
   578→        };
   579→        if (!res.ok || !d.integrations) {
   580→          toast({ title: d.error ?? "Delete failed", variant: "destructive" });
   581→          return;
   582→        }
   583→        setRows(d.integrations);
   584→        setConfirmDelete(null);
   585→        if (editingKey === row.key) {
   586→          setEditingKey(null);
   587→          setEditState(null);
   588→        }
   589→        toast({ title: "Integration deleted", description: row.name });
   590→      } catch {
   591→        toast({ title: "Network error", variant: "destructive" });
   592→      } finally {
   593→        setBusyKey(null);
   594→      }
   595→    },
   596→    [apiKey, editingKey, toast]
   597→  );
   598→
   599→  const configuredCount = (row: IntegrationRow) =>
   600→    Object.values(row.config).filter((v) => v !== "").length;
