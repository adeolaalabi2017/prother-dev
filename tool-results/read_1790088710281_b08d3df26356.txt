   228→// ── The wizard ───────────────────────────────────────────────────────────
   229→
   230→export function SubmitWizard() {
   231→  const { toast } = useToast();
   232→  const router = useRouter();
   233→  const open = useExplorer((s) => s.submitOpen);
   234→  const setOpen = useExplorer((s) => s.setSubmitOpen);
   235→  const submitPrefill = useExplorer((s) => s.submitPrefill);
   236→  const openTool = useExplorer((s) => s.openTool);
   237→  const setTrackOpen = useExplorer((s) => s.setTrackOpen);
   238→
   239→  const [step, setStep] = useState(0);
   240→  const [form, setForm] = useState<SubmitForm>(INITIAL_SUBMIT_FORM);
   241→  const [errors, setErrors] = useState<Record<string, string>>({});
   242→  const [dup, setDup] = useState<DuplicateInfo | null>(null);
   243→  const [checkingDup, setCheckingDup] = useState(false);
   244→  const [submitting, setSubmitting] = useState(false);
   245→  const [result, setResult] = useState<{ id: string; position: number } | null>(
   246→    null
   247→  );
   248→
   249→  const set = useCallback(
   250→    <K extends keyof SubmitForm>(key: K, value: SubmitForm[K]) => {
   251→      setForm((f) => ({ ...f, [key]: value }));
   252→      setErrors((e) => {
   253→        if (!e[key as string]) return e;
   254→        const next = { ...e };
   255→        delete next[key as string];
   256→        return next;
   257→      });
   258→    },
   259→    []
   260→  );
   261→
   262→  const reset = useCallback(() => {
   263→    setForm(INITIAL_SUBMIT_FORM);
   264→    setStep(0);
   265→    setErrors({});
   266→    setDup(null);
   267→    setResult(null);
   268→  }, []);
   269→
   270→  const onOpenChange = useCallback(
   271→    (next: boolean) => {
   272→      setOpen(next);
   273→      if (!next) window.setTimeout(reset, 250);
   274→    },
   275→    [setOpen, reset]
   276→  );
   277→
   278→  // "Resubmit with fixes" (status tracker): a draft queued in the store is
   279→  // applied once when the wizard opens with one — confirm checkboxes stay
   280→  // unticked so the maker re-attests after addressing the cited standards.
   281→  // `toast` is a module-level function (stable identity), so it's a safe dep.
   282→  useEffect(() => {
   283→    if (!open || !submitPrefill) return;
   284→    setForm({ ...INITIAL_SUBMIT_FORM, ...submitPrefill });
   285→    setStep(0);
   286→    setErrors({});
   287→    setDup(null);
   288→    setResult(null);
   289→    toast({
   290→      title: "Draft loaded",
   291→      description: `Pre-filled from your rejected ${submitPrefill.name} submission.`,
   292→    });
   293→  }, [open, submitPrefill, toast]);
   294→
   295→  // PRD §11: duplicate check on Step-1 URL blur.
   296→  const checkDuplicate = useCallback(async () => {
   297→    const domain = domainOf(form.websiteUrl);
   298→    if (!domain || !domain.includes(".")) return;
   299→    setCheckingDup(true);
   300→    try {
   301→      const res = await fetch(
   302→        `/api/submit/check?url=${encodeURIComponent(form.websiteUrl.trim())}`
   303→      );
   304→      const data = (await res.json()) as { valid: boolean; duplicate: DuplicateInfo | null };
   305→      setDup(data.duplicate);
   306→    } catch {
   307→      // Network hiccup — non-blocking; server re-checks on submit.
   308→    } finally {
   309→      setCheckingDup(false);
   310→    }
   311→  }, [form.websiteUrl]);
   312→
   313→  const goNext = useCallback(() => {
   314→    const errs = validateStep(step, form);
   315→    setErrors(errs);
   316→    if (Object.keys(errs).length > 0) return;
   317→    if (step === 0 && dup?.kind === "submission") return; // blocked by interstitial
   318→    setStep((s) => Math.min(4, s + 1));
   319→  }, [step, form, dup]);
   320→
   321→  const onSubmit = useCallback(async () => {
   322→    const errs = validateStep(4, form);
   323→    setErrors(errs);
   324→    if (Object.keys(errs).length > 0) return;
   325→    setSubmitting(true);
   326→    try {
   327→      const res = await fetch("/api/submit", {
   328→        method: "POST",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify(form),
   331→      });
   332→      if (res.status === 201) {
   333→        const data = (await res.json()) as { id: string; position: number };
   334→        setResult({ id: data.id, position: data.position });
   335→        return;
   336→      }
   337→      const data = (await res.json().catch(() => ({}))) as {
   338→        error?: string;
   339→        fields?: Record<string, string>;
   340→        duplicate?: DuplicateInfo;
   341→      };
   342→      if (data.fields) setErrors(data.fields);
   343→      if (data.duplicate) {
   344→        setDup(data.duplicate);
   345→        setStep(0);
   346→      }
   347→      toast({
   348→        title: data.error ?? "Submission failed",
   349→        description: "Fix the highlighted fields and try again.",
   350→        variant: "destructive",
   351→      });
   352→    } catch {
   353→      toast({
   354→        title: "Submission failed",
   355→        description: "Network error — please try again.",
   356→        variant: "destructive",
   357→      });
   358→    } finally {
   359→      setSubmitting(false);
   360→    }
   361→  }, [form, toast]);
   362→
   363→  const taglineLen = form.tagline.trim().length;
   364→  const descLen = form.description.trim().length;
   365→
   366→  const toggleTag = useCallback(
   367→    (tag: TagVocab) => {
   368→      setForm((f) => {
   369→        const has = f.tags.includes(tag);
   370→        if (has) return { ...f, tags: f.tags.filter((t) => t !== tag) };
   371→        if (f.tags.length >= 5) return f; // ≤5 controlled vocab (PRD §13)
   372→        return { ...f, tags: [...f.tags, tag] };
   373→      });
   374→    },
   375→    []
   376→  );
   377→
   378→  const canProceed = useMemo(
   379→    () => Object.keys(validateStep(step, form)).length === 0,
   380→    [step, form]
   381→  );
   382→
   383→  return (
   384→    <Dialog open={open} onOpenChange={onOpenChange}>
   385→      <DialogContent
   386→        showCloseButton={!result}
   387→        className="max-h-[92vh] gap-0 overflow-y-auto border-white/10 bg-coal p-0 text-white max-w-[calc(100vw-2rem)] sm:max-w-3xl"
   388→      >
   389→        <DialogTitle className="sr-only">Submit your tool</DialogTitle>
   390→        <DialogDescription className="sr-only">
   391→          Five-step submission wizard: product, description, pricing, media,
   392→          confirmation. Editors review within 24 hours.
   393→        </DialogDescription>
   394→
   395→        {result ? (
   396→          // ── Post-submit state (PRD §11: "in review, typically 24h") ──
   397→          <div className="flex flex-col items-center px-8 py-14 text-center">
   398→            <motion.div
   399→              initial={{ scale: 0.6, opacity: 0 }}
   400→              animate={{ scale: 1, opacity: 1 }}
   401→              transition={{ type: "spring", stiffness: 260, damping: 18 }}
   402→            >
   403→              <CheckCircle2 className="size-14 text-ember" aria-hidden />
   404→            </motion.div>
   405→            <h2 className="mt-5 text-2xl font-black tracking-tight">
   406→              Queued for review
   407→            </h2>
   408→            <p className="mt-2 max-w-sm text-sm text-white/60">
   409→              <span className="font-semibold text-white">{form.name.trim()}</span> is
   410→              in the moderation queue. Editors check the six standards and reply
   411→              to <span className="font-mono text-ember">{form.email.trim()}</span> —
   412→              typically within 24h. Approved launches get 72h notice + a launch kit.
   413→            </p>
   414→            <div className="mt-6 grid w-full max-w-sm grid-cols-2 gap-3">
   415→              <div className="rounded-xl border border-white/10 bg-white/[0.03] p-3">
   416→                <p className="font-mono text-[10px] tracking-widest text-white/40">QUEUE</p>
   417→                <p className="mt-1 font-mono text-2xl font-bold tabular-nums text-ember">
   418→                  #{result.position}
   419→                </p>
   420→              </div>
   421→              <div className="rounded-xl border border-white/10 bg-white/[0.03] p-3">
   422→                <p className="font-mono text-[10px] tracking-widest text-white/40">TICKET</p>
   423→                <p className="mt-1.5 font-mono text-sm text-white/70">
   424→                  {result.id.slice(0, 8).toUpperCase()}
   425→                </p>
   426→              </div>
   427→            </div>
   428→            <div className="mt-7 flex flex-wrap justify-center gap-3">
   429→              <Button
   430→                type="button"
   431→                onClick={() => onOpenChange(false)}
   432→                className="rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   433→              >
   434→                Done — back to the feed
   435→              </Button>
   436→              <Button
   437→                type="button"
   438→                variant="outline"
   439→                onClick={() => {
   440→                  // Hand off to the maker status tracker with the email
   441→                  // pre-filled (PRD §11 status tracking).
   442→                  const email = form.email.trim();
   443→                  onOpenChange(false);
   444→                  window.setTimeout(() => setTrackOpen(true, email), 80);
   445→                }}
   446→                className="rounded-lg border-white/15 bg-transparent text-white/70 hover:bg-white/5 hover:text-white"
   447→              >
   448→                Track this submission
   449→              </Button>
   450→              <Button
   451→                type="button"
   452→                variant="outline"
   453→                onClick={reset}
   454→                className="rounded-lg border-white/15 bg-transparent text-white/70 hover:bg-white/5 hover:text-white"
   455→              >
   456→                Submit another tool
   457→              </Button>
   458→            </div>
   459→          </div>
   460→        ) : (
   461→          <div className="grid lg:grid-cols-[1fr_270px]">
   462→            {/* ── Left: steps ─────────────────────────────────────────── */}
   463→            <div className="min-w-0">
   464→              {/* Progress header */}
   465→              <div className="border-b border-white/10 px-5 pb-4 pt-5 sm:px-6">
   466→                <div className="flex items-center justify-between">
   467→                  <p className="font-mono text-[11px] tracking-[0.25em] text-ember">
   468→                    SUBMIT YOUR TOOL
   469→                  </p>
   470→                  <p className="font-mono text-[11px] text-white/40">
   471→                    STEP {String(step + 1).padStart(2, "0")}/05
   472→                  </p>
   473→                </div>
   474→                <div className="mt-3 flex gap-1.5" role="presentation">
   475→                  {STEP_TITLES.map((t, i) => (
   476→                    <div
   477→                      key={t}
   478→                      className={cn(
   479→                        "h-1 flex-1 rounded-full transition-colors duration-300",
   480→                        i < step ? "bg-ember/70" : i === step ? "bg-ember" : "bg-white/10"
   481→                      )}
   482→                    />
   483→                  ))}
   484→                </div>
   485→                <p className="mt-2.5 text-sm font-semibold text-white">
   486→                  {STEP_TITLES[step]}
   487→                  <span className="ml-2 font-mono text-[10px] font-normal text-white/35">
   488→                    {step === 0 && "Where does it live?"}
   489→                    {step === 1 && "The 5-second scan"}
   490→                    {step === 2 && "Honest pricing only (S4)"}
   491→                    {step === 3 && "Make it recognizable"}
   492→                    {step === 4 && "Two confirmations, then queue"}
   493→                  </span>
   494→                </p>
   495→              </div>
   496→
   497→              <div className="space-y-5 px-5 py-5 sm:px-6">
   498→                {/* STEP 1 — Product */}
   499→                {step === 0 && (
   500→                  <>
   501→                    <div className="space-y-1.5">
   502→                      <FieldLabel htmlFor="su-url" hint="we auto-check for duplicates">
   503→                        WEBSITE URL *
   504→                      </FieldLabel>
   505→                      <div className="relative">
   506→                        <Input
   507→                          id="su-url"
   508→                          value={form.websiteUrl}
   509→                          onChange={(e) => set("websiteUrl", e.target.value)}
   510→                          onBlur={checkDuplicate}
   511→                          placeholder="yourtool.ai"
   512→                          inputMode="url"
   513→                          autoComplete="url"
   514→                          className={cn(inputCls, "pr-9")}
   515→                        />
   516→                        {checkingDup && (
   517→                          <Loader2
   518→                            className="absolute right-3 top-1/2 size-4 -translate-y-1/2 animate-spin text-white/30"
   519→                            aria-hidden
   520→                          />
   521→                        )}
   522→                      </div>
   523→                      <FieldError msg={errors.websiteUrl} />
   524→                    </div>
   525→
   526→                    <AnimatePresence>
   527→                      {dup && (
   528→                        <DuplicateBanner
   529→                          dup={dup}
   530→                          onViewTool={(slug) => {
   531→                            onOpenChange(false);
   532→                            window.setTimeout(() => openTool(slug), 120);
   533→                          }}
   534→                          onContinue={() => setDup(null)}
   535→                        />
   536→                      )}
   537→                    </AnimatePresence>
   538→
   539→                    <div className="space-y-1.5">
   540→                      <FieldLabel htmlFor="su-name">TOOL NAME *</FieldLabel>
   541→                      <Input
   542→                        id="su-name"
   543→                        value={form.name}
   544→                        onChange={(e) => set("name", e.target.value)}
   545→                        placeholder="Promptly"
   546→                        className={inputCls}
   547→                        maxLength={40}
   548→                      />
   549→                      <FieldError msg={errors.name} />
   550→                    </div>
