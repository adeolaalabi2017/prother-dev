  1036→function PostEditor({
  1037→  apiKey,
  1038→  draft,
  1039→  onDone,
  1040→}: {
  1041→  apiKey: string;
  1042→  draft: PostDraft;
  1043→  onDone: (openSlug?: string) => void;
  1044→}) {
  1045→  const { toast } = useToast();
  1046→  const [f, setF] = useState<PostDraft>(draft);
  1047→  const [busy, setBusy] = useState(false);
  1048→  const isNew = !f.id;
  1049→
  1050→  const save = async (publishOverride?: string) => {
  1051→    setBusy(true);
  1052→    try {
  1053→      const payload = {
  1054→        title: f.title,
  1055→        slug: f.slug || undefined,
  1056→        excerpt: f.excerpt,
  1057→        // Only send the body when rewritten — a metadata-only edit of an
  1058→        // existing post keeps the stored markdown server-side.
  1059→        ...(f.body.trim().length >= 50 ? { body: f.body } : {}),
  1060→        category: f.category,
  1061→        tags: f.tags,
  1062→        coverEmoji: f.coverEmoji,
  1063→        coverGradient: f.coverGradient,
  1064→        author: f.author,
  1065→        status: publishOverride ?? f.status,
  1066→        seoTitle: f.seoTitle || undefined,
  1067→        seoDescription: f.seoDescription || undefined,
  1068→        keywords: f.keywords || undefined,
  1069→      };
  1070→      const res = await adminFetch(
  1071→        apiKey,
  1072→        isNew ? "/api/admin/posts" : `/api/admin/posts/${f.id}`,
  1073→        { method: isNew ? "POST" : "PATCH", body: JSON.stringify(payload) }
  1074→      );
  1075→      const d = (await res.json()) as { ok?: boolean; error?: string; slug?: string };
  1076→      if (!res.ok || !d.ok) {
  1077→        toast({ title: d.error ?? "Save failed", variant: "destructive" });
  1078→        return;
  1079→      }
  1080→      toast({
  1081→        title: publishOverride === "published" ? "Published" : "Saved",
  1082→        description: d.slug ? `/journal/${d.slug}` : undefined,
  1083→      });
  1084→      onDone(publishOverride === "published" ? d.slug : undefined);
  1085→    } catch {
  1086→      toast({ title: "Network error", variant: "destructive" });
  1087→    } finally {
  1088→      setBusy(false);
  1089→    }
  1090→  };
  1091→
  1092→  const remove = async () => {
  1093→    if (!f.id) return;
  1094→    setBusy(true);
  1095→    try {
  1096→      const res = await adminFetch(apiKey, `/api/admin/posts/${f.id}`, { method: "DELETE" });
  1097→      if (res.ok) {
  1098→        toast({ title: "Post deleted" });
  1099→        onDone();
  1100→      }
  1101→    } finally {
  1102→      setBusy(false);
  1103→    }
  1104→  };
  1105→
  1106→  const seoTitle = f.seoTitle || f.title;
  1107→  const seoDesc = f.seoDescription || f.excerpt;
  1108→
  1109→  return (
  1110→    <div className="space-y-5">
  1111→      <div className="grid gap-3 sm:grid-cols-2">
  1112→        <Field label="Title (≤120)">
  1113→          <Input value={f.title} maxLength={120} onChange={(e) => setF({ ...f, title: e.target.value })} className={inputCx} placeholder="How to launch an AI tool…" />
  1114→        </Field>
  1115→        <Field label="Slug (auto from title if blank)">
  1116→          <Input value={f.slug} onChange={(e) => setF({ ...f, slug: e.target.value })} className={cn(inputCx, "font-mono")} placeholder="how-to-launch-an-ai-tool" />
  1117→        </Field>
  1118→      </div>
  1119→
  1120→      <Field label="Excerpt (20–200 · cards + meta description)">
  1121→        <Textarea value={f.excerpt} rows={2} maxLength={200} onChange={(e) => setF({ ...f, excerpt: e.target.value })} className={inputCx} />
  1122→        <p className="text-[10px] text-white/30">{f.excerpt.length}/200</p>
  1123→      </Field>
  1124→
  1125→      <Field label="Body (markdown: ##, lists, **bold**, `code`, ``` blocks)">
  1126→        <Textarea
  1127→          value={f.body}
  1128→          rows={10}
  1129→          onChange={(e) => setF({ ...f, body: e.target.value })}
  1130→          className={cn(inputCx, "font-mono text-xs leading-relaxed")}
  1131→        />
  1132→        <p className="text-[10px] text-white/30">
  1133→          {f.body.split(/\s+/).filter(Boolean).length} words · ~
