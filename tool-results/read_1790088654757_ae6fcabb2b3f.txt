    35→// ── Formatting helpers (server-only, deterministic UTC — hydration-safe) ─
    36→
    37→const MONTHS = [
    38→  "Jan", "Feb", "Mar", "Apr", "May", "Jun",
    39→  "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
    40→] as const;
    41→
    42→function utcDateLabel(v: Date | string): string {
    43→  const d = typeof v === "string" ? new Date(v) : v;
    44→  if (Number.isNaN(d.getTime())) return "";
    45→  return `${MONTHS[d.getUTCMonth()]} ${d.getUTCDate()}, ${d.getUTCFullYear()}`;
    46→}
    47→
    48→const PRICING_LABEL: Record<string, string> = {
    49→  free: "Free",
    50→  freemium: "Freemium",
    51→  paid: "Paid",
    52→  open_source: "Open Source",
    53→};
    54→
    55→function pricingLine(model: string, price: string | null): string {
    56→  const label = PRICING_LABEL[model] ?? "Free";
    57→  if ((model === "paid" || model === "freemium") && price) {
    58→    return `${label} · from ${price}`;
    59→  }
    60→  return label;
    61→}
    62→
    63→/** Mono chip shared by badges / tags / ratings. */
    64→function chipCx(extra?: string): string {
    65→  return cn(
    66→    "inline-flex items-center gap-1 rounded-full border border-white/15 bg-white/5 px-2.5 py-1 font-mono text-[10px] tracking-wider uppercase",
    67→    extra
    68→  );
    69→}

    73→// ── Metadata ──────────────────────────────────────────────────────────────
    74→
    75→export async function generateMetadata({ params }: Params): Promise<Metadata> {
    76→  const { slug } = await params;
    77→  const tool = await db.tool.findUnique({
    78→    where: { slug },
    79→    include: {
    80→      launch: { include: { _count: { select: { votes: true } } } },
    81→      category: { select: { name: true } },
    82→    },
    83→  });
    84→  // Unknown slug or removed/unlisted tool → 404 covers the meta too.
    85→  if (!tool || tool.status !== "live") notFound();
    86→
    87→  // Tomorrow teaser: keep the URL alive but out of the index (thin content).
    88→  if (tool.launch?.scheduled) {
    89→    return {
    90→      title: `${tool.name} — launching soon | Prother`,
    91→      robots: { index: false, follow: true },
    92→    };
    93→  }
    94→
    95→  const votes = (tool.launch?.baseUpvotes ?? 0) + (tool.launch?._count.votes ?? 0);
    96→  const title = `${tool.name} — ${tool.tagline} | Prother`;
    97→  const description = clamp(
    98→    `Live on Prother · ${tool.category.name} · ▲ ${votes} votes. ${tool.description || tool.tagline}`,
    99→    200,
   100→  );
   101→  const keywords = tool.tags
   102→    .split("|")
   103→    .map((t) => t.trim())
   104→    .filter(Boolean);
   105→
   106→  return {
   107→    title,
   108→    description,
   109→    keywords: keywords.length > 0 ? keywords : undefined,
   110→    alternates: { canonical: `/tools/${slug}` },
   111→    openGraph: {
   112→      title,
   113→      description,
   114→      type: "article",
   115→      siteName: "Prother",
   116→      images: [
   117→        { url: `/api/og?tool=${encodeURIComponent(slug)}`, width: 1200, height: 630 },
   118→      ],
   119→    },
   120→    twitter: {
   121→      card: "summary_large_image",
   122→      title,
   123→      description,
   124→      images: [`/api/og?tool=${encodeURIComponent(slug)}`],
   125→    },
   126→  };
   127→}
