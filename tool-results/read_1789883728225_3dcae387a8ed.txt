   349→   349→                    const p2 = { x: path.isLeft ? centerX * 0.8 : width - centerX * 0.8, y: centerY };
   350→   350→                    const p3 = { x: centerX, y: centerY };
   351→   351→
   352→   352→                    ctx.beginPath();
   353→   353→                    ctx.moveTo(p0.x, p0.y);
   354→   354→                    ctx.bezierCurveTo(p1.x, p1.y, p2.x, p2.y, p3.x, p3.y);
   355→   355→                    ctx.strokeStyle = 'rgba(255, 255, 255, 0.35)';
   356→   356→                    ctx.lineWidth = 1.2;
   357→   357→                    ctx.setLineDash([1, 4]);
   358→   358→                    ctx.stroke();
   359→   359→                    ctx.setLineDash([]);
   360→   360→
   361→   361→                    path.particles.forEach(p => {
   362→   362→                        p.t += p.speed;
   363→   363→                        if (p.t > 1) {
   364→   364→                            p.t = 0;
   365→   365→                            path.startY += (Math.random() - 0.5) * 10;
   366→   366→                        }
   367→   367→
   368→   368→                        let pos = getBezierPoint(p.t, p0, p1, p2, p3);
   369→   369→
   370→   370→                        let dxTotal = 0, dyTotal = 0;
   371→   371→                        explosions.forEach(exp => {
   372→   372→                            let dx = pos.x - exp.x;
   373→   373→                            let dy = pos.y - exp.y;
   374→   374→                            let dist = Math.hypot(dx, dy);
   375→   375→                            if (dist < exp.radius + 120 && dist > exp.radius - 120) {
   376→   376→                                let force = (1 - Math.abs(dist - exp.radius) / 120) * exp.life;
   377→   377→                                dxTotal += (dx / dist) * force * 80;
   378→   378→                                dyTotal += (dy / dist) * force * 80;
   379→   379→                            }
   380→   380→                        });
   381→   381→                        
   382→   382→                        pos.x += dxTotal;
   383→   383→                        pos.y += dyTotal;
   384→   384→
   385→   385→                        ctx.fillStyle = `rgba(255, 255, 255, 0.7)`;
   386→   386→                        ctx.fillRect(pos.x - 1.5, pos.y - 1.5, 3, 3);
   387→   387→                    });
   388→   388→                });
   389→   389→                
   390→   390→                requestAnimationFrame(render);
   391→   391→            }
   392→   392→            
   393→   393→            render();
   394→   394→        });
   395→   395→    </script>
   396→   396→</body>
   397→   397→</html>`;
   398→   398→
   399→   399→const GATEWAY_FLOW_DEFINITION: EffectDefinition = {
   400→   400→  title: "Gateway Flow",
   401→   401→  source: gatewayFlowSource,
   402→   402→  supportsMode: true,
   403→   403→  background: (mode) => (mode === "light" ? LIGHT_PAPER : "#000000"),
   404→   404→  targets: [{ selector: "#flow-canvas", role: "background" }],
   405→   405→  patch(source, { size, density, mode }) {
   406→   406→    let next = source

   434→   434→function buildFocusedDocument(
   435→   435→  definition: EffectDefinition,
   436→   436→  knobs: BakeKnobs & { speed: number; opacity: number },
   437→   437→) {
   438→   438→  const mode = knobs.mode;
   439→   439→  const background = resolveBackground(definition.background, mode);
   440→   440→  const targetJson = JSON.stringify(definition.targets).replace(
   441→   441→    /</g,
   442→   442→    "\\u003c",
   443→   443→  );
   444→   444→  const controlsJson = JSON.stringify({
   445→   445→    mode,
   446→   446→    speed: knobs.speed,
   447→   447→    size: knobs.size,
   448→   448→    gap: knobs.gap,
   449→   449→    length: knobs.length,
   450→   450→    density: knobs.density,
   451→   451→    strokeWidth: knobs.strokeWidth,
   452→   452→    opacity: knobs.opacity,
   453→   453→  }).replace(/</g, "\\u003c");
   454→   454→  const patchedSource = definition.patch
   455→   455→    ? definition.patch(definition.source, {
   456→   456→        size: knobs.size,
   457→   457→        gap: knobs.gap,
   458→   458→        length: knobs.length,
   459→   459→        density: knobs.density,
   460→   460→        strokeWidth: knobs.strokeWidth,
   461→   461→        mode,
   462→   462→      })
   463→   463→    : definition.source;
   464→   464→  const focusStyle = `<style data-threeui-focus>
   465→   465→html, body { width: 100% !important; height: 100% !important; min-height: 0 !important; margin: 0 !important; padding: 0 !important; overflow: hidden !important; background: ${background} !important; }
   466→   466→body { position: relative !important; display: flex !important; align-items: center !important; justify-content: center !important; }
   467→   467→body > * { visibility: hidden !important; }
   468→   468→body[data-threeui-ready] > [data-threeui-role] { visibility: visible !important; }
   469→   469→[data-threeui-residual] { display: none !important; }
   470→   470→[data-threeui-role="background"] { position: fixed !important; inset: 0 !important; width: 100% !important; height: 100% !important; max-width: none !important; max-height: none !important; z-index: 0 !important; opacity: 1 !important; pointer-events: none !important; }
   471→   471→[data-threeui-role="ui"] { position: relative !important; z-index: 1 !important; width: min(calc(100% - 32px), var(--threeui-target-width, 1040px)) !important; max-width: none !important; max-height: calc(100% - 32px) !important; margin: auto !important; overflow: auto !important; opacity: 1 !important; transform: none !important; filter: none !important; flex: none !important; box-sizing: border-box !important; }
   472→   472→${definition.focusCss ?? ""}
   473→   473→</style>`;
   474→   474→  const controlScript = `<script data-threeui-controls>
   475→   475→(function () {
   476→   476→  var controls = ${controlsJson};
   477→   477→  window.__SF_CONTROLS = controls;
   478→   478→  var origin = performance.now();
   479→   479→  var virtual = 0;
   480→   480→  var last = origin;
   481→   481→  var performanceNow = performance.now.bind(performance);
   482→   482→  var dateNow = Date.now.bind(Date);
   483→   483→  var dateOrigin = dateNow();
   484→   484→  performance.now = function () {
   485→   485→    var real = performanceNow();
   486→   486→    virtual += (real - last) * (controls.speed || 1);
   487→   487→    last = real;
   488→   488→    return origin + virtual;
   489→   489→  };
   490→   490→  Date.now = function () {
   491→   491→    return dateOrigin + (performance.now() - origin);
   492→   492→  };
   493→   493→  var raf = window.requestAnimationFrame.bind(window);
   494→   494→  window.requestAnimationFrame = function (callback) {
   495→   495→    return raf(function () {
   496→   496→      callback(performance.now());
   497→   497→    });
   498→   498→  };
   499→   499→  function applyVisual() {
   500→   500→    var opacity = controls.opacity == null ? 1 : controls.opacity;
   501→   501→    var size = controls.size == null ? 1 : controls.size;
   502→   502→    Array.prototype.forEach.call(document.querySelectorAll('[data-threeui-role]'), function (element) {
   503→   503→      element.style.opacity = String(opacity);
   504→   504→      if (element.getAttribute('data-threeui-role') === 'ui') {
   505→   505→        element.style.transform = 'scale(' + size + ')';
   506→   506→        element.style.transformOrigin = 'center center';
   507→   507→      }
   508→   508→    });
   509→   509→  }
   510→   510→  window.addEventListener('message', function (event) {
   511→   511→    if (!event.data || event.data.type !== 'threeui-controls') return;
   512→   512→    var next = event.data.controls || {};
   513→   513→    Object.keys(next).forEach(function (key) { controls[key] = next[key]; });
   514→   514→    applyVisual();
   515→   515→  });
   516→   516→  window.__SF_APPLY_CONTROLS = applyVisual;
   517→   517→})();
   518→   518→</script>`;
   519→   519→  const focusScript = `<script data-threeui-focus>
   520→   520→(function () {
   521→   521→  var isolated = false;
   522→   522→  function isolate() {
   523→   523→    if (isolated) return;
   524→   524→    var specs = ${targetJson};
   525→   525→    var roots = [];
   526→   526→    specs.forEach(function (spec) {
   527→   527→      var element = document.querySelector(spec.selector);
   528→   528→      if (!element) return;
   529→   529→      element.setAttribute('data-threeui-role', spec.role);
   530→   530→      if (spec.width) element.style.setProperty('--threeui-target-width', spec.width);
   531→   531→      if (!roots.some(function (root) { return root.contains(element); })) roots.push(element);
   532→   532→    });
   533→   533→    if (!roots.length) return;
   534→   534→    isolated = true;
   535→   535→    roots.forEach(function (root) { document.body.appendChild(root); });
   536→   536→    Array.from(document.body.children).forEach(function (element) {
   537→   537→      if (roots.indexOf(element) !== -1) return;
   538→   538→      element.setAttribute('data-threeui-residual', '');
   539→   539→      element.setAttribute('aria-hidden', 'true');
   540→   540→      if ('inert' in element) element.inert = true;
   541→   541→    });
   542→   542→    document.body.setAttribute('data-threeui-ready', '');
   543→   543→    if (window.__SF_APPLY_CONTROLS) window.__SF_APPLY_CONTROLS();
   544→   544→    requestAnimationFrame(function () { window.dispatchEvent(new Event('resize')); });
   545→   545→  }
   546→   546→  function scheduleIsolation() { setTimeout(isolate, 100); }
   547→   547→  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', scheduleIsolation, { once: true });
   548→   548→  else scheduleIsolation();
   549→   549→  window.addEventListener('load', isolate, { once: true });
   550→   550→})();
   551→   551→</script>`;
   552→   552→  return patchedSource
   553→   553→    .replace(/<head([^>]*)>/i, `<head$1>${controlScript}${focusStyle}`)
   554→   554→    .replace(/<\/body>/i, `${focusScript}</body>`);
   555→   555→}

   557→   557→function GatewayFlowFrame({
   558→   558→  definition,
   559→   559→  mode,
   560→   560→  speed = GATEWAY_FLOW_DEFAULTS.speed,
   561→   561→  size = GATEWAY_FLOW_DEFAULTS.size,
   562→   562→  gap = GATEWAY_FLOW_DEFAULTS.gap,
   563→   563→  length = GATEWAY_FLOW_DEFAULTS.length,
   564→   564→  density = GATEWAY_FLOW_DEFAULTS.density,
   565→   565→  strokeWidth = GATEWAY_FLOW_DEFAULTS.strokeWidth,
   566→   566→  opacity = GATEWAY_FLOW_DEFAULTS.opacity,
   567→   567→  hue = GATEWAY_FLOW_DEFAULTS.hue,
   568→   568→  saturation = GATEWAY_FLOW_DEFAULTS.saturation,
   569→   569→  brightness = GATEWAY_FLOW_DEFAULTS.brightness,
   570→   570→  className,
   571→   571→  style,
   572→   572→}: GatewayFlowProps & { definition: EffectDefinition }) {
   573→   573→  const iframeRef = useRef<HTMLIFrameElement>(null);
   574→   574→  const requestedMode =
   575→   575→    mode ?? definition.defaultMode ?? GATEWAY_FLOW_DEFAULTS.mode;
   576→   576→  const automaticMode = useAutomaticMode(requestedMode === "auto");
   577→   577→  const resolvedMode =
   578→   578→    requestedMode === "auto"
   579→   579→      ? automaticMode
   580→   580→      : resolveMode(requestedMode, GATEWAY_FLOW_DEFAULTS.mode);
   581→   581→  const background = resolveBackground(definition.background, resolvedMode);
   582→   582→  const safeSpeed = clamp(speed, 0, 3);
   583→   583→  const safeSize = clamp(size, 0.05, 200);
   584→   584→  const safeGap = clamp(gap, 0, 64);
   585→   585→  const safeLength = clamp(length, 0.35, 2.5);
   586→   586→  const safeDensity = clamp(density, 0.25, 2.5);
   587→   587→  const safeStrokeWidth = clamp(strokeWidth, 0.25, 8);
   588→   588→  const safeOpacity = clamp(opacity, 0.05, 1);
   589→   589→  const safeHue = clamp(hue, -180, 180);
   590→   590→  const safeSaturation = clamp(saturation, 0, 2);
   591→   591→  const safeBrightness = clamp(brightness, 0.35, 1.65);
   592→   592→
   593→   593→  // Rebuild when baked geometry/mode knobs change. Speed/opacity stay live via postMessage + time wrap.
   594→   594→  const source = useMemo(
   595→   595→    () =>
   596→   596→      buildFocusedDocument(definition, {
   597→   597→        mode: resolvedMode,
   598→   598→        speed: GATEWAY_FLOW_DEFAULTS.speed,
   599→   599→        size: safeSize,
   600→   600→        gap: safeGap,
   601→   601→        length: safeLength,
   602→   602→        density: safeDensity,
   603→   603→        strokeWidth: safeStrokeWidth,
   604→   604→        opacity: GATEWAY_FLOW_DEFAULTS.opacity,
   605→   605→      }),
   606→   606→    [
   607→   607→      definition,
   608→   608→      resolvedMode,
   609→   609→      safeDensity,
   610→   610→      safeGap,
   611→   611→      safeLength,
   612→   612→      safeSize,
   613→   613→      safeStrokeWidth,
   614→   614→    ],
   615→   615→  );
   616→   616→
   617→   617→  useEffect(() => {
   618→   618→    const frame = iframeRef.current?.contentWindow;
   619→   619→    if (!frame) return;
   620→   620→    frame.postMessage(
   621→   621→      {
   622→   622→        type: "threeui-controls",
   623→   623→        controls: {
   624→   624→          mode: resolvedMode,
   625→   625→          speed: safeSpeed,
   626→   626→          size: safeSize,
   627→   627→          gap: safeGap,
   628→   628→          length: safeLength,
   629→   629→          density: safeDensity,
   630→   630→          strokeWidth: safeStrokeWidth,
   631→   631→          opacity: safeOpacity,
   632→   632→        },
   633→   633→      },
   634→   634→      "*",
   635→   635→    );
   636→   636→  }, [
   637→   637→    resolvedMode,
   638→   638→    safeDensity,
   639→   639→    safeGap,
   640→   640→    safeLength,
   641→   641→    safeOpacity,
   642→   642→    safeSize,
   643→   643→    safeSpeed,
   644→   644→    safeStrokeWidth,
   645→   645→    source,
   646→   646→  ]);
   647→   647→
   648→   648→  const filter =
   649→   649→    safeHue === 0 && safeSaturation === 1 && safeBrightness === 1
   650→   650→      ? undefined
   651→   651→      : `hue-rotate(${safeHue}deg) saturate(${safeSaturation}) brightness(${safeBrightness})`;
   652→   652→
   653→   653→  return (
   654→   654→    <iframe
   655→   655→      ref={iframeRef}
   656→   656→      className={className}
   657→   657→      title={definition.title}
   658→   658→      srcDoc={source}
   659→   659→      sandbox="allow-scripts"
   660→   660→      loading="eager"
   661→   661→      style={{
   662→   662→        display: "block",
   663→   663→        width: "100%",
   664→   664→        height: "100%",
   665→   665→        border: 0,
   666→   666→        background,
   667→   667→        filter,
   668→   668→        ...style,
   669→   669→      }}
   670→   670→    />
   671→   671→  );
   672→   672→}
   673→   673→
   674→   674→export default function GatewayFlow(props: GatewayFlowProps) {
   675→   675→  return <GatewayFlowFrame {...props} definition={GATEWAY_FLOW_DEFINITION} />;
   676→   676→}
