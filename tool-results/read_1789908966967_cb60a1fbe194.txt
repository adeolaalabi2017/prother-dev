    93→  "from-rose-400 to-red-600",

   100→function avatarGradient(author: string): string {
   101→  let h = 0;
   102→  for (let i = 0; i < author.length; i++) h = (h * 31 + author.charCodeAt(i)) >>> 0;
   103→  return AVATAR_GRADIENTS[h % AVATAR_GRADIENTS.length];
   104→}

   106→function relTime(iso: string): string {
   107→  const s = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000);
   108→  if (s < 60) return "just now";
   109→  const m = Math.floor(s / 60);
   110→  if (m < 60) return `${m}m ago`;
   111→  const h = Math.floor(m / 60);
   112→  if (h < 24) return `${h}h ago`;
   113→  return `${Math.floor(h / 24)}d ago`;
   114→}

   128→  const [body, setBody] = useState("");
   129→  const [posting, setPosting] = useState(false);
   130→  const [postedFlash, setPostedFlash] = useState(false);
   131→
   132→  useEffect(() => {
   133→    let alive = true;
   134→    setLoading(true);
   135→    setLoadError(false);
   136→    setItems(null);
   137→    fetch(`/api/tools/${encodeURIComponent(slug)}/comments`)
   138→      .then(async (res) => {
   139→        if (!res.ok) throw new Error("Failed to load discussion");

   141→      })
   142→      .then((data) => {
   143→        if (alive) setItems(data.items);
   144→      })
   145→      .catch(() => {
   146→        if (alive) setLoadError(true);
   147→      })
   148→      .finally(() => {
   149→        if (alive) setLoading(false);
   150→      });
   151→    return () => {
   152→      alive = false;
   153→    };
   154→  }, [slug]);
   155→
   156→  // Remember the commenter's display name (per browser).
   157→  useEffect(() => {
   158→    setName(window.localStorage.getItem(COMMENT_NAME_STORAGE) ?? "");
   159→  }, []);
   160→
   161→  const canPost =
   162→    name.trim().length >= 2 && body.trim().length >= 4 && !posting;
   163→
   164→  const onPost = useCallback(async () => {
   165→    if (!canPost) return;
   166→    setPosting(true);
   167→    try {
   168→      const res = await fetch(
   169→        `/api/tools/${encodeURIComponent(slug)}/comments`,
   170→        {
   171→          method: "POST",
   172→          headers: { "Content-Type": "application/json" },
   173→          body: JSON.stringify({ author: name.trim(), body: body.trim() }),
   174→        },
   175→      );
   176→      const data = (await res.json()) as CommentRow & { error?: string };
   177→      if (!res.ok)
   178→        throw new Error(data.error ?? "Could not post — please try again.");
   179→      setItems((prev) => [...(prev ?? []), data]);
   180→      setBody("");
   181→      window.localStorage.setItem(COMMENT_NAME_STORAGE, name.trim());
   182→      setPostedFlash(true);
   183→      window.setTimeout(() => setPostedFlash(false), 1600);
   184→      toast({ title: "Comment posted" });
   185→    } catch (err) {
   186→      toast({
   187→        title: "Could not post comment",
   188→        description:
   189→          err instanceof Error ? err.message : "Please try again.",
   190→        variant: "destructive",
   191→      });
   192→    } finally {
   193→      setPosting(false);
   194→    }
   195→  }, [canPost, slug, name, body, toast]);
   196→
   197→  const count = items?.length ?? 0;
   198→  const counterTone =
   199→    body.length >= COMMENT_BODY_MAX
   200→      ? "text-red-400"
   201→      : body.length >= COMMENT_BODY_WARN
   202→        ? "text-ember"
   203→        : "text-white/35";
   204→
   205→  return (
   206→    <div>
   207→      <div className="flex items-center justify-between gap-2">
   208→        <p className="font-mono text-[10px] tracking-widest text-white/40">
   209→          DISCUSSION ({count})
   210→        </p>
   211→        <p className="font-mono text-[10px] tracking-wider text-white/25">
   212→          MODERATED PER S6
   213→        </p>
   214→      </div>
   215→
   216→      <div
   217→        className="mt-2 max-h-96 space-y-2 overflow-y-auto"
   218→        aria-live="polite"
   219→        aria-label={`Discussion for this tool, ${count} comments`}
   220→      >
   221→        {loading && (
   222→          <>
   223→            <Skeleton className="h-16 w-full rounded-xl bg-white/5" />
   224→            <Skeleton className="h-16 w-5/6 rounded-xl bg-white/5" />
   225→          </>
   226→        )}
   227→
   228→        {!loading && loadError && (
   229→          <p className="rounded-lg border border-red-500/25 bg-red-500/[0.04] p-3 text-xs text-red-300">
   230→            Couldn&apos;t load the discussion. It will appear next time you
   231→            open this listing.
   232→          </p>
   233→        )}
   234→
   235→        {!loading && !loadError && count === 0 && (
   236→          <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed border-white/15 p-6 text-center">
   237→            <MessageSquare className="size-4 text-white/30" aria-hidden />
   238→            <p className="text-sm text-white/70">No comments yet.</p>
   239→            <p className="text-xs text-white/40">
   240→              Start the discussion — ask {makerHandle} anything.
   241→            </p>
   242→          </div>
   243→        )}
   244→
   245→        {items?.map((c) => (
   246→          <motion.div
   247→            key={c.id}
   248→            initial={{ opacity: 0, y: 8 }}
   249→            animate={{ opacity: 1, y: 0 }}
   250→            transition={{ duration: 0.25, ease: "easeOut" }}
   251→            className="flex gap-3 rounded-xl border border-white/10 bg-white/[0.02] p-3.5 transition-colors hover:border-ember/25"
   252→          >
   253→            <span
   254→              aria-hidden
   255→              className={cn(
   256→                "flex size-8 shrink-0 items-center justify-center rounded-full bg-gradient-to-br text-xs font-bold text-black",
   257→                avatarGradient(c.author),
   258→              )}
   259→            >
   260→              {c.author.replace(/^@/, "").charAt(0).toUpperCase()}
   261→            </span>
   262→            <div className="min-w-0 flex-1">
   263→              <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
   264→                <span className="text-sm font-semibold text-white/90">
   265→                  {c.author}
   266→                </span>
   267→                {c.isMaker && (
   268→                  <span className="rounded-full border border-ember/40 bg-ember/15 px-1.5 py-px font-mono text-[9px] tracking-wider text-ember">
   269→                    MAKER

   273→                  {relTime(c.createdAt)}
   274→                </span>
   275→              </div>
   276→              <p className="mt-1 whitespace-pre-wrap break-words text-sm leading-relaxed text-white/75">
   277→                {c.body}
   278→              </p>
   279→            </div>
   280→          </motion.div>
   281→        ))}
   282→      </div>
   283→
   284→      {/* Composer */}
   285→      <div className="mt-3 rounded-xl border border-white/10 bg-white/[0.02] p-3 transition-colors focus-within:border-ember/40">
   286→        <label htmlFor={`c-name-${slug}`} className="sr-only">
   287→          Your display name
   288→        </label>
   289→        <input
   290→          id={`c-name-${slug}`}
   291→          value={name}
   292→          onChange={(e) => setName(e.target.value.slice(0, COMMENT_NAME_MAX))}
   293→          placeholder="Your name"
   294→          maxLength={COMMENT_NAME_MAX}
   295→          autoComplete="name"
   296→          className="w-full border-b border-white/10 bg-transparent pb-2 font-mono text-sm text-white placeholder:text-white/30 focus:border-ember/60 focus:outline-none"
   297→        />
   298→        <label htmlFor={`c-body-${slug}`} className="sr-only">
   299→          Write a comment
   300→        </label>
   301→        <textarea
   302→          id={`c-body-${slug}`}
   303→          value={body}
   304→          onChange={(e) =>
   305→            setBody(e.target.value.slice(0, COMMENT_BODY_MAX))

   307→          onKeyDown={(e) => {
   308→            if ((e.metaKey || e.ctrlKey) && e.key === "Enter") {
   309→              e.preventDefault();
   310→              void onPost();
   311→            }
   312→          }}
   313→          placeholder="Add to the discussion…"
   314→          rows={3}
   315→          maxLength={COMMENT_BODY_MAX}
   316→          className={cn(
   317→            "mt-2 w-full resize-none bg-transparent text-sm leading-relaxed text-white placeholder:text-white/30 focus:outline-none",
   318→            body.length >= COMMENT_BODY_MAX && "text-red-300",
   319→          )}
   320→        />
   321→        <div className="mt-2 flex items-center justify-between gap-3">
   322→          <p className="font-mono text-[10px] text-white/30">
   323→            ⌘↵ TO POST · BE CONSTRUCTIVE (S6)
   324→          </p>
   325→          <div className="flex items-center gap-3">
   326→            <span
   327→              className={cn(
   328→                "font-mono text-[10px] tabular-nums transition-colors",
   329→                counterTone,
   330→              )}
   331→              aria-live="polite"
   332→            >
   333→              {body.length}/{COMMENT_BODY_MAX}
   334→            </span>
   335→            <button
   336→              type="button"
   337→              onClick={() => void onPost()}
   338→              disabled={!canPost}
   339→              aria-label="Post comment"
   340→              className={cn(
   341→                "inline-flex h-8 items-center gap-1.5 rounded-lg px-3 text-xs font-semibold transition active:scale-95",

   348→              {posting ? (
   349→                <>
   350→                  <Loader2 className="size-3.5 animate-spin" aria-hidden />
   351→                  Posting…
   352→                </>
   353→              ) : postedFlash ? (
   354→                <>
   355→                  <Check className="size-3.5" aria-hidden />
   356→                  Posted
   357→                </>
   358→              ) : (
   359→                <>
   360→                  <Send className="size-3.5" aria-hidden />
   361→                  Post
   362→                </>
   363→              )}
   364→            </button>
   365→          </div>
   366→        </div>
   367→      </div>
   368→    </div>
   369→  );
   370→}

   393→  const { toast } = useToast();
   394→  const openTool = useExplorer((s) => s.openTool);
   395→  const setCategoryFilter = useExplorer((s) => s.setCategoryFilter);
   396→  const [detail, setDetail] = useState<ToolDetailResponse | null>(null);
   397→  const [loading, setLoading] = useState(true);
   398→  const [error, setError] = useState<string | null>(null);
   399→  const [vote, setVote] = useState<{ votes: number; voted: boolean } | null>(
   400→    null,
   401→  );
   402→  const [copied, setCopied] = useState(false);
   403→
   404→  useEffect(() => {
   405→    let alive = true;
   406→    fetch(`/api/tools/${slug}?vk=${encodeURIComponent(getVoterKey())}`)
   407→      .then(async (res) => {
   408→        if (!res.ok)
   409→          throw new Error(
   410→            res.status === 404 ? "Tool not found" : "Failed to load tool",
   411→          );
   412→        return (await res.json()) as ToolDetailResponse;
   413→      })
   414→      .then((data) => {
   415→        if (alive) {
   416→          setDetail(data);
   417→          setVote({ votes: data.votes, voted: data.voted });
   418→        }
   419→      })
   420→      .catch((err: unknown) => {
   421→        if (alive)
   422→          setError(err instanceof Error ? err.message : "Failed to load tool");
   423→      })
   424→      .finally(() => {
   425→        if (alive) setLoading(false);
   426→      });
   427→    return () => {
   428→      alive = false;

   430→  }, [slug]);

   432→  const onVote = useCallback(async () => {
   433→    if (!detail?.launchId || !vote) return;

   435→    setVote({ votes: vote.votes + (nextVoted ? 1 : -1), voted: nextVoted });

   437→      const res = await fetch("/api/vote", {
   438→        method: "POST",
   439→        headers: { "Content-Type": "application/json" },
   440→        body: JSON.stringify({
   441→          launchId: detail.launchId,
   442→          voterKey: getVoterKey(),
   443→        }),
   444→      });
   445→      const data = (await res.json()) as { voted: boolean; votes: number };
   446→      setVote({ votes: data.votes, voted: data.voted });
   447→      // Sync the feed list behind the modal.
   448→      window.dispatchEvent(
   449→        new CustomEvent("prother:vote", {
   450→          detail: { launchId: detail.launchId, votes: data.votes, voted: data.voted },
   451→        }),
   452→      );
   453→    } catch {
   454→      setVote(vote);
   455→      toast({
   456→        title: "Vote failed",
   457→        description: "Please try again.",
   458→        variant: "destructive",
   459→      });
   460→    }
   461→  }, [detail, vote, toast]);
   462→
   463→  const passedCount = detail?.standards.filter((s) => s.passed).length ?? 0;
   464→
   465→  const shareUrl =
   466→    typeof window !== "undefined"
   467→      ? `${window.location.origin}/?tool=${encodeURIComponent(slug)}`
   468→      : "";

   480→        title: "Could not copy the link",

   482→        variant: "destructive",
   483→      });
   484→    }
   485→  }, [shareUrl, toast]);

   495→        .querySelector("#feed")

   498→  }, [detail, onClose, setCategoryFilter]);

   505→        <div className="p-10 text-center">
   506→          <p className="text-sm text-white/60">{error}</p>
   507→          <button
   508→            type="button"
   509→            onClick={onClose}
   510→            className="mt-4 font-mono text-xs text-ember hover:underline"
   511→          >
   512→            ← Close
   513→          </button>
   514→        </div>
   515→      )}
   516→
   517→      {!loading && !error && detail && vote && (
   518→        <>
   519→          {/* Header with gradient banner + ember glow / dot texture */}
   520→          <div className="relative">
   521→            <div
   522→              aria-hidden
   523→              className={cn(
   524→                "h-20 w-full bg-gradient-to-br opacity-30",
   525→                detail.gradient,
   526→              )}
   527→            />
   528→            <div
   529→              aria-hidden
   530→              className="pointer-events-none absolute inset-x-0 top-0 h-20 bg-[radial-gradient(340px_120px_at_85%_0%,rgba(255,106,0,0.22),transparent_70%)]"
   531→            />
   532→            <div

   535→            />
   536→            <div className="relative flex items-end gap-4 px-6 pb-4">
   537→              <div
   538→                aria-hidden
   539→                className={cn(
   540→                  "-mt-9 flex size-16 shrink-0 items-center justify-center rounded-2xl border-4 border-coal bg-gradient-to-br text-2xl shadow-xl",
   541→                  detail.gradient,
   542→                )}
   543→              >
   544→                {detail.emoji}
   545→              </div>
   546→              <div className="min-w-0 pb-1">
   547→                <div className="flex flex-wrap items-center gap-2">
   548→                  <h2 className="text-2xl font-black tracking-tight">
   549→                    {detail.name}
   550→                  </h2>
   551→                  {detail.badges.editorsPick && (
   552→                    <span className="rounded-full border border-ember/30 bg-ember/15 px-2 py-0.5 font-mono text-[10px] text-ember">

  1106→// ── Mounted-once explorer + ⌘K shortcut ──────────────────────────────────

  1108→export function ToolExplorer() {
  1109→  const setSearch = useExplorer((s) => s.setSearch);
  1110→  const openTool = useExplorer((s) => s.openTool);
  1111→
  1112→  // ⌘K / ctrl+K opens the command palette.
  1113→  useEffect(() => {
  1114→    const onKey = (e: KeyboardEvent) => {
  1115→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
  1116→        e.preventDefault();
  1117→        setSearch(true);
  1118→      }
  1119→    };
  1120→    window.addEventListener("keydown", onKey);
  1121→    return () => window.removeEventListener("keydown", onKey);
  1122→  }, [setSearch]);

  1124→  // Shareable deep links open the detail modal:
  1125→  //  - /#tool=<slug>     hash form (works on load AND same-document nav)
  1126→  //  - /?tool=<slug>     canonical form (server-reachable → real OG unfurl;
  1127→  //                      copied by the modal's Copy-link action)
  1128→  // Clearing the hash (e.g. browser Back) closes the modal again.
  1129→  useEffect(() => {
  1130→    const applyHash = (initial = false) => {
  1131→      const m = window.location.hash.match(/^#tool=([^&]+)$/);
  1132→      if (m?.[1]) openTool(decodeURIComponent(m[1]));
  1133→      else if (!initial) useExplorer.getState().closeTool();
  1134→    };
  1135→    applyHash(true);
  1136→    const onHash = () => applyHash(false);
  1137→    window.addEventListener("hashchange", onHash);
  1138→    return () => window.removeEventListener("hashchange", onHash);
  1139→  }, [openTool]);
  1140→
  1141→  // Canonical ?tool= form: open the modal once at mount, then normalize the
  1142→  // URL to the hash form (keeps ?cat=/?tab=, drops the query param) so the
  1143→  // address bar matches what in-app navigation produces.
  1144→  useEffect(() => {
  1145→    const t = new URLSearchParams(window.location.search).get("tool");
  1146→    if (!t) return;
  1147→    openTool(t);
  1148→    const url = new URL(window.location.href);
  1149→    url.searchParams.delete("tool");
  1150→    window.history.replaceState(null, "", url.pathname + url.search + url.hash);
  1151→  }, [openTool]);
  1152→
  1153→  // Restore a shared/filtered category from the URL on load (?cat=<slug>)
  1154→  // and jump straight to the feed — a shared ?cat= link is an intent to browse.
  1155→  useEffect(() => {
  1156→    const cat = new URLSearchParams(window.location.search).get("cat");
  1157→    if (cat && CATEGORIES.some((c) => c.slug === cat)) {
  1158→      useExplorer.getState().setCategoryFilter(cat);
  1159→      // Wait a beat for the feed to render before smooth-scrolling to it.
  1160→      const t = window.setTimeout(() => {
  1161→        document
  1162→          .querySelector("#feed")
  1163→          ?.scrollIntoView({ behavior: "smooth" });
  1164→      }, 500);
  1165→      return () => window.clearTimeout(t);
  1166→    }
  1167→  }, []);
  1168→

  1170→    <>
  1171→      <CommandPalette />
  1172→      <ToolDetailDialog />
  1173→    </>
  1174→  );
