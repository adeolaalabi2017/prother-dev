   273→  1473→type Delta = { text: string; dir: "up" | "down" | "flat"; title: string };
   274→  1474→
   275→  1475→/** Honest delta vs yesterday — no fabrication when the base is zero. */
   276→  1476→function computeDelta(cur: number, prev: number): Delta {
   277→  1477→  if (prev === 0 && cur === 0)
   278→  1478→    return { text: "±0", dir: "flat", title: "No change vs yesterday" };
   279→  1479→  if (prev === 0)
   280→  1480→    return { text: "↑ new", dir: "up", title: `${cur} today · none yesterday` };
   281→  1481→  const pct = Math.round(((cur - prev) / prev) * 100);
   282→  1482→  if (pct === 0)
   283→  1483→    return { text: "±0%", dir: "flat", title: `${cur} today vs ${prev} yesterday` };
   284→  1484→  return pct > 0
   285→  1485→    ? { text: `↑ ${pct}%`, dir: "up", title: `${cur} today vs ${prev} yesterday` }
   286→  1486→    : { text: `↓ ${Math.abs(pct)}%`, dir: "down", title: `${cur} today vs ${prev} yesterday` };
   287→  1487→}
   288→  1488→

   290→  1490→  up: "bg-emerald-400/10 text-emerald-400/80",
   291→  1491→  down: "bg-red-400/10 text-red-400/80",
   292→  1492→  flat: "bg-white/5 text-white/60",
   293→  1493→};
