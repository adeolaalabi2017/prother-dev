   109→function urlOk(value: string): boolean {
   110→  try {
   111→    const u = new URL(value.trim());
   112→    return u.protocol === "http:" || u.protocol === "https:";
   113→  } catch {
   114→    return false;
   115→  }
   116→}

   118→const fmtCtr = (ctr: number) => `${ctr.toFixed(1)}%`;

   120→function fmtMoney(cents: number | null | undefined): string {
   121→  if (!cents || cents <= 0) return "no cap";
   122→  const dollars = cents / 100;
   123→  return `$${dollars % 1 === 0 ? dollars.toLocaleString("en-US") : dollars.toFixed(2)} cap`;
   124→}

   444→// ── Serving controls (Task 27) ─────────────────────────────────────────

   446→const SERVING_MASTER_KEY = "ads.master";
   447→const servingPlacementKey = (p: Placement) => `ads.placement.${p}`;
   448→
   449→/**
   450→ * Live kill switches, backed by the SiteSetting KV via /api/admin/settings —
   451→ * the same keys /api/ads/serve reads before counting an impression. Flipping
   452→ * one takes effect on the next slot request, no deploy. When a placement is
   453→ * off, pages don't even mount the client island (zero ad JS).
   454→ */
   455→function ServingControls({ apiKey }: { apiKey: string }) {
   456→  const { toast } = useToast();
   457→  const [config, setConfig] = useState<{
   458→    master: boolean;
   459→    placements: Record<Placement, boolean>;
   460→  } | null>(null);
   461→  const [busyKey, setBusyKey] = useState<string | null>(null);
   462→
   463→  useEffect(() => {
   464→    let alive = true;
   465→    adminFetch(apiKey, "/api/admin/settings")
   466→      .then((r) => {
   467→        if (!r.ok) throw new Error(String(r.status));
   468→        return r.json() as Promise<{ settings?: Record<string, string> }>;
   469→      })
   470→      .then((d) => {
   471→        if (!alive) return;
   472→        const s = d.settings ?? {};
   473→        setConfig({
   474→          master: s[SERVING_MASTER_KEY] !== "0",
   475→          placements: Object.fromEntries(
   476→            PLACEMENTS.map((p) => [p.value, s[servingPlacementKey(p.value)] !== "0"])
   477→          ) as Record<Placement, boolean>,
   478→        });
   479→      })
   480→      .catch(() => {
   481→        if (!alive) return;
   482→        setConfig({
   483→          master: true,
   484→          placements: Object.fromEntries(
   485→            PLACEMENTS.map((p) => [p.value, true])
   486→          ) as Record<Placement, boolean>,
   487→        });
   488→      });
   489→    return () => {
   490→      alive = false;
   491→    };
   492→  }, [apiKey]);
   493→
   494→  const put = useCallback(
   495→    async (key: string, on: boolean, label: string) => {
   496→      setBusyKey(key);
   497→      try {
   498→        const res = await adminFetch(apiKey, "/api/admin/settings", {
   499→          method: "PUT",
   500→          body: JSON.stringify({ key, value: on ? "1" : "0" }),
   501→        });
   502→        const d = (await res.json()) as { ok?: boolean; error?: string };
   503→        if (!res.ok || !d.ok) throw new Error(d.error ?? "Save failed");
   504→        setConfig((cur) =>
   505→          !cur
   506→            ? cur
   507→            : key === SERVING_MASTER_KEY
   508→              ? { ...cur, master: on }
   509→              : {
   510→                  ...cur,
   511→                  placements: {
   512→                    ...cur.placements,
   513→                    [key.slice("ads.placement.".length) as Placement]: on,
   514→                  },
   515→                }
   516→        );
   517→        toast({
   518→          title: `${label} ${on ? "enabled" : "disabled"}`,
   519→          description: on ? undefined : "Slots render nothing while off.",
   520→        });
   521→      } catch (err) {
   522→        toast({
   523→          title: err instanceof Error ? err.message : "Network error",
   524→          variant: "destructive",
   525→        });
   526→      } finally {
   527→        setBusyKey(null);
   528→      }
   529→    },
   530→    [apiKey, toast]
   531→  );
   532→
   533→  return (
   534→    <div className="rounded-xl border border-white/10 bg-white/[0.02] p-4">
   535→      <div className="flex items-center justify-between gap-3">
   536→        <div className="min-w-0">
   537→          <p className="text-sm font-bold text-white">Serving controls</p>
   538→          <p className="text-xs text-white/45">
   539→            Live kill switches — no deploy. Off slots render nothing and spend no
   540→            impressions.
   541→          </p>
   542→        </div>
   543→        <div className="flex shrink-0 items-center gap-2">
   544→          <Power
   545→            className={cn("size-4", config?.master ? "text-ember" : "text-white/30")}
   546→            aria-hidden
   547→          />
   548→          <Switch
   549→            checked={config?.master ?? false}
   550→            disabled={!config || busyKey === SERVING_MASTER_KEY}
   551→            onCheckedChange={(v) => void put(SERVING_MASTER_KEY, v, "All ad serving")}
   552→            aria-label="Master ad serving switch"
   553→          />
   554→        </div>
   555→      </div>
   556→
   557→      <div className="mt-3 grid gap-1.5 border-t border-white/10 pt-3 sm:grid-cols-2">
   558→        {PLACEMENTS.map((p) => (
   559→          <div
   560→            key={p.value}
   561→            className="flex items-center justify-between gap-3 rounded-lg border border-white/5 bg-white/[0.02] px-3 py-2"
   562→          >
   563→            <span className="min-w-0">
   564→              <span className="block truncate font-mono text-[11px] tracking-wider text-white/70 uppercase">
   565→                {p.label}
   566→              </span>
   567→              <span className="block truncate font-mono text-[9px] text-white/30">
   568→                {p.value}
   569→              </span>
   570→            </span>
   571→            <Switch
   572→              checked={config?.placements[p.value] ?? false}
   573→              disabled={!config || busyKey === servingPlacementKey(p.value)}
   574→              onCheckedChange={(v) =>
   575→                void put(servingPlacementKey(p.value), v, p.label)
   576→              }
   577→              aria-label={`Serve ${p.label} placement`}
   578→            />
   579→          </div>
   580→        ))}
   581→      </div>
   582→    </div>
   583→  );

   588→/**
   589→ * 7-day fill accounting per placement, read from the same GET as the
   590→ * campaign list (measurement field). "Unfill" = requested slots that did
   591→ * NOT yield a direct-sold campaign — house fills included, since the slot
   592→ * was requested and no advertiser paid for it. That number is the sales
   593→ * pitch: it is the inventory currently going out at $0.
   594→ */
   595→function MeasurementCard({ m }: { m: Measurement | null }) {
   596→  const maxServed = m
   597→    ? Math.max(1, ...m.perPlacement.map((p) => p.served))
   598→    : 1;
   599→  return (
   600→    <div className="rounded-xl border border-white/10 bg-white/[0.02] p-4">
   601→      <div className="flex items-center justify-between gap-3">
   602→        <div className="min-w-0">
   603→          <p className="text-sm font-bold text-white">Measurement — 7-day fill</p>
   604→          <p className="text-xs text-white/45">
   605→            Served vs. unsold demand per placement. Unfill = no campaign (house fills
   606→            included) — that gap is open selling inventory.
   607→          </p>
   608→        </div>
   609→        <span className="shrink-0 rounded-full border border-white/10 px-2 py-0.5 font-mono text-[9px] tracking-wider text-white/40 uppercase">
   610→          {m ? `${m.days[0]?.day} → ${m.days[m.days.length - 1]?.day}` : "…"}
   611→        </span>
   612→      </div>
   613→
   614→      <div className="mt-3 space-y-1.5 border-t border-white/10 pt-3">
   615→        {(m?.perPlacement ?? PLACEMENTS.map((p) => ({
   616→          placement: p.value,
   617→          served: 0,
   618→          house: 0,
   619→          unfilled: 0,
   620→          unfillRate: 0,
   621→        }))).map((row) => {
   622→          const label =
   623→            PLACEMENTS.find((p) => p.value === row.placement)?.label ?? row.placement;
   624→          const requested = row.served + row.house + row.unfilled;
   625→          return (
   626→            <div
   627→              key={row.placement}
   628→              className="flex items-center gap-3 rounded-lg border border-white/5 bg-white/[0.02] px-3 py-2"
   629→            >
   630→              <span className="w-28 shrink-0 truncate font-mono text-[11px] tracking-wider text-white/70 uppercase">
   631→                {label}
   632→              </span>
   633→              <span className="w-16 shrink-0 text-right font-mono text-[11px] tabular-nums text-white">
   634→                {row.served.toLocaleString("en-US")}
   635→                <span className="ml-1 text-white/35">imp</span>
   636→              </span>
   637→              <span className="hidden w-14 shrink-0 text-right font-mono text-[11px] tabular-nums text-white/45 sm:block">
   638→                {row.house} hs
   639→              </span>
   640→              <span className="hidden w-14 shrink-0 text-right font-mono text-[11px] tabular-nums text-white/45 sm:block">
   641→                {row.unfilled} un
   642→              </span>
   643→              {/* served share bar + unfill rate readout */}
   644→              <span className="flex min-w-0 flex-1 items-center gap-2">
   645→                <span
   646→                  aria-hidden
   647→                  className="h-1 min-w-0 flex-1 overflow-hidden rounded-full bg-white/10"
   648→                >
   649→                  <span
   650→                    className="block h-full rounded-full bg-ember transition-all"
   651→                    style={{ width: `${Math.round((row.served / maxServed) * 100)}%` }}
   652→                  />
   653→                </span>
   654→                <span
   655→                  className={cn(
   656→                    "w-16 shrink-0 text-right font-mono text-[10px] tabular-nums",
   657→                    requested === 0
   658→                      ? "text-white/30"
   659→                      : row.unfillRate >= 50
   660→                        ? "text-amber-400"
   661→                        : "text-emerald-400"
   662→                  )}
   663→                  title="Unfill rate — (house + unfilled) / requested"
   664→                >
   665→                  {requested === 0 ? "—" : `${row.unfillRate}% unfill`}
   666→                </span>
   667→              </span>
   668→            </div>
   669→          );
   670→        })}
   671→      </div>
   672→
   673→      <p className="mt-3 font-mono text-[10px] leading-relaxed text-white/25">
   674→        VIEWABILITY (VW) IS COUNTED MRC-STYLE — ≥50% OF THE CREATIVE ON SCREEN FOR ≥1S —
   675→        PER CAMPAIGN BELOW · DISABLED SLOTS COUNT NOTHING (AN OFF SWITCH IS NOT A FILL)
   676→      </p>
   677→    </div>
   678→  );
   679→}

   681→// ── Ads tab ──────────────────────────────────────────────────────────────
   682→
   683→export function AdsTab({ apiKey, onChanged }: { apiKey: string; onChanged: () => void }) {
   684→  const { toast } = useToast();
   685→  const [campaigns, setCampaigns] = useState<AdCampaign[] | null>(null);
   686→  const [stats, setStats] = useState<AdsStats | null>(null);
   687→  const [measurement, setMeasurement] = useState<Measurement | null>(null);
   688→  const [error, setError] = useState(false);
   689→  const [busy, setBusy] = useState<string | null>(null);
   690→  const [dialog, setDialog] = useState<{ open: boolean; campaign: AdCampaign | null }>({
   691→    open: false,
   692→    campaign: null,
   693→  });
   694→  const [confirmDelete, setConfirmDelete] = useState<string | null>(null);
   695→  const deleteTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   696→
   697→  const load = useCallback(() => {
   698→    adminFetch(apiKey, "/api/admin/ads")
   699→      .then((r) => {
   700→        if (!r.ok) throw new Error(String(r.status));
   701→        return r.json() as Promise<{
   702→          campaigns: AdCampaign[];
   703→          stats: AdsStats;
   704→          measurement?: Measurement;
   705→        }>;
   706→      })
   707→      .then((d) => {
   708→        setCampaigns(d.campaigns);
   709→        setStats(d.stats);
   710→        setMeasurement(d.measurement ?? null);
   711→        setError(false);
   712→      })
   713→      .catch(() => setError(true));
   714→  }, [apiKey]);
   715→
   716→  useEffect(load, [load]);
   717→
   718→  const retry = useCallback(() => {
   719→    setCampaigns(null);
   720→    setError(false);
   721→    load();
   722→  }, [load]);
   723→
   724→  const patchStatus = useCallback(
   725→    async (c: AdCampaign, nextStatus: CampaignStatus, note?: string) => {
   726→      const snapshot = campaigns;
   727→      setCampaigns((cur) =>
   728→        cur?.map((x) => (x.id === c.id ? { ...x, status: nextStatus } : x)) ?? cur
   729→      );
   730→      setBusy(c.id);
   731→      try {
   732→        const res = await adminFetch(apiKey, `/api/admin/ads/${c.id}`, {
   733→          method: "PATCH",
   734→          body: JSON.stringify({ status: nextStatus }),
   735→        });
   736→        const d = (await res.json()) as { ok?: boolean; error?: string };
   737→        if (!res.ok || !d.ok) throw new Error(d.error ?? "Status update failed");
   738→        toast({ title: `${c.name} → ${nextStatus}`, description: note });
   739→        onChanged();
   740→      } catch (err) {
   741→        setCampaigns(snapshot);
   742→        toast({
   743→          title: err instanceof Error ? err.message : "Network error",
   744→          variant: "destructive",
   745→        });
   746→      } finally {
   747→        setBusy(null);
   748→      }
   749→    },
   750→    [apiKey, campaigns, toast, onChanged]
   751→  );
   752→
   753→  const doDelete = useCallback(
   754→    async (c: AdCampaign) => {
   755→      setBusy(c.id);
   756→      try {
   757→        const res = await adminFetch(apiKey, `/api/admin/ads/${c.id}`, { method: "DELETE" });
   758→        const d = (await res.json()) as { ok?: boolean; error?: string };
   759→        if (!res.ok || !d.ok) throw new Error(d.error ?? "Delete failed");
   760→        toast({ title: `${c.name} deleted` });
   761→        setCampaigns((cur) => cur?.filter((x) => x.id !== c.id) ?? cur);
   762→        onChanged();
   763→      } catch (err) {
   764→        toast({
   765→          title: err instanceof Error ? err.message : "Network error",
   766→          variant: "destructive",
   767→        });
   768→      } finally {
   769→        setBusy(null);
   770→      }
   771→    },
   772→    [apiKey, toast, onChanged]
   773→  );
   774→
   775→  const onDeleteClick = useCallback(
   776→    (c: AdCampaign) => {
   777→      if (confirmDelete === c.id) {
   778→        if (deleteTimer.current) clearTimeout(deleteTimer.current);
   779→        setConfirmDelete(null);
   780→        void doDelete(c);
   781→      } else {
   782→        setConfirmDelete(c.id);
   783→        if (deleteTimer.current) clearTimeout(deleteTimer.current);
   784→        deleteTimer.current = setTimeout(() => setConfirmDelete(null), 3000);
   785→      }
   786→    },
   787→    [confirmDelete, doDelete]
   788→  );
   789→
   790→  return (
   791→    <div className="space-y-3">
   792→      {/* stats strip */}
   793→      <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-4">
   794→        <MiniStat label="Active campaigns" value={stats ? `${stats.active}/${stats.total}` : "…"} />
   795→        <MiniStat label="Impressions" value={stats?.impressions ?? "…"} />
   796→        <MiniStat label="Clicks" value={stats?.clicks ?? "…"} />
   797→        <MiniStat label="Avg CTR" value={stats ? fmtCtr(stats.ctr) : "…"} accent />
   798→      </div>
   799→
   800→      {/* serving controls — master + per-placement kill switches (Task 27) */}
   801→      <ServingControls apiKey={apiKey} />
   802→
   803→      {/* 7-day fill/unfill accounting per placement (Task 28) */}
   804→      <MeasurementCard m={measurement} />
   805→
   806→      <div className="flex items-center justify-between">
   807→        <p className="font-mono text-[10px] tracking-wider text-white/35 uppercase">
   808→          {campaigns ? `${campaigns.length} campaign${campaigns.length === 1 ? "" : "s"}` : "…"} ·
   809→          weight picks the slot winner
   810→        </p>
   811→        <Button
   812→          size="sm"
   813→          onClick={() => setDialog({ open: true, campaign: null })}
   814→          className="rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   815→        >
   816→          <Plus className="size-4" aria-hidden />
   817→          New campaign
   818→        </Button>
   819→      </div>
   820→
   821→      {error ? (
   822→        <LoadError label="Campaigns" onRetry={retry} />
   823→      ) : campaigns === null ? (
   824→        <CampaignsSkeleton />
   825→      ) : campaigns.length === 0 ? (
   826→        <div className="rounded-xl border border-dashed border-white/15 bg-white/[0.02] p-10 text-center">
   827→          <Megaphone className="mx-auto size-8 text-white/25" aria-hidden />
   828→          <p className="mt-3 font-mono text-sm text-white/60">
   829→            No campaigns yet — create the first one
   830→          </p>
   831→          <Button
   832→            onClick={() => setDialog({ open: true, campaign: null })}
   833→            className="mt-4 rounded-lg bg-ember font-semibold text-black shadow-none hover:bg-ember-hot dark:text-black"
   834→          >
   835→            <Plus className="size-4" aria-hidden />
   836→            New campaign
   837→          </Button>
   838→        </div>
   839→      ) : (
   840→        <div className="max-h-[52vh] space-y-2 overflow-y-auto pr-1">
   841→          {campaigns.map((c) => (
   842→            <div
   843→              key={c.id}
   844→              className="rounded-xl border border-white/10 bg-white/[0.02] p-3.5"
   845→            >
   846→              <div className="flex flex-wrap items-center gap-x-3 gap-y-2">
   847→                <span aria-hidden className="text-lg">
   848→                  {c.emoji || "📢"}
   849→                </span>
   850→                <div className="min-w-44 flex-1">
   851→                  <p className="truncate text-sm font-bold text-white">{c.name}</p>
   852→                  <p className="truncate text-xs text-white/45">{c.headline}</p>
   853→                </div>
   854→
   855→                <div className="flex shrink-0 flex-wrap items-center gap-1.5">
   856→                  <span className="rounded-full border border-white/15 bg-white/5 px-2 py-0.5 font-mono text-[9px] tracking-wider text-white/60 uppercase">
   857→                    {PLACEMENT_LABEL[c.placement]}
   858→                  </span>
   859→                  {c.windowState !== "none" && (
   860→                    <span
   861→                      className={cn(
   862→                        "rounded-full border px-2 py-0.5 font-mono text-[9px] tracking-wider uppercase",
   863→                        WINDOW_CX[c.windowState]

   865→                    >
